13static int reflect_index(int index, int length)
14{
15 while (index < 0 || index >= length) {
16 if (index < 0) {
17 index = -index;
18 } else {
19 index = 2 * length - index - 2;
20 }
21 }
22 return index;
23}

25int audio_whisper_stft_power_reference_f32(
26 const float *samples,
27 int n_samples,
28 float *power,
29 int n_frames)
30{
31 if (samples == NULL || power == NULL) {
32 return -1;
33 }
34 if (n_samples <= CK_AUDIO_WHISPER_N_FFT / 2 || n_frames <= 0) {
35 return -2;
36 }
37 if (n_frames != n_samples / CK_AUDIO_WHISPER_HOP_LENGTH) {
38 return -3;
39 }
40
41 const int center = CK_AUDIO_WHISPER_N_FFT / 2;
42 for (int frame = 0; frame < n_frames; ++frame) {
43 for (int bin = 0; bin < CK_AUDIO_WHISPER_POWER_BINS; ++bin) {
44 float real = 0.0f;
45 float imag = 0.0f;
46 for (int sample = 0; sample < CK_AUDIO_WHISPER_N_FFT; ++sample) {
47 const int source = reflect_index(
48 frame * CK_AUDIO_WHISPER_HOP_LENGTH + sample - center,
49 n_samples);
50 const float window = 0.5f - 0.5f * cosf(
51 2.0f * CK_AUDIO_PI_F * (float)sample /
52 (float)CK_AUDIO_WHISPER_N_FFT);
53 const float value = samples[source] * window;
54 const float angle = -2.0f * CK_AUDIO_PI_F *
55 (float)(bin * sample) / (float)CK_AUDIO_WHISPER_N_FFT;
56 real = fmaf(value, cosf(angle), real);
57 imag = fmaf(value, sinf(angle), imag);
58 }
59 power[(size_t)frame * CK_AUDIO_WHISPER_POWER_BINS + bin] =
60 fmaf(real, real, imag * imag);
61 }
62 }
63 return 0;
64}

66int audio_whisper_log_mel_from_power_reference_f32(
67 const float *power,
68 const float *mel_filters,
69 int n_mels,
70 int n_frames,
71 float *log_mel)
72{
73 if (power == NULL || mel_filters == NULL || log_mel == NULL) {
74 return -1;
75 }
76 if (n_mels <= 0 || n_frames <= 0) {
77 return -2;
78 }
79
80 float maximum = -INFINITY;
81 for (int mel = 0; mel < n_mels; ++mel) {
82 const float *filter = mel_filters + (size_t)mel * CK_AUDIO_WHISPER_POWER_BINS;
83 float *output = log_mel + (size_t)mel * n_frames;
84 for (int frame = 0; frame < n_frames; ++frame) {
85 const float *spectrum = power + (size_t)frame * CK_AUDIO_WHISPER_POWER_BINS;
86 float sum = 0.0f;
87 for (int bin = 0; bin < CK_AUDIO_WHISPER_POWER_BINS; ++bin) {
88 sum = fmaf(filter[bin], spectrum[bin], sum);
89 }
90 const float value = log10f(fmaxf(sum, 1.0e-10f));
91 output[frame] = value;
92 maximum = fmaxf(maximum, value);
93 }
94 }
95
96 const float floor = maximum - 8.0f;
97 for (int mel = 0; mel < n_mels; ++mel) {
98 float *output = log_mel + (size_t)mel * n_frames;
99 for (int frame = 0; frame < n_frames; ++frame) {
100 output[frame] = (fmaxf(output[frame], floor) + 4.0f) / 4.0f;
101 }
102 }
103 return 0;
104}
