10static int rotate_rows(const float *input, size_t input_elements,
11 const float *cosine, const float *sine, size_t table_elements,
12 float *output, size_t output_elements, size_t tokens, size_t heads,
13 size_t head_dim, int bf16)
14{
15 size_t ni, nt;
16 if (!head_dim || (head_dim & 1) || !heads || heads > SIZE_MAX / head_dim ||
17 !ck_checked_span(tokens, heads * head_dim, heads * head_dim, &ni) ||
18 !ck_checked_span(tokens, head_dim / 2, head_dim / 2, &nt) ||
19 input_elements < ni || output_elements < ni || table_elements < nt) return -2;
20 if (!ck_checked_region(input, ni) || !ck_checked_region(output, ni) ||
21 !ck_checked_region(cosine, nt) || !ck_checked_region(sine, nt) ||
22 ck_checked_overlap(input, ni, output, ni) ||
23 ck_checked_overlap(cosine, nt, output, ni) ||
24 ck_checked_overlap(sine, nt, output, ni)) return -1;
25 if (!ck_checked_finite(input, ni) || !ck_checked_finite(cosine, nt) ||
26 !ck_checked_finite(sine, nt)) return -3;
27 for (int pass = 0; pass < 2; ++pass)
28 for (size_t t = 0; t < tokens; ++t)
29 for (size_t h = 0; h < heads; ++h)
30 for (size_t d = 0; d < head_dim; d += 2) {
31 size_t at = (t * heads + h) * head_dim + d;
32 size_t phase = t * (head_dim / 2) + d / 2;
33 float a = input[at], b = input[at + 1];
34 float c = cosine[phase], s = sine[phase];
35 if (bf16 && (rounded(a, 1) != a || rounded(b, 1) != b ||
36 rounded(c, 1) != c || rounded(s, 1) != s)) return -3;
37 float left = rounded(rounded(a * c, bf16) - rounded(b * s, bf16), bf16);
38 float right = rounded(rounded(b * c, bf16) + rounded(a * s, bf16), bf16);
39 if (!isfinite(left) || !isfinite(right)) return -3;
40 if (pass) { output[at] = left; output[at + 1] = right; }
41 }
42 return 0;
43}
