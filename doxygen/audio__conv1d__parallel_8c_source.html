24static void audio_conv1d_compute_parallel(const ck_audio_conv1d_job *job,
25 void *opaque)
26{
27 const ck_audio_conv1d_parallel_context *context = opaque;
28 const int channels = (int)job->output_channels;
29 int workers = context->workers;
30 if (workers > ck_threadpool_capacity(context->pool))
31 workers = ck_threadpool_capacity(context->pool);
32 if (workers > channels) workers = channels;
33 /* Dispatch overhead dominates tiny independent work. Both paths use the
34 * identical per-channel AVX2 implementation and ordered reduction. */
35 if (workers < 2 || channels < 8 || job->output_frames < 16) {
36 audio_conv1d_compute_avx2(job, NULL);
37 return;
38 }
39 const int grain = (channels + workers - 1) / workers;
40 ck_threadpool_parallel_for_n(context->pool, workers, 0, channels, grain,
41 audio_conv1d_channel_range, (void *)job);
42}
