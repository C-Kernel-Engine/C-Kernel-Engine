25/* Packed blocks are read in GGUF byte order and expanded inside the dot.
26 * The FP32 activation and ordered FP64 accumulation policy is shared. */
27static float expert_dot(const float *input, const unsigned char *weight,
28 size_t width, enum expert_storage storage)
29{
30 if (storage == EXPERT_Q8_0) return ck_packed_q8_dot_ordered(input, weight, width);
31 double sum = 0.0;
32 if (storage == EXPERT_F32) {
33 for (size_t i = 0; i < width; ++i) {
34 float value;
35 memcpy(&value, weight + i * sizeof(float), sizeof(float));
36 sum += (double)input[i] * (double)value;
37 }
38 } else {
39 const size_t block_bytes = packed_block_bytes(storage);
40 for (size_t i = 0; i < width; i += QK8_0, weight += block_bytes) {
41 const float scale = ck_fp16_to_fp32_soft(
42 (uint16_t)(weight[0] | ((uint16_t)weight[1] << 8)));
43 for (size_t j = 0; j < QK8_0; ++j) {
44 const unsigned pair = weight[2 + j % 16];
45 const float expanded = scale * (float)iq4_nl_values[j < 16 ? pair & 15 : pair >> 4];
46 sum += (double)input[i + j] * (double)expanded;
47 }
48 }
49 }
50 return (float)sum;
51}

72static int routed_softcapped_experts(ROUTED_EXPERT_ARGS, enum expert_storage storage)
73{
74 /* Bounds keep workspace arithmetic valid on supported size_t widths. */
75 if (!rows || rows > 16 || !hidden_dim || hidden_dim > 8192 ||
76 !inner || inner > 8192 || !experts || experts > 1024 || !slots ||
77 slots > experts || !top_k || top_k > experts ||
78 !isfinite(gate_cap) || gate_cap <= 0.0f || !isfinite(up_cap) || up_cap <= 0.0f)
79 return -1;
80 const int packed = storage != EXPERT_F32;
81 if (packed && (hidden_dim % QK8_0 || inner % QK8_0)) return -1;
82 const size_t block_bytes = packed ? packed_block_bytes(storage) : sizeof(float);
83 const size_t block_elements = packed ? QK8_0 : 1;
84 const size_t gate_row = hidden_dim / block_elements * block_bytes;
85 const size_t down_row = inner / block_elements * block_bytes;
86 const size_t expert_bytes = inner * gate_row;
87 const size_t route_count = rows * top_k;
88 const size_t output_elements = rows * hidden_dim;
89 const size_t row_outputs = route_count * hidden_dim;
90 const size_t scratch_elements = 3 * inner + row_outputs + output_elements;
91 if (hidden_elements < output_elements || id_elements < route_count || route_elements < route_count ||
92 expert_slot_elements < experts || expert_output_elements < row_outputs ||
93 output_capacity < output_elements || work_elements < scratch_elements) return -1;
94 if (expert_bytes != hidden_dim * down_row || expert_bytes > SIZE_MAX / slots ||
95 matrix_bytes != expert_bytes * slots) return -1;
96 const void *buffers[] = {hidden, ids, route_weights, expert_slots, gate, up, down,
97 expert_output, output, work};
98 const size_t sizes[] = {output_elements * sizeof(float), route_count * sizeof(int32_t),
99 route_count * sizeof(float), experts * sizeof(int32_t),
100 matrix_bytes, matrix_bytes, matrix_bytes, row_outputs * sizeof(float),
101 output_elements * sizeof(float),
102 scratch_elements * sizeof(float)};
103 for (size_t i = 0; i < sizeof(buffers) / sizeof(buffers[0]); ++i) {
104 if (!region(buffers[i], sizes[i])) return -1;
105 if (i < 4 || i >= 7) {
106 if ((uintptr_t)buffers[i] % _Alignof(float)) return -1;
107 }
108 if (i < 7) continue;
109 for (size_t j = 0; j < i; ++j)
110 if (contains_overlap(buffers[i], sizes[i], buffers[j], sizes[j])) return -1;
111 }
112 if (!ck_checked_finite(hidden, output_elements) ||
113 !ck_checked_finite(route_weights, route_count)) return -2;
114
115 unsigned char assigned[1024] = {0};
116 for (size_t id = 0; id < experts; ++id) {
117 const int32_t slot = expert_slots[id];
118 if (slot == -1) continue;
119 if (slot < 0 || (size_t)slot >= slots || assigned[slot]) return -2;
120 assigned[slot] = 1;
121 }
122 for (size_t row = 0; row < rows; ++row) {
123 for (size_t k = 0; k < top_k; ++k) {
124 const size_t route = row * top_k + k;
125 const int32_t id = ids[route];
126 if (id < 0 || (size_t)id >= experts || expert_slots[id] < 0 ||
127 (size_t)expert_slots[id] >= slots || route_weights[route] < 0.0f) return -2;
128 for (size_t prev = 0; prev < k; ++prev)
129 if (id == ids[row * top_k + prev]) return -2;
130 }
131 }
132
133 float *gate_values = work;
134 float *up_values = work + inner;
135 float *activation = work + 2 * inner;
136 float *staged_experts = work + 3 * inner;
137 float *staged_output = staged_experts + row_outputs;
138 memset(staged_output, 0, output_elements * sizeof(float));
139 for (size_t row = 0; row < rows; ++row) {
140 const float *x = hidden + row * hidden_dim;
141 float *y = staged_output + row * hidden_dim;
142 /* The reference sums contributions in ascending expert-ID order. */
143 for (size_t id = 0; id < experts; ++id) {
144 for (size_t index = 0; index < top_k; ++index) {
145 const size_t route = row * top_k + index;
146 if ((size_t)ids[route] != id) continue;
147 const size_t offset = (size_t)expert_slots[id] * expert_bytes;
148 const unsigned char *g = (const unsigned char *)gate + offset;
149 const unsigned char *u = (const unsigned char *)up + offset;
150 const unsigned char *d = (const unsigned char *)down + offset;
151 for (size_t i = 0; i < inner; ++i) {
152 gate_values[i] = expert_dot(x, g + i * gate_row, hidden_dim, storage);
153 up_values[i] = expert_dot(x, u + i * gate_row, hidden_dim, storage);
154 }
155 if (softcapped_swiglu_rows_checked_f32(gate_values, inner, up_values, inner,
156 activation, inner, inner, gate_cap, up_cap)) return -3;
157 for (size_t i = 0; i < hidden_dim; ++i) {
158 const float v = expert_dot(activation, d + i * down_row, inner, storage);
159 staged_experts[route * hidden_dim + i] = v;
160 y[i] += v * route_weights[route];
161 if (!isfinite(v) || !isfinite(y[i])) return -3;
162 }
163 }
164 }
165 }
166 memcpy(expert_output, staged_experts, row_outputs * sizeof(float));
167 memcpy(output, staged_output, output_elements * sizeof(float));
168 return 0;
169}
