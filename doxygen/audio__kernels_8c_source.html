13static int reflect_index(int index, int length)
14{
15 while (index < 0 || index >= length) {
16 if (index < 0) {
17 index = -index;
18 } else {
19 index = 2 * length - index - 2;
20 }
21 }
22 return index;
23}

25int audio_pcm_s16_to_mono_f32(
26 const int16_t *interleaved,
27 int n_frames,
28 int n_channels,
29 float *mono)
30{
31 if (interleaved == NULL || mono == NULL) {
32 return -1;
33 }
34 if (n_frames <= 0 || n_channels <= 0) {
35 return -2;
36 }
37 const float scale = 1.0f / 32768.0f;
38 for (int frame = 0; frame < n_frames; ++frame) {
39 float sum = 0.0f;
40 for (int channel = 0; channel < n_channels; ++channel) {
41 sum += (float)interleaved[(size_t)frame * n_channels + channel];
42 }
43 mono[frame] = (sum / (float)n_channels) * scale;
44 }
45 return 0;
46}

59int audio_resample_linear_f32(
60 const float *input,
61 int input_frames,
62 int input_rate,
63 float *output,
64 int output_frames,
65 int output_rate)
66{
67 if (input == NULL || output == NULL) {
68 return -1;
69 }
70 const int expected = audio_resampled_frame_count(input_frames, input_rate, output_rate);
71 if (expected <= 0 || output_frames != expected) {
72 return -2;
73 }
74 for (int frame = 0; frame < output_frames; ++frame) {
75 const long long numerator = (long long)frame * input_rate;
76 const int left = (int)(numerator / output_rate);
77 const int right = left + 1 < input_frames ? left + 1 : left;
78 const float fraction = (float)(numerator % output_rate) / (float)output_rate;
79 output[frame] = fmaf(input[right] - input[left], fraction, input[left]);
80 }
81 return 0;
82}

84int audio_stft_precompute_tables_f32(
85 int n_fft,
86 float *window,
87 float *cos_table,
88 float *sin_table)
89{
90 if (window == NULL || cos_table == NULL || sin_table == NULL) {
91 return -1;
92 }
93 if (n_fft <= 0 || (n_fft & 1) != 0) {
94 return -2;
95 }
96 const int bins = n_fft / 2 + 1;
97 for (int sample = 0; sample < n_fft; ++sample) {
98 window[sample] = 0.5f - 0.5f * cosf(
99 2.0f * CK_AUDIO_PI_F * (float)sample / (float)n_fft);
100 }
101 for (int bin = 0; bin < bins; ++bin) {
102 for (int sample = 0; sample < n_fft; ++sample) {
103 const float angle = -2.0f * CK_AUDIO_PI_F *
104 (float)(bin * sample) / (float)n_fft;
105 const size_t index = (size_t)bin * n_fft + sample;
106 cos_table[index] = cosf(angle);
107 sin_table[index] = sinf(angle);
108 }
109 }
110 return 0;
111}

113int audio_stft_power_precomputed_f32(
114 const float *samples,
115 int n_samples,
116 const float *window,
117 const float *cos_table,
118 const float *sin_table,
119 int n_fft,
120 int hop_length,
121 float *power,
122 int n_frames)
123{
124 if (samples == NULL || window == NULL || cos_table == NULL ||
125 sin_table == NULL || power == NULL) {
126 return -1;
127 }
128 if (n_fft <= 0 || hop_length <= 0 || n_samples <= n_fft / 2 ||
129 (n_fft & 1) != 0 || n_frames <= 0) {
130 return -2;
131 }
132 if (n_frames != n_samples / hop_length) {
133 return -3;
134 }
135 const int bins = n_fft / 2 + 1;
136 const int center = n_fft / 2;
137 for (int frame = 0; frame < n_frames; ++frame) {
138 for (int bin = 0; bin < bins; ++bin) {
139 const float *cos_row = cos_table + (size_t)bin * n_fft;
140 const float *sin_row = sin_table + (size_t)bin * n_fft;
141 float real = 0.0f;
142 float imag = 0.0f;
143 for (int sample = 0; sample < n_fft; ++sample) {
144 const int source = reflect_index(
145 frame * hop_length + sample - center, n_samples);
146 const float value = samples[source] * window[sample];
147 real = fmaf(value, cos_row[sample], real);
148 imag = fmaf(value, sin_row[sample], imag);
149 }
150 power[(size_t)frame * bins + bin] =
151 fmaf(real, real, imag * imag);
152 }
153 }
154 return 0;
155}

157int audio_conv1d_channel_major_f32(
158 const float *input,
159 const float *weight,
160 const float *bias,
161 float *output,
162 int input_channels,
163 int output_channels,
164 int input_frames,
165 int kernel_size,
166 int stride,
167 int padding,
168 int output_frames)
169{
170 if (input == NULL || weight == NULL || output == NULL) {
171 return -1;
172 }
173 if (input_channels <= 0 || output_channels <= 0 || input_frames <= 0 ||
174 kernel_size <= 0 || stride <= 0 || padding < 0 || output_frames <= 0) {
175 return -2;
176 }
177 const int expected = (input_frames + 2 * padding - kernel_size) / stride + 1;
178 if (output_frames != expected) {
179 return -3;
180 }
181 for (int out_channel = 0; out_channel < output_channels; ++out_channel) {
182 const float *weight_channel = weight +
183 (size_t)out_channel * input_channels * kernel_size;
184 float *output_channel = output + (size_t)out_channel * output_frames;
185 for (int out_frame = 0; out_frame < output_frames; ++out_frame) {
186 float sum = bias != NULL ? bias[out_channel] : 0.0f;
187 for (int in_channel = 0; in_channel < input_channels; ++in_channel) {
188 const float *input_channel = input + (size_t)in_channel * input_frames;
189 const float *weight_row = weight_channel + (size_t)in_channel * kernel_size;
190 for (int kernel = 0; kernel < kernel_size; ++kernel) {
191 const int in_frame = out_frame * stride + kernel - padding;
192 if (in_frame >= 0 && in_frame < input_frames) {
193 sum = fmaf(input_channel[in_frame], weight_row[kernel], sum);
194 }
195 }
196 }
197 output_channel[out_frame] = sum;
198 }
199 }
200 return 0;
201}

224int audio_whisper_stft_power_reference_f32(
225 const float *samples,
226 int n_samples,
227 float *power,
228 int n_frames)
229{
230 if (samples == NULL || power == NULL) {
231 return -1;
232 }
233 if (n_samples <= CK_AUDIO_WHISPER_N_FFT / 2 || n_frames <= 0) {
234 return -2;
235 }
236 if (n_frames != n_samples / CK_AUDIO_WHISPER_HOP_LENGTH) {
237 return -3;
238 }
239
240 const int center = CK_AUDIO_WHISPER_N_FFT / 2;
241 for (int frame = 0; frame < n_frames; ++frame) {
242 for (int bin = 0; bin < CK_AUDIO_WHISPER_POWER_BINS; ++bin) {
243 float real = 0.0f;
244 float imag = 0.0f;
245 for (int sample = 0; sample < CK_AUDIO_WHISPER_N_FFT; ++sample) {
246 const int source = reflect_index(
247 frame * CK_AUDIO_WHISPER_HOP_LENGTH + sample - center,
248 n_samples);
249 const float window = 0.5f - 0.5f * cosf(
250 2.0f * CK_AUDIO_PI_F * (float)sample /
251 (float)CK_AUDIO_WHISPER_N_FFT);
252 const float value = samples[source] * window;
253 const float angle = -2.0f * CK_AUDIO_PI_F *
254 (float)(bin * sample) / (float)CK_AUDIO_WHISPER_N_FFT;
255 real = fmaf(value, cosf(angle), real);
256 imag = fmaf(value, sinf(angle), imag);
257 }
258 power[(size_t)frame * CK_AUDIO_WHISPER_POWER_BINS + bin] =
259 fmaf(real, real, imag * imag);
260 }
261 }
262 return 0;
263}

265int audio_whisper_log_mel_from_power_reference_f32(
266 const float *power,
267 const float *mel_filters,
268 int n_mels,
269 int n_frames,
270 float *log_mel)
271{
272 if (power == NULL || mel_filters == NULL || log_mel == NULL) {
273 return -1;
274 }
275 if (n_mels <= 0 || n_frames <= 0) {
276 return -2;
277 }
278
279 float maximum = -INFINITY;
280 for (int mel = 0; mel < n_mels; ++mel) {
281 const float *filter = mel_filters + (size_t)mel * CK_AUDIO_WHISPER_POWER_BINS;
282 float *output = log_mel + (size_t)mel * n_frames;
283 for (int frame = 0; frame < n_frames; ++frame) {
284 const float *spectrum = power + (size_t)frame * CK_AUDIO_WHISPER_POWER_BINS;
285 float sum = 0.0f;
286 for (int bin = 0; bin < CK_AUDIO_WHISPER_POWER_BINS; ++bin) {
287 sum = fmaf(filter[bin], spectrum[bin], sum);
288 }
289 const float value = log10f(fmaxf(sum, 1.0e-10f));
290 output[frame] = value;
291 maximum = fmaxf(maximum, value);
292 }
293 }
294
295 const float floor = maximum - 8.0f;
296 for (int mel = 0; mel < n_mels; ++mel) {
297 float *output = log_mel + (size_t)mel * n_frames;
298 for (int frame = 0; frame < n_frames; ++frame) {
299 output[frame] = (fmaxf(output[frame], floor) + 4.0f) / 4.0f;
300 }
301 }
302 return 0;
303}
