11int rmsnorm_rows_checked_f32(const float *input, size_t input_elements,

20 input_elements < ni || output_elements < no || gamma_elements < width)
21 return -2;

23 !ck_checked_region(input, ni) || !ck_checked_region(gamma, width) ||
24 !ck_checked_region(output, no) ||

27 if (!ck_checked_finite(gamma, width)) return -3;

29 const float *x = input + row * input_stride;
30 if (!ck_checked_finite(x, width)) return -3;
31 float sum = 0;
32 for (size_t col = 0; col < width; ++col) sum += x[col] * x[col];
33 const float scale = 1.0f / sqrtf(sum / (float)width + eps);
34 if (!isfinite(scale)) return -3;

36 if (!isfinite(x[col] * scale * gamma[col])) return -3;

40 float *y = output + row * output_stride;
41 float sum = 0;

43 const float scale = 1.0f / sqrtf(sum / (float)width + eps);
44 for (size_t col = 0; col < width; ++col)
45 y[col] = x[col] * scale * gamma[col];
46 }
47 return 0;

162 return -2;

164 const size_t nk = kv_heads * tokens * head_dim;

167 !ck_checked_region(q, nq) || !ck_checked_region(k, nk) ||

169 /* The unchecked provider evaluates pos_offset + token as signed int and
170 * computes theta with powf before writing in place. Prove every angle is
171 * representable first so an invalid frequency cannot partially rotate Q/K.
172 */
173 const float theta_scale = powf(freq_base, -2.0f / (float)head_dim);
174 const float last_position = (float)(pos_offset + tokens - 1);
175 if (!isfinite(theta_scale) || theta_scale <= 0.0f) return -3;
176 for (size_t index = 0; index < head_dim / 2; ++index) {
177 const float factor = powf(theta_scale, (float)index);
178 const float angle = last_position * factor;
179 if (!isfinite(factor) || !isfinite(angle) ||
180 !isfinite(cosf(angle)) || !isfinite(sinf(angle))) return -3;
181 }
182 for (size_t i = 0; i < nq; ++i)
183 if (!isfinite(q[i]) || fabsf(q[i]) > FLT_MAX / 4) return -3;
184 for (size_t i = 0; i < nk; ++i)
185 if (!isfinite(k[i]) || fabsf(k[i]) > FLT_MAX / 4) return -3;
186 rope_forward_qk_split_direct_f32(q, k, NULL, 0, (int)heads,
187 (int)kv_heads, (int)tokens, (int)head_dim, (int)head_dim,
188 (int)pos_offset, (int)head_dim, freq_base);

231int attention_sliding_bounded_checked_f32(
232 const float *q, size_t q_elements,
233 const float *k, size_t k_elements,
234 const float *v, size_t v_elements,
235 float *work, size_t work_elements,
236 float *output, size_t output_elements,
237 size_t heads, size_t kv_heads, size_t tokens, size_t head_dim,
238 size_t sliding_window)
239{
240 if (!heads || !kv_heads || heads < kv_heads || heads % kv_heads ||
241 !tokens || !head_dim || !sliding_window ||
242 heads > INT_MAX || kv_heads > INT_MAX || tokens > INT_MAX ||
243 head_dim > INT_MAX || sliding_window > INT_MAX ||
244 heads > SIZE_MAX / tokens || kv_heads > SIZE_MAX / tokens ||
245 heads * tokens > SIZE_MAX / head_dim ||
246 kv_heads * tokens > SIZE_MAX / head_dim)
247 return -2;
248 const size_t nq = heads * tokens * head_dim;
249 const size_t nk = kv_heads * tokens * head_dim;
250 if (q_elements < nq || k_elements < nk || v_elements < nk ||
251 work_elements < nq || output_elements < nq) return -2;
252 if (!ck_checked_region(q, nq) || !ck_checked_region(k, nk) ||
253 !ck_checked_region(v, nk) || !ck_checked_region(work, nq) ||
254 !ck_checked_region(output, nq) ||
255 ck_checked_overlap(q, nq, work, nq) ||
256 ck_checked_overlap(k, nk, work, nq) ||
257 ck_checked_overlap(v, nk, work, nq) ||
258 ck_checked_overlap(work, nq, output, nq) ||
259 ck_checked_overlap(q, nq, output, nq) ||
260 ck_checked_overlap(k, nk, output, nq) ||
261 ck_checked_overlap(v, nk, output, nq)) return -1;
262 if (!ck_checked_finite(q, nq) || !ck_checked_finite(k, nk) ||
263 !ck_checked_finite(v, nk)) return -3;
264
265 const size_t group = heads / kv_heads;
266 const float scale = 1.0f / sqrtf((float)head_dim);
267 for (size_t head = 0; head < heads; ++head) {
268 const size_t kv_head = head / group;
269 for (size_t row = 0; row < tokens; ++row) {
270 const float *query = q + (head * tokens + row) * head_dim;
271 float *acc = work + (head * tokens + row) * head_dim;
272 memset(acc, 0, head_dim * sizeof(float));
273 float maximum = -INFINITY;
274 float sum = 0.0f;
275 const size_t begin = row + 1 > sliding_window
276 ? row + 1 - sliding_window : 0;
277 for (size_t col = begin; col <= row; ++col) {
278 const float *key = k + (kv_head * tokens + col) * head_dim;
279 const float *value = v + (kv_head * tokens + col) * head_dim;
280 float score = 0.0f;
281 for (size_t dim = 0; dim < head_dim; ++dim)
282 score += query[dim] * key[dim];
283 score *= scale;
284 if (!isfinite(score)) return -3;
285 if (score > maximum) {
286 const float factor = maximum == -INFINITY
287 ? 0.0f : expf(maximum - score);
288 sum = sum * factor + 1.0f;
289 for (size_t dim = 0; dim < head_dim; ++dim)
290 acc[dim] = acc[dim] * factor + value[dim];
291 maximum = score;
292 } else {
293 const float factor = expf(score - maximum);
294 sum += factor;
295 for (size_t dim = 0; dim < head_dim; ++dim)
296 acc[dim] += factor * value[dim];
297 }
298 if (!isfinite(sum) || !ck_checked_finite(acc, head_dim))
299 return -3;
300 }
301 if (sum <= 0.0f || !isfinite(sum)) return -3;
302 const float inv_sum = 1.0f / sum;
303 for (size_t dim = 0; dim < head_dim; ++dim)
304 acc[dim] *= inv_sum;
305 if (!ck_checked_finite(acc, head_dim)) return -3;
306 }
307 }
308 memcpy(output, work, nq * sizeof(float));
309 return 0;
310}
