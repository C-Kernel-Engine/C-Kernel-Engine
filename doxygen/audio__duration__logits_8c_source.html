7static float sigmoid_f32(float x) {
8 if (x >= 0.0f) return 1.0f / (1.0f + expf(-x));
9 const float e = expf(x);
10 return e / (1.0f + e);
11}

13static int token_frames(const float *logits, size_t bins, float speed,
14 size_t max_duration_per_token, int32_t *frames) {
15 float sum = 0.0f;
16 for (size_t bin = 0; bin < bins; ++bin) {
17 const float value = logits[bin];
18 if (!isfinite(value)) return CK_AUDIO_EXTENT_INVALID;
19 sum += sigmoid_f32(value);
20 }
21 const float scaled = sum / speed;
22 if (!isfinite(scaled) || scaled >= 2147483648.0f)
23 return CK_AUDIO_EXTENT_LIMIT;
24
25 /* Nonnegative scaled values only. Implement nearest-even independently of
26 * the process floating-point rounding mode. */
27 const float lower = floorf(scaled);
28 const float fraction = scaled - lower;
29 float rounded = lower;
30 if (fraction > 0.5f ||
31 (fraction == 0.5f && fmodf(lower, 2.0f) != 0.0f))
32 rounded += 1.0f;
33 if (rounded < 1.0f) rounded = 1.0f;
34 /* rounded is below 2^31, so the integer conversion is exact and safe.
35 * Compare integers: converting a large limit to float can round it up. */
36 const int32_t candidate = (int32_t)rounded;
37 if ((size_t)candidate > max_duration_per_token)
38 return CK_AUDIO_EXTENT_LIMIT;
39 *frames = candidate;
40 return CK_AUDIO_EXTENT_OK;
41}
