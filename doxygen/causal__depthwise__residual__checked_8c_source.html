5/* Ordered FP64 arithmetic on FP32 input storage. Weight taps and history run
6 * oldest first. The history contains pre-convolution rows, never outputs. */
7static double evaluate(const float *input, const float *history, const float *weight,
8 size_t token, size_t channel, size_t channels, size_t taps,
9 double divisor)
10{
11 const float *filter = weight + channel * taps;
12 double sum = 0.0;
13 for (size_t tap = 0; tap < taps; ++tap) {
14 const size_t row = token + tap;
15 const float value = row < taps - 1
16 ? history[row * channels + channel]
17 : input[(row - (taps - 1)) * channels + channel];
18 /* Keep the same multiply and add sequence in every build. */
19 const volatile double product = (double)value * (double)filter[tap];
20 sum += product;
21 }
22 return (sum + (double)input[token * channels + channel]) / divisor;
23}

45static void store_history(float *history, const float *input, size_t tokens, size_t taps,
46 size_t rows, size_t state)
47{
48 if (tokens >= taps - 1) {
49 memcpy(history, input + rows - state, state * sizeof(float));
50 } else {
51 memmove(history, history + rows, (state - rows) * sizeof(float));
52 memcpy(history + state - rows, input, rows * sizeof(float));
53 }
54}

116/* GGUF static taps are oldest first; per-row dynamic taps start at lag zero. */
117static double evaluate_dynamic(const float *input, const float *history, const float *weight,
118 const float *dynamic, size_t token, size_t channel, size_t channels,
119 size_t heads, size_t taps, double divisor)
120{
121 const float *filter = weight + channel * taps;
122 const size_t head = channel / (channels / heads);
123 const float *delta = dynamic + (token * heads + head) * taps;
124 double static_sum = 0.0, dynamic_sum = 0.0;
125 for (size_t tap = 0; tap < taps; ++tap) {
126 const size_t row = token + tap;
127 const float value = row < taps - 1
128 ? history[row * channels + channel]
129 : input[(row - (taps - 1)) * channels + channel];
130 const volatile double product = (double)value * (double)filter[tap];
131 static_sum += product;
132 }
133 for (size_t lag = 0; lag < taps; ++lag) {
134 const size_t row = token + taps - 1 - lag;
135 const float value = row < taps - 1
136 ? history[row * channels + channel]
137 : input[(row - (taps - 1)) * channels + channel];
138 const volatile double product = (double)value * (double)delta[lag];
139 dynamic_sum += product;
140 }
141 return (static_sum + dynamic_sum + (double)input[token * channels + channel]) / divisor;
142}
