17static inline float recurrent_softplus(float x) {
18 if (x > 20.0f) {
19 return x;
20 }
21 if (x < -20.0f) {
22 return expf(x);
23 }
24 return logf(1.0f + expf(x));
25}

27static inline float recurrent_sigmoid(float x) {
28 if (x >= 0.0f) {
29 float z = expf(-x);
30 return 1.0f / (1.0f + z);
31 }
32 {
33 float z = expf(x);
34 return z / (1.0f + z);
35 }
36}

38void recurrent_dt_gate_forward(const float *alpha,
39 const float *dt_bias,
40 const float *a,
41 float *gate,
42 int rows,
43 int num_heads,
44 int state_dim) {
45 const int dim = num_heads * state_dim;
46 for (int row = 0; row < rows; ++row) {
47 const float *alpha_row = alpha + (size_t) row * (size_t) dim;
48 float *gate_row = gate + (size_t) row * (size_t) dim;
49 for (int col = 0; col < dim; ++col) {
50 const float x = alpha_row[col] + dt_bias[col];
51 gate_row[col] = recurrent_softplus(x) * a[col];
52 }
53 }
54}

56void recurrent_dt_gate_expanded_forward(const float *alpha,
57 const float *dt_bias,
58 const float *a,
59 float *gate,
60 int rows,
61 int num_heads,
62 int state_dim) {
63 for (int row = 0; row < rows; ++row) {
64 const float *alpha_row = alpha + (size_t) row * (size_t) num_heads;
65 float *gate_row = gate + (size_t) row * (size_t) num_heads * (size_t) state_dim;
66 for (int h = 0; h < num_heads; ++h) {
67 const float sp = recurrent_softplus(alpha_row[h] + dt_bias[h]);
68 const float *a_head = a + (size_t) h * (size_t) state_dim;
69 float *gate_head = gate_row + (size_t) h * (size_t) state_dim;
70 for (int col = 0; col < state_dim; ++col) {
71 gate_head[col] = sp * a_head[col];
72 }
73 }
74 }

77void recurrent_dt_gate_backward(const float *d_gate,
78 const float *alpha,
79 const float *dt_bias,
80 const float *a,
81 float *d_alpha,
82 float *d_dt_bias,
83 float *d_a,
84 int rows,
85 int dim) {
86 for (int col = 0; col < dim; ++col) {
87 d_dt_bias[col] = 0.0f;
88 d_a[col] = 0.0f;
89 }
90
91 for (int row = 0; row < rows; ++row) {
92 const float *d_gate_row = d_gate + (size_t) row * (size_t) dim;
93 const float *alpha_row = alpha + (size_t) row * (size_t) dim;
94 float *d_alpha_row = d_alpha + (size_t) row * (size_t) dim;
95 for (int col = 0; col < dim; ++col) {
96 const float x = alpha_row[col] + dt_bias[col];
97 const float sp = recurrent_softplus(x);
98 const float sig = recurrent_sigmoid(x);
99 const float d_out = d_gate_row[col];
100 d_a[col] += d_out * sp;
101 {
102 const float d_x = d_out * a[col] * sig;
103 d_alpha_row[col] = d_x;
104 d_dt_bias[col] += d_x;
105 }
106 }
107 }

146void recurrent_silu_forward_pytorch_bf16_storage(const float *x,
147 float *out,
148 int rows,
149 int dim)
150{
151 if (!x || !out || rows < 0 || dim < 0) {
152 fprintf(stderr, "HARD KERNEL CONTRACT FAULT: invalid PyTorch BF16 recurrent SiLU arguments\n");
153 abort();
154 }
155#if defined(__AVX512F__)
156 pthread_once(&ck_recurrent_sleef_once, ck_bind_recurrent_pytorch_sleef);
157 if (!ck_recurrent_pytorch_expf16) {
158 fprintf(stderr,
159 "HARD KERNEL CONTRACT FAULT: PyTorch BF16 recurrent SiLU requires "
160 "SLEEF Sleef_expf16_u10; set CK_SLEEF_LIBRARY\n");
161 abort();
162 }
163#endif
164 for (int row = 0; row < rows; ++row) {
165 const float *src = x + (size_t)row * (size_t)dim;
166 float *dst = out + (size_t)row * (size_t)dim;
167 int col = 0;
168#if defined(__AVX512F__)
169 for (; col + 16 <= dim; col += 16) {
170 float lanes[16] __attribute__((aligned(64)));
171 for (int lane = 0; lane < 16; ++lane) {
172 lanes[lane] = bf16_to_float(float_to_bf16(src[col + lane]));
173 }
174 const __m512 values = _mm512_load_ps(lanes);
175 const __m512 denominator = _mm512_add_ps(
176 _mm512_set1_ps(1.0f),
177 ck_recurrent_pytorch_expf16(
178 _mm512_sub_ps(_mm512_setzero_ps(), values)));
179 _mm512_store_ps(lanes, _mm512_div_ps(values, denominator));
180 for (int lane = 0; lane < 16; ++lane) {
181 dst[col + lane] = bf16_to_float(float_to_bf16(lanes[lane]));
182 }
183 }
184#endif
185 for (; col < dim; ++col) {
186 const float value = bf16_to_float(float_to_bf16(src[col]));
187 const float silu = value / (1.0f + expf(-value));
188 dst[col] = bf16_to_float(float_to_bf16(silu));
189 }
190 }
191}

234/* Match ggml's AVX2 exp approximation exactly; it is part of the provider ABI. */
235static inline __m256 recurrent_ggml_expf_avx2(__m256 x) {
236 const __m256 r = _mm256_set1_ps(0x1.8p23f);
237 const __m256 z = _mm256_fmadd_ps(x, _mm256_set1_ps(0x1.715476p+0f), r);
238 const __m256 n = _mm256_sub_ps(z, r);
239 const __m256 b = _mm256_fnmadd_ps(
240 n, _mm256_set1_ps(0x1.7f7d1cp-20f),
241 _mm256_fnmadd_ps(n, _mm256_set1_ps(0x1.62e4p-1f), x));
242 const __m256i e = _mm256_slli_epi32(_mm256_castps_si256(z), 23);
243 const __m256 k = _mm256_castsi256_ps(
244 _mm256_add_epi32(e, _mm256_castps_si256(_mm256_set1_ps(1.0f))));
245 const __m256i c = _mm256_castps_si256(
246 _mm256_cmp_ps(_mm256_andnot_ps(_mm256_set1_ps(-0.0f), n),
247 _mm256_set1_ps(126.0f), _CMP_GT_OQ));
248 const __m256 u = _mm256_mul_ps(b, b);
249 const __m256 j = _mm256_fmadd_ps(
250 _mm256_fmadd_ps(
251 _mm256_fmadd_ps(_mm256_set1_ps(0x1.0e4020p-7f), b,
252 _mm256_set1_ps(0x1.573e2ep-5f)),
253 u,
254 _mm256_fmadd_ps(_mm256_set1_ps(0x1.555e66p-3f), b,
255 _mm256_set1_ps(0x1.fffdb6p-2f))),
256 u, _mm256_mul_ps(_mm256_set1_ps(0x1.ffffecp-1f), b));
257 if (!_mm256_movemask_ps(_mm256_castsi256_ps(c))) {
258 return _mm256_fmadd_ps(j, k, k);
259 }
260 const __m256i g = _mm256_and_si256(
261 _mm256_castps_si256(_mm256_cmp_ps(n, _mm256_setzero_ps(), _CMP_LE_OQ)),
262 _mm256_set1_epi32((int) 0x82000000u));
263 const __m256 s1 = _mm256_castsi256_ps(
264 _mm256_add_epi32(g, _mm256_set1_epi32(0x7f000000)));
265 const __m256 s2 = _mm256_castsi256_ps(_mm256_sub_epi32(e, g));
266 const __m256i d = _mm256_castps_si256(
267 _mm256_cmp_ps(_mm256_andnot_ps(_mm256_set1_ps(-0.0f), n),
268 _mm256_set1_ps(192.0f), _CMP_GT_OQ));
269 return _mm256_or_ps(
270 _mm256_and_ps(_mm256_castsi256_ps(d), _mm256_mul_ps(s1, s1)),
271 _mm256_andnot_ps(
272 _mm256_castsi256_ps(d),
273 _mm256_or_ps(
274 _mm256_and_ps(
275 _mm256_castsi256_ps(c),
276 _mm256_mul_ps(_mm256_fmadd_ps(s2, j, s2), s1)),
277 _mm256_andnot_ps(
278 _mm256_castsi256_ps(c), _mm256_fmadd_ps(k, j, k)))));
279}
280#endif

282void recurrent_silu_forward_ggml(const float *x,
283 float *out,
284 int rows,
285 int dim) {
286 for (int row = 0; row < rows; ++row) {
287 const float *x_row = x + (size_t) row * (size_t) dim;
288 float *out_row = out + (size_t) row * (size_t) dim;
289 int col = 0;
290#if defined(__AVX2__) && defined(__FMA__)
291 for (; col + 8 <= dim; col += 8) {
292 const __m256 xv = _mm256_loadu_ps(x_row + col);
293 const __m256 neg = _mm256_sub_ps(_mm256_setzero_ps(), xv);
294 const __m256 denom = _mm256_add_ps(
295 _mm256_set1_ps(1.0f), recurrent_ggml_expf_avx2(neg));
296 _mm256_storeu_ps(out_row + col, _mm256_div_ps(xv, denom));
297 }
298#endif
299 for (; col < dim; ++col) {
300 const float xv = x_row[col];
301 out_row[col] = xv / (1.0f + expf(-xv));
302 }
303 }

306void recurrent_silu_backward(const float *d_out,
307 const float *x,
308 float *d_x,
309 int rows,
310 int dim) {
311 for (int row = 0; row < rows; ++row) {
312 const float *d_out_row = d_out + (size_t) row * (size_t) dim;
313 const float *x_row = x + (size_t) row * (size_t) dim;
314 float *d_x_row = d_x + (size_t) row * (size_t) dim;
315 for (int col = 0; col < dim; ++col) {
316 const float xv = x_row[col];
317 const float sig = recurrent_sigmoid(xv);
318 d_x_row[col] = d_out_row[col] * (sig + xv * sig * (1.0f - sig));
319 }
320 }
