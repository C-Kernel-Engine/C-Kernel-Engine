14static int valid_span(tensor_span s)
15{

20static int overlap(tensor_span a, tensor_span b)
21{
22 return (uintptr_t)a.ptr < (uintptr_t)b.ptr + b.elements * b.width &&
23 (uintptr_t)b.ptr < (uintptr_t)a.ptr + a.elements * a.width;

26static int checked_spans(const tensor_span *spans, size_t reads, size_t count)
27{
28 for (size_t i = 0; i < count; ++i) {
29 if (!valid_span(spans[i])) return 0;
30 if (i >= reads)
31 for (size_t j = 0; j < i; ++j)
32 if (overlap(spans[i], spans[j])) return 0;
33 }
34 return 1;
35}

37int ck_diffusion_vp_condition_inputs_checked(
38 const double *latent, const double *gamma, const float *self_condition,
39 const float *frequencies, float *normalized, float *prior_input,
40 float *scaled_self, float *time_features, size_t rows, size_t width,
41 size_t bands, size_t elements, size_t time_elements, size_t gamma_elements)
42{
43 if (!rows || !width || !bands || bands > SIZE_MAX / 2 || rows > SIZE_MAX / width ||
44 rows > SIZE_MAX / (2 * bands) || elements != rows * width ||
45 time_elements != rows * 2 * bands || gamma_elements != 2) return -1;
46 const tensor_span spans[] = {{latent,elements,8}, {gamma,2,8}, {self_condition,elements,4},
47 {frequencies,bands,4}, {normalized,elements,4}, {prior_input,elements,4},
48 {scaled_self,elements,4}, {time_features,time_elements,4}};
49 if (!checked_spans(spans, 4, sizeof spans / sizeof spans[0]) ||
50 !isfinite(gamma[0]) || !isfinite(gamma[1]) || gamma[1] > gamma[0]) return -1;
51 const double alpha_sq = vp_sigmoid(-gamma[0]), sigma_sq = vp_sigmoid(gamma[0]);
52 const double denominator = sqrt(alpha_sq / (double)width + sigma_sq);
53 const double ratio = sqrt(alpha_sq) / sigma_sq;
54 if (!isfinite(ratio) || ratio > FLT_MAX || fabs(gamma[0]) > FLT_MAX || !(denominator > 0.0))
55 return -1;
56 const float prior_gain = (float)ratio;
57 const float self_gain = sqrtf((float)width);
58 for (size_t i = 0; i < elements; ++i) {
59 if (!isfinite(latent[i]) || fabs(latent[i]) > FLT_MAX ||
60 fabs(latent[i] / denominator) > FLT_MAX ||
61 !isfinite((float)latent[i] * prior_gain) || !isfinite(self_condition[i] * self_gain))
62 return -1;
63 }
64 for (size_t i = 0; i < bands; ++i)
65 if (!isfinite((float)gamma[0] * frequencies[i])) return -1;
66
67 for (size_t i = 0; i < elements; ++i) {
68 normalized[i] = (float)(latent[i] / denominator);
69 prior_input[i] = (float)latent[i] * prior_gain;
70 scaled_self[i] = self_condition[i] * self_gain;
71 }
72 for (size_t i = 0; i < bands; ++i) {
73 const float angle = (float)gamma[0] * frequencies[i];
74 const float sine = sinf(angle), cosine = cosf(angle);
75 for (size_t r = 0; r < rows; ++r) {
76 time_features[r * 2 * bands + i] = sine;
77 time_features[r * 2 * bands + bands + i] = cosine;
78 }
79 }
80 return 0;
81}

85 float *logits, float *mean, int32_t *ids, float *probabilities,

91 scores_elements != rows * vocabulary || embedding_elements != vocabulary * width ||
92 mean_elements != rows * width) return -1;
93 const tensor_span spans[] = {{model_scores,scores_elements,2}, {prior_scores,scores_elements,4},
94 {embedding,embedding_elements,4}, {logits,scores_elements,4}, {mean,mean_elements,4},
95 {ids,rows,4}, {probabilities,vocabulary,4}};
96 if (!checked_spans(spans, 3, sizeof spans / sizeof spans[0])) return -1;
97 for (size_t i = 0; i < embedding_elements; ++i) {
98 /* The table contains prepared unit-vector coordinates. */
99 if (!isfinite(embedding[i]) || fabsf(embedding[i]) > 1.0f) return -1;
100 }
101 for (size_t i = 0; i < scores_elements; ++i)
102 if (!isfinite(bf16_to_float(model_scores[i]) + prior_scores[i])) return -1;
103
104 for (size_t r = 0; r < rows; ++r) {
105 const size_t start = r * vocabulary;
106 float largest = -INFINITY;
107 size_t best = 0;
108 for (size_t i = 0; i < vocabulary; ++i) {
109 const float value = bf16_to_float(model_scores[start + i]) + prior_scores[start + i];
110 logits[start + i] = value;
111 if (value > largest) {
112 largest = value;
113 best = i;
114 }
115 }
116 double denominator = 0.0;
117 for (size_t i = 0; i < vocabulary; ++i) {
118 probabilities[i] = expf(logits[start + i] - largest);
119 denominator += probabilities[i];
120 }
121 for (size_t i = 0; i < vocabulary; ++i)
122 probabilities[i] = (float)(probabilities[i] / denominator);
123 for (size_t d = 0; d < width; ++d) {
124 double sum = 0.0;
125 for (size_t i = 0; i < vocabulary; ++i) {
126 /* Product of two FP32 values is exact in FP64. */
127 sum += (double)probabilities[i] * embedding[i * width + d];
128 }
129 mean[r * width + d] = (float)sum;
130 }
131 ids[r] = (int32_t)best;
132 }

136static float guidance_value(float conditional, float unconditional, float scale)
137{
138 if (scale == 1.0f) return conditional;
139 const volatile float delta = conditional - unconditional;
140 const volatile float weighted = scale * delta;
141 return unconditional + weighted;
142}

144int ck_diffusion_guidance_checked(const float *conditional, const float *unconditional,
145 const float *scale, double *output, size_t elements,
146 size_t scale_elements)
147{
148 if (!elements || scale_elements != 1) return -1;

152 for (size_t i = 0; i < elements; ++i)

154 !isfinite(guidance_value(conditional[i], unconditional[i], *scale))) return -1;
155 for (size_t i = 0; i < elements; ++i)
156 output[i] = guidance_value(conditional[i], unconditional[i], *scale);
