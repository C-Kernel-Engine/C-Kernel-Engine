35 !kernel_size || !stride || !dilation || !output_frames ||
36 input_channels > INT_MAX || output_channels > INT_MAX ||
37 input_frames > INT_MAX || kernel_size > INT_MAX ||
38 stride > INT_MAX || padding > INT_MAX || dilation > INT_MAX ||
39 output_frames > INT_MAX)
40 return CK_AUDIO_CONV_CHECKED_INVALID;
41 const int64_t effective_kernel = 1 +

43 const int64_t padded = (int64_t)input_frames +
44 2 * (int64_t)padding - effective_kernel;
45 if (padded < 0 || padded / stride + 1 != output_frames)
46 return CK_AUDIO_CONV_CHECKED_INVALID;
47 if (!ck_checked_span((size_t)input_channels, (size_t)input_frames,
48 input_stride, &input_span) ||
49 !ck_checked_span((size_t)output_channels, (size_t)output_frames,
50 output_stride, &output_span) ||
51 audio_conv1d_checked_workspace((size_t)input_channels,
52 (size_t)output_channels, (size_t)input_frames,
53 (size_t)output_frames, &scratch_need))
54 return CK_AUDIO_CONV_CHECKED_OVERFLOW;
55 if ((size_t)output_channels > SIZE_MAX / (size_t)input_channels ||
56 (size_t)output_channels * (size_t)input_channels > SIZE_MAX / (size_t)kernel_size)
57 return CK_AUDIO_CONV_CHECKED_OVERFLOW;
58 weight_count = (size_t)output_channels * (size_t)input_channels * (size_t)kernel_size;
59 if (weight_count > SIZE_MAX / sizeof(float)) return CK_AUDIO_CONV_CHECKED_OVERFLOW;
60 if (input_elements < input_span || weight_elements < weight_count ||
61 bias_elements < (size_t)output_channels || output_elements < output_span ||
62 scratch_elements < scratch_need)
63 return CK_AUDIO_CONV_CHECKED_CAPACITY;
64 if (!ck_checked_region(input, input_span) ||
65 !ck_checked_region(weight, weight_count) ||
66 !ck_checked_region(bias, (size_t)output_channels) ||
67 !ck_checked_region(output, output_span) ||
68 !ck_checked_region(scratch, scratch_need))
69 return CK_AUDIO_CONV_CHECKED_INVALID;
70 if (ck_checked_overlap(output, output_span, input, input_span) ||
71 ck_checked_overlap(output, output_span, weight, weight_count) ||
72 ck_checked_overlap(output, output_span, bias, (size_t)output_channels) ||
73 ck_checked_overlap(output, output_span, scratch, scratch_need) ||
74 ck_checked_overlap(scratch, scratch_need, input, input_span) ||
75 ck_checked_overlap(scratch, scratch_need, weight, weight_count) ||
76 ck_checked_overlap(scratch, scratch_need, bias, (size_t)output_channels))
77 return CK_AUDIO_CONV_CHECKED_INVALID;
78 if (!ck_checked_finite(weight, weight_count) ||
79 !ck_checked_finite(bias, (size_t)output_channels))
80 return CK_AUDIO_CONV_CHECKED_NONFINITE;
81 for (size_t channel = 0; channel < input_channels; ++channel)
82 if (!ck_checked_finite(input + (size_t)channel * input_stride,
83 (size_t)input_frames))
84 return CK_AUDIO_CONV_CHECKED_NONFINITE;
85 /* Scalar channel/kernel order matches the existing Conv1D reference. No
86 * global thread-pool initialization or allocation occurs in this call. */
87 float *staged = scratch;
88 for (size_t oc = 0; oc < output_channels; ++oc) {
89 for (size_t frame = 0; frame < output_frames; ++frame) {
90 float sum = bias[oc];
91 for (size_t ic = 0; ic < input_channels; ++ic) {
92 const float *row = weight +
93 ((size_t)oc * (size_t)input_channels + (size_t)ic) *
94 (size_t)kernel_size;
95 for (size_t tap = 0; tap < kernel_size; ++tap) {
96 int64_t source = (int64_t)frame * stride +
97 (int64_t)tap * dilation - padding;
98 if (source >= 0 && (size_t)source < input_frames)
99 sum = fmaf(input[(size_t)ic * input_stride +
100 (size_t)source], row[tap], sum);
101 }
102 }
103 staged[(size_t)oc * (size_t)output_frames + (size_t)frame] = sum;
104 }
105 }
106 if (!ck_checked_finite(staged, (size_t)output_channels * (size_t)output_frames))
107 return CK_AUDIO_CONV_CHECKED_NONFINITE;
108 for (size_t channel = 0; channel < output_channels; ++channel)
109 memcpy(output + (size_t)channel * output_stride,
110 staged + (size_t)channel * (size_t)output_frames,
111 (size_t)output_frames * sizeof(float));
112 return CK_AUDIO_CONV_CHECKED_OK;
113}
