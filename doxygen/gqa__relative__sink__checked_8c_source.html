24static __attribute__((target("avx2"))) int finite_cache_avx2(const float *data, size_t length)
25{
26 const __m256i magnitude_mask = _mm256_set1_epi32(0x7fffffff);
27 const __m256i largest_finite = _mm256_set1_epi32(0x7f7fffff);
28 size_t i = 0;
29 for (; length - i >= 8; i += 8) {
30 const __m256i values = _mm256_castps_si256(_mm256_loadu_ps(data + i));
31 const __m256i magnitude = _mm256_and_si256(values, magnitude_mask);
32 if (_mm256_movemask_epi8(_mm256_cmpgt_epi32(magnitude, largest_finite)))
33 return 0;
34 }
35 return ck_checked_finite(data + i, length - i);
36}

38/* Each SIMD lane owns a key. Advance every lane in channel order. */
39static inline __attribute__((target("avx2"))) void scores_for_four_keys(
40 const float *query, const float *keys, size_t dim, double scores[4])
41{
42 __m256d sums = _mm256_setzero_pd();
43 size_t channel = 0;
44 for (; channel + 4 <= dim; channel += 4) {
45 __m128 k0 = _mm_loadu_ps(keys + channel);
46 __m128 k1 = _mm_loadu_ps(keys + dim + channel);
47 __m128 k2 = _mm_loadu_ps(keys + 2 * dim + channel);
48 __m128 k3 = _mm_loadu_ps(keys + 3 * dim + channel);
49 _MM_TRANSPOSE4_PS(k0, k1, k2, k3);
50 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel]), _mm256_cvtps_pd(k0)));
51 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel + 1]), _mm256_cvtps_pd(k1)));
52 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel + 2]), _mm256_cvtps_pd(k2)));
53 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel + 3]), _mm256_cvtps_pd(k3)));
54 }
55 for (; channel < dim; ++channel) {
56 const __m256d k = _mm256_setr_pd(keys[channel], keys[dim + channel],
57 keys[2 * dim + channel], keys[3 * dim + channel]);
58 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel]), k));
59 }
60 _mm256_storeu_pd(scores, sums);
61}

63static __attribute__((target("avx2"))) void attention_row_avx2(
64 const float *query, const float *key, const float *value, const float *relative, float sink,
65 float *output, double *accumulator, size_t dim, size_t begin, size_t position,
66 size_t relative_extent, double scale, size_t ring_capacity)
67{
68 double max_logit = sink, denom = 1.0;
69 for (size_t d = 0; d < dim; ++d) accumulator[d] = 0.0;
70 for (size_t first = begin; first <= position;) {
71 const size_t physical = ring_capacity ? first % ring_capacity : first;
72 size_t available = position - first + 1;
73 if (ring_capacity && available > ring_capacity - physical)
74 available = ring_capacity - physical;
75 double dots[4];
76 if (available >= 4) {
77 scores_for_four_keys(query, key + physical * dim, dim, dots);
78 } else {
79 for (size_t lane = 0; lane < available; ++lane) {
80 const float *keys = key + (physical + lane) * dim;
81 double dot = 0.0;
82 for (size_t d = 0; d < dim; ++d) dot += (double)query[d] * keys[d];
83 dots[lane] = dot;
84 }
85 }
86 const size_t count = available < 4 ? available : 4;
87 for (size_t lane = 0; lane < count; ++lane) {
88 const size_t distance = position - first - lane;
89 const double score = dots[lane] * scale +
90 (distance < relative_extent ? relative[distance] : 0.0);
91 const double next_max = fmax(score, max_logit);
92 const double old_factor = max_logit == next_max ? 1.0 : exp(max_logit - next_max);
93 const double new_factor = score == next_max ? 1.0 : exp(score - next_max);
94 denom = denom * old_factor + new_factor;
95 const float *values = value + (physical + lane) * dim;
96 const __m256d old_vector = _mm256_set1_pd(old_factor), new_vector = _mm256_set1_pd(new_factor);
97 size_t d = 0;
98 for (; d + 4 <= dim; d += 4) {
99 const __m256d previous = _mm256_loadu_pd(accumulator + d);
100 const __m256d current = _mm256_cvtps_pd(_mm_loadu_ps(values + d));
101 _mm256_storeu_pd(accumulator + d, _mm256_add_pd(
102 _mm256_mul_pd(previous, old_vector), _mm256_mul_pd(current, new_vector)));
103 }
104 for (; d < dim; ++d)
105 accumulator[d] = accumulator[d] * old_factor + (double)values[d] * new_factor;
106 max_logit = next_max;
107 }
108 first += count;
109 }
110 for (size_t d = 0; d < dim; ++d) output[d] = (float)(accumulator[d] / denom);
111}

142static void evaluate_attention_rows(size_t start, size_t stop, void *opaque)
143{
144 const struct gqa_rows *task = opaque;
145 const float *query = task->query, *key = task->key, *value = task->value;
146 const float *relative = task->relative, *sink = task->sink;
147 float *output = task->output;
148 const size_t heads = task->heads, group = task->group, dim = task->dim;
149 const size_t capacity = task->capacity, position = task->position;
150 const size_t relative_extent = task->relative_extent, window = task->window;
151 const double scale = task->scale;
152 const int circular = task->circular;
153#ifdef CK_GQA_X86_AVX2
154 const int use_avx2 = task->use_avx2;
155#endif
156 for (size_t row = start; row < stop; ++row) {
157 const size_t qpos = position + row / heads;
158 const size_t begin = window && qpos >= window ? qpos - window + 1 : 0;
159 const float *q = query + row * dim;
160 const float *rel = relative + row * relative_extent;
161 const size_t cache = ((row % heads) / group) * capacity;
162 double *accumulator = task->work + (task->use_row_work ? row * dim : 0);
163#ifdef CK_GQA_X86_AVX2
164 if (use_avx2 && qpos - begin >= 7) {
165 attention_row_avx2(q, key + cache * dim, value + cache * dim, rel, sink[row],
166 output + row * dim, accumulator, dim, begin, qpos, relative_extent, scale, circular ? capacity : 0);
167 continue;
168 }
169#endif
170 double max_logit = sink[row], denom = 1.0;
171 for (size_t d = 0; d < dim; ++d) accumulator[d] = 0.0;
172 for (size_t k = begin; k <= qpos; ++k) {
173 const size_t row_index = circular ? k % capacity : k;
174 const float *keys = key + (cache + row_index) * dim;
175 const float *values = value + (cache + row_index) * dim;
176 double dot = 0.0;
177 /* Each product of finite FP32 operands is exact in FP64. */
178 for (size_t d = 0; d < dim; ++d) dot += (double)q[d] * (double)keys[d];
179 const size_t distance = qpos - k;
180 const double score = dot * scale + (distance < relative_extent ? rel[distance] : 0.0);
181 const double next_max = fmax(score, max_logit);
182 const double old_factor = max_logit == next_max ? 1.0 : exp(max_logit - next_max);
183 const double new_factor = score == next_max ? 1.0 : exp(score - next_max);
184 denom = denom * old_factor + new_factor;
185 for (size_t d = 0; d < dim; ++d)
186 accumulator[d] = accumulator[d] * old_factor + (double)values[d] * new_factor;
187 max_logit = next_max;
188 }
189 for (size_t d = 0; d < dim; ++d)
190 output[row * dim + d] = (float)(accumulator[d] / denom);
191 }
192}

194/* Read the causal span in absolute order, including across a ring wrap. */
195static int gqa_relative_sink_checked(GQA_RELATIVE_ARGS, int circular)
196{
197 if (!heads || heads > 256 || !kv_heads || heads % kv_heads ||
198 !dim || dim > 4096 || !capacity || capacity > SIZE_MAX / kv_heads ||
199 !tokens || tokens > capacity || position > SIZE_MAX - tokens ||
200 (!circular && position > capacity - tokens) || (circular && !window) ||
201 !relative_extent || relative_extent > 16384 || window > 16384)
202 return -2;
203
204 size_t rows, q_size, kv_size, rel_size;
205 if (!ck_checked_span(tokens, heads, heads, &rows) ||
206 !ck_checked_span(rows, dim, dim, &q_size) ||
207 !ck_checked_span(kv_heads * capacity, dim, dim, &kv_size) ||
208 !ck_checked_span(rows, relative_extent, relative_extent, &rel_size) ||
209 q_size > query_elements || q_size > output_elements ||
210 kv_size > key_elements || kv_size > value_elements ||
211 rel_size > relative_elements || rows > sink_elements || dim > work_elements)
212 return -2;
213
214 const size_t first = window && position >= window ? position - window + 1 : 0;
215 const size_t visible = position + tokens - first;
216 /* The writer commits the entire chunk first, so every query's visible key
217 * must fit at once. Otherwise the newest rows can evict an earlier query. */
218 if (circular && visible > capacity) return -2;
219
220 size_t workspace = dim;
221#ifdef CK_GQA_OPTIONAL_POOL
222 const int parallel_work = tokens == 1 ? rows >= 8 && visible >= 128
223 : rows >= 64 && position + tokens >= 64;
224 const int parallel = parallel_work &&
225 q_size <= SIZE_MAX / sizeof(double) && work_elements >= q_size &&
226 ck_get_num_threads && ck_threadpool_global && ck_threadpool_in_dispatch &&
227 ck_threadpool_n_threads && ck_threadpool_parallel_for_n && ck_get_num_threads() > 1;
228 if (parallel) workspace = q_size;
229#endif
230 const size_t scratch_bytes = workspace * sizeof(double);
231 if (!work || (uintptr_t)work % _Alignof(double) ||
232 (uintptr_t)work > UINTPTR_MAX - scratch_bytes ||
233 !ck_checked_region(output, q_size)) return -1;
234 const float *inputs[] = {query, key, value, relative, sink, output};
235 const size_t counts[] = {q_size, kv_size, kv_size, rel_size, rows, q_size};
236 for (size_t i = 0; i < 6; ++i) {
237 if (!ck_checked_region(inputs[i], counts[i])) return -1;
238 const uintptr_t a = (uintptr_t)inputs[i], b = (uintptr_t)work;
239 if ((a < b + scratch_bytes && b < a + counts[i] * sizeof(float)) ||
240 (i != 5 && ck_checked_overlap(output, q_size, inputs[i], counts[i])))
241 return -1;
242 }
243
244 if (!ck_checked_finite(query, q_size) || !ck_checked_finite(relative, rel_size) ||
245 !ck_checked_finite(sink, rows)) return -3;
246 const size_t physical = circular ? first % capacity : first;
247 const size_t contiguous = visible < capacity - physical ? visible : capacity - physical;
248 for (size_t h = 0; h < kv_heads; ++h) {
249 const size_t offset = (h * capacity + physical) * dim;
250 const size_t prefix = contiguous * dim;
251 const size_t suffix = (visible - contiguous) * dim;
252 const size_t head_start = h * capacity * dim;
253 if (!finite_cache(key + offset, prefix) || !finite_cache(value + offset, prefix) ||
254 (suffix && (!finite_cache(key + head_start, suffix) ||
255 !finite_cache(value + head_start, suffix))))
256 return -3;
257 }

259 struct gqa_rows work_rows = {
260 .query = query, .key = key, .value = value, .relative = relative, .sink = sink,
261 .output = output, .work = work, .heads = heads, .group = heads / kv_heads,
262 .dim = dim, .capacity = capacity, .position = position, .relative_extent = relative_extent,
263 .window = window, .scale = 1.0 / sqrt((double)dim), .circular = circular,
264 .use_row_work = workspace > dim,
265#ifdef CK_GQA_X86_AVX2
266 .use_avx2 = dim >= 16 && __builtin_cpu_supports("avx2"),
267#endif
268 };
269#ifdef CK_GQA_OPTIONAL_POOL
270 if (parallel)
271 ck_checked_parallel_for(rows, tokens == 1 ? 1 : 4, evaluate_attention_rows, &work_rows);
272 else
273#endif
274 evaluate_attention_rows(0, rows, &work_rows);
275 return 0;
276}
