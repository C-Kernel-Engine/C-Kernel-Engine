41/* =============================================================================
42 * AXPY: y = y + alpha * x
43 *
44 * Core operation for MoE expert gathering:
45 * output = sum_i(weight_i * expert_output_i)
46 *
47 * Implemented as: output += weight * expert_output (called for each expert)
48 * ============================================================================= */
49
50/**
51 * @brief In-place AXPY: y += alpha * x
52 * @test test_axpy.py::TestAXPY::test_axpy_f32
53 * @test test_axpy.py::TestAXPY::test_axpy_vs_naive
54 *
55 * In-place scaled vector addition: y += alpha * x
56 * BLAS-like axpy operation.
57 *
58 * After changes: make test
59 */
60void axpy_f32(float *y,
61 const float *x,
62 float alpha,
63 int n)
64{
65 if (!y || !x || n <= 0) {
66 return;
67 }
68
69 int i = 0;
70
71#ifdef __AVX512F__
72 __m512 valpha = _mm512_set1_ps(alpha);
73 for (; i + 16 <= n; i += 16) {
74 __m512 vy = _mm512_loadu_ps(&y[i]);
75 __m512 vx = _mm512_loadu_ps(&x[i]);
76 vy = _mm512_fmadd_ps(vx, valpha, vy); /* y = y + alpha * x */
77 _mm512_storeu_ps(&y[i], vy);
78 }
79#endif
80
81#ifdef __AVX2__
82 __m256 valpha256 = _mm256_set1_ps(alpha);
83 for (; i + 8 <= n; i += 8) {
84 __m256 vy = _mm256_loadu_ps(&y[i]);
85 __m256 vx = _mm256_loadu_ps(&x[i]);
86 vy = _mm256_fmadd_ps(vx, valpha256, vy);
87 _mm256_storeu_ps(&y[i], vy);
88 }
89#endif
90
91 /* Scalar remainder */
92 for (; i < n; i++) {
93 y[i] += alpha * x[i];
94 }
95}

97/* =============================================================================
98 * Scaled copy: y = alpha * x
99 *
100 * First step when accumulating: initialize output with first expert's result.
101 * ============================================================================= */
102
103/**
104 * @brief Scaled copy: y = alpha * x
105 *
106 * @param y Output vector [n]
107 * @param x Input vector [n]
108 * @param alpha Scalar multiplier
109 * @param n Vector length
110 */
111void scal_copy_f32(float *y,
112 const float *x,
113 float alpha,
114 int n)
115{
116 if (!y || !x || n <= 0) {
117 return;
118 }
119
120 int i = 0;
121
122#ifdef __AVX512F__
123 __m512 valpha = _mm512_set1_ps(alpha);
124 for (; i + 16 <= n; i += 16) {
125 __m512 vx = _mm512_loadu_ps(&x[i]);
126 __m512 vy = _mm512_mul_ps(vx, valpha);
127 _mm512_storeu_ps(&y[i], vy);
128 }
129#endif
130
131#ifdef __AVX2__
132 __m256 valpha256 = _mm256_set1_ps(alpha);
133 for (; i + 8 <= n; i += 8) {
134 __m256 vx = _mm256_loadu_ps(&x[i]);
135 __m256 vy = _mm256_mul_ps(vx, valpha256);
136 _mm256_storeu_ps(&y[i], vy);
137 }
138#endif
139
140 for (; i < n; i++) {
141 y[i] = alpha * x[i];
142 }
143}

145/* =============================================================================
146 * Weighted sum: y = sum_i(weights[i] * x[i])
147 *
148 * Combine multiple expert outputs with their routing weights in one pass.
149 * More efficient than multiple axpy calls when all inputs are available.
150 * ============================================================================= */
151
152/**
153 * @brief Weighted sum of k vectors: y = sum_i(weights[i] * vectors[i])
154 *
155 * @param y Output vector [n]
156 * @param vectors Array of k input vector pointers, each [n]
157 * @param weights Array of k scalar weights
158 * @param k Number of vectors to combine
159 * @param n Vector length
160 */
161void weighted_sum_f32(float *y,
162 const float **vectors,
163 const float *weights,
164 int k,
165 int n)
166{
167 if (!y || !vectors || !weights || k <= 0 || n <= 0) {
168 return;
169 }
170
171 /* Initialize with first vector */
172 scal_copy_f32(y, vectors[0], weights[0], n);
173
174 /* Accumulate rest */
175 for (int i = 1; i < k; i++) {
176 axpy_f32(y, vectors[i], weights[i], n);
177 }
178}

180/* =============================================================================
181 * Zero-initialized AXPY accumulation
182 *
183 * Zero output first, then accumulate. Useful when output may contain garbage.
184 * ============================================================================= */
185
186/**
187 * @brief Zero output then accumulate: y = 0; y += alpha * x
188 *
189 * @param y Output vector [n], zeroed then accumulated
190 * @param x Input vector [n]
191 * @param alpha Scalar multiplier
192 * @param n Vector length
193 */
194void axpy_zero_f32(float *y,
195 const float *x,
196 float alpha,
197 int n)
198{
199 if (!y || n <= 0) {
200 return;
201 }
202
203 memset(y, 0, n * sizeof(float));
204
205 if (x) {
206 axpy_f32(y, x, alpha, n);
207 }
208}

210/* =============================================================================
211 * 2D batched AXPY for [tokens, hidden] shaped tensors
212 *
213 * Process multiple tokens at once, common in transformer inference.
214 * ============================================================================= */
215
216/**
217 * @brief Batched AXPY for 2D tensors: Y[t,:] += alpha * X[t,:]
218 *
219 * @param Y Output tensor [num_tokens, dim]
220 * @param X Input tensor [num_tokens, dim]
221 * @param alpha Scalar multiplier
222 * @param num_tokens Number of tokens
223 * @param dim Hidden dimension
224 * @param y_stride Stride between Y rows (for alignment)
225 * @param x_stride Stride between X rows
226 */
227void axpy_2d_f32(float *Y,
228 const float *X,
229 float alpha,
230 int num_tokens,
231 int dim,
232 int y_stride,
233 int x_stride)
234{
235 if (!Y || !X || num_tokens <= 0 || dim <= 0) {
236 return;
237 }
238
239 /* Default strides if not specified */
240 if (y_stride <= 0) y_stride = dim;
241 if (x_stride <= 0) x_stride = dim;
242
243 for (int t = 0; t < num_tokens; t++) {
244 axpy_f32(Y + t * y_stride, X + t * x_stride, alpha, dim);
245 }
246}

271/* =============================================================================
272 * Routed MoE expert MLP with ReLU2 activation.
273 *
274 * expert_up layout: [n_experts, intermediate_dim, hidden_dim]
275 * expert_down layout: [n_experts, hidden_dim, intermediate_dim]
276 * output layout: [rows, hidden_dim]
277 * ============================================================================= */
278static inline size_t ck_moe_up_idx(int e, int i, int h, int intermediate_dim, int hidden_dim)
279{
280 return ((size_t)e * (size_t)intermediate_dim + (size_t)i) * (size_t)hidden_dim + (size_t)h;
281}
282
283static inline size_t ck_moe_down_idx(int e, int h, int i, int hidden_dim, int intermediate_dim)
284{
285 return ((size_t)e * (size_t)hidden_dim + (size_t)h) * (size_t)intermediate_dim + (size_t)i;
286}

294static void ck_moe_debug_finite(const char *name, const float *x, size_t n)
295{
296 if (!ck_moe_debug_enabled() || !x) {
297 return;
298 }
299 size_t finite = 0;
300 size_t nan = 0;
301 size_t inf = 0;
302 float min_v = 0.0f;
303 float max_v = 0.0f;
304 int have = 0;
305 for (size_t i = 0; i < n; ++i) {
306 const float v = x[i];
307 if (isnan(v)) {
308 ++nan;
309 } else if (isinf(v)) {
310 ++inf;
311 } else {
312 ++finite;
313 if (!have || v < min_v) min_v = v;
314 if (!have || v > max_v) max_v = v;
315 have = 1;
316 }
317 }
318 fprintf(stderr,
319 "[CK_DEBUG_MOE] %s finite=%zu/%zu nan=%zu inf=%zu min=%g max=%g\n",
320 name,
321 finite,
322 n,
323 nan,
324 inf,
325 have ? min_v : 0.0f,
326 have ? max_v : 0.0f);
327}

329void moe_relu2_expert_forward_f32(const float *hidden,
330 const int *indices,
331 const float *routing_weights,
332 const float *expert_up,
333 const float *expert_down,
334 float *output,
335 int rows,
336 int hidden_dim,
337 int intermediate_dim,
338 int n_experts,
339 int top_k)
340{
341 if (!hidden || !indices || !routing_weights || !expert_up || !expert_down || !output ||
342 rows <= 0 || hidden_dim <= 0 || intermediate_dim <= 0 || n_experts <= 0 || top_k <= 0) {
343 return;
344 }
345
346 const size_t out_count = (size_t)rows * (size_t)hidden_dim;
347 for (size_t p = 0; p < out_count; ++p) output[p] = 0.0f;
348
349 float pre[intermediate_dim];
350 float act[intermediate_dim];
351
352 for (int r = 0; r < rows; ++r) {
353 const float *x = hidden + (size_t)r * (size_t)hidden_dim;
354 float *y = output + (size_t)r * (size_t)hidden_dim;
355 for (int slot = 0; slot < top_k; ++slot) {
356 const int e = indices[(size_t)r * (size_t)top_k + (size_t)slot];
357 if (e < 0 || e >= n_experts) continue;
358 const float route_w = routing_weights[(size_t)r * (size_t)top_k + (size_t)slot];
359
360 for (int i = 0; i < intermediate_dim; ++i) {
361 float v = 0.0f;
362 for (int h = 0; h < hidden_dim; ++h) {
363 v += expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] * x[h];
364 }
365 pre[i] = v;
366 act[i] = (v > 0.0f) ? v * v : 0.0f;
367 }
368
369 for (int h = 0; h < hidden_dim; ++h) {
370 float v = 0.0f;
371 for (int i = 0; i < intermediate_dim; ++i) {
372 v += expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)] * act[i];
373 }
374 y[h] += route_w * v;
375 }
376 }
377 }
378}

544void moe_relu2_shared_forward_q5_1_q8_0(const float *hidden,
545 const float *routed,
546 const void *shared_up,
547 const void *shared_down,
548 float *output,
549 int rows,
550 int hidden_dim,
551 int intermediate_dim)
552{
553 if (!hidden || !shared_up || !shared_down || !output || rows <= 0 || hidden_dim <= 0 || intermediate_dim <= 0) {
554 return;
555 }
556
557 const size_t up_row_bytes = ck_dtype_row_bytes(CK_DT_Q5_1, (size_t)hidden_dim);
558 const size_t down_row_bytes = ck_dtype_row_bytes(CK_DT_Q8_0, (size_t)intermediate_dim);

561
562 float up_row[hidden_dim];

564 float act[intermediate_dim];
565
566 for (int r = 0; r < rows; ++r) {

569 float *y = output + (size_t)r * (size_t)hidden_dim;

612void moe_relu2_expert_backward_f32(const float *d_output,
613 const float *hidden,
614 const int *indices,
615 const float *routing_weights,
616 const float *expert_up,
617 const float *expert_down,
618 float *d_hidden,
619 float *d_routing_weights,
620 float *d_expert_up,
621 float *d_expert_down,
622 int rows,
623 int hidden_dim,
624 int intermediate_dim,
625 int n_experts,
626 int top_k)
627{
628 if (!d_output || !hidden || !indices || !routing_weights || !expert_up || !expert_down ||
629 !d_hidden || !d_routing_weights || !d_expert_up || !d_expert_down ||
630 rows <= 0 || hidden_dim <= 0 || intermediate_dim <= 0 || n_experts <= 0 || top_k <= 0) {
631 return;
632 }
633
634 for (size_t p = 0; p < (size_t)rows * (size_t)hidden_dim; ++p) d_hidden[p] = 0.0f;
635 for (size_t p = 0; p < (size_t)rows * (size_t)top_k; ++p) d_routing_weights[p] = 0.0f;
636 for (size_t p = 0; p < (size_t)n_experts * (size_t)intermediate_dim * (size_t)hidden_dim; ++p) d_expert_up[p] = 0.0f;
637 for (size_t p = 0; p < (size_t)n_experts * (size_t)hidden_dim * (size_t)intermediate_dim; ++p) d_expert_down[p] = 0.0f;
638
639 float pre[intermediate_dim];
640 float act[intermediate_dim];
641 float d_act[intermediate_dim];
642 float d_pre[intermediate_dim];
643 float expert_out[hidden_dim];
644
645 for (int r = 0; r < rows; ++r) {
646 const float *x = hidden + (size_t)r * (size_t)hidden_dim;
647 const float *dy = d_output + (size_t)r * (size_t)hidden_dim;
648 float *dx = d_hidden + (size_t)r * (size_t)hidden_dim;
649
650 for (int slot = 0; slot < top_k; ++slot) {
651 const int e = indices[(size_t)r * (size_t)top_k + (size_t)slot];
652 if (e < 0 || e >= n_experts) continue;
653 const float route_w = routing_weights[(size_t)r * (size_t)top_k + (size_t)slot];
654
655 for (int i = 0; i < intermediate_dim; ++i) {
656 float v = 0.0f;
657 for (int h = 0; h < hidden_dim; ++h) {
658 v += expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] * x[h];
659 }
660 pre[i] = v;
661 act[i] = (v > 0.0f) ? v * v : 0.0f;
662 d_act[i] = 0.0f;
663 }
664
665 for (int h = 0; h < hidden_dim; ++h) {
666 float v = 0.0f;
667 for (int i = 0; i < intermediate_dim; ++i) {
668 v += expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)] * act[i];
669 }
670 expert_out[h] = v;
671 }
672
673 float d_route = 0.0f;
674 for (int h = 0; h < hidden_dim; ++h) {
675 const float d_expert_out = dy[h] * route_w;
676 d_route += dy[h] * expert_out[h];
677 for (int i = 0; i < intermediate_dim; ++i) {
678 d_expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)] += d_expert_out * act[i];
679 d_act[i] += d_expert_out * expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)];
680 }
681 }
682 d_routing_weights[(size_t)r * (size_t)top_k + (size_t)slot] += d_route;
683
684 for (int i = 0; i < intermediate_dim; ++i) {
685 d_pre[i] = (pre[i] > 0.0f) ? d_act[i] * 2.0f * pre[i] : 0.0f;
686 }
687
688 for (int i = 0; i < intermediate_dim; ++i) {
689 const float dpi = d_pre[i];
690 for (int h = 0; h < hidden_dim; ++h) {
691 d_expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] += dpi * x[h];
692 dx[h] += dpi * expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)];
693 }
694 }
695 }
696 }
