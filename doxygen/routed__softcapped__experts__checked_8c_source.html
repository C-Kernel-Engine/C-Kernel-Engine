26/* Packed blocks are read in GGUF byte order and expanded inside the dot.
27 * The FP32 activation and ordered FP64 accumulation policy is shared. */
28static float expert_dot(const float *input, const unsigned char *weight,
29 size_t width, enum expert_storage storage, const unsigned char *activation)
30{
31 if (activation) return ck_q8_activation_dot(weight, activation, width);
32 if (storage == EXPERT_Q8_0) return ck_packed_q8_dot_ordered(input, weight, width);
33 double sum = 0.0;
34 if (storage == EXPERT_F32) {
35 for (size_t i = 0; i < width; ++i) {
36 float value;
37 memcpy(&value, weight + i * sizeof(float), sizeof(float));
38 sum += (double)input[i] * (double)value;
39 }
40 } else {
41 const size_t block_bytes = packed_block_bytes(storage);
42 for (size_t i = 0; i < width; i += QK8_0, weight += block_bytes) {
43 const float scale = ck_fp16_to_fp32_soft(
44 (uint16_t)(weight[0] | ((uint16_t)weight[1] << 8)));
45 for (size_t j = 0; j < QK8_0; ++j) {
46 const unsigned pair = weight[2 + j % 16];
47 const float expanded = scale * (float)iq4_nl_values[j < 16 ? pair & 15 : pair >> 4];
48 sum += (double)input[i + j] * (double)expanded;
49 }
50 }
51 }
52 return (float)sum;
53}

74static int routed_softcapped_experts(ROUTED_EXPERT_ARGS, enum expert_storage storage, int quantize, int reference_moe)
75{
76 /* Bounds keep workspace arithmetic valid on supported size_t widths. */
77 if (!rows || rows > 16 || !hidden_dim || hidden_dim > 8192 ||
78 !inner || inner > 8192 || !experts || experts > 1024 || !slots ||
79 slots > experts || !top_k || top_k > experts ||
80 !isfinite(gate_cap) || gate_cap <= 0.0f || !isfinite(up_cap) || up_cap <= 0.0f)
81 return -1;
82 const int packed = storage != EXPERT_F32;
83 if (packed && (hidden_dim % QK8_0 || inner % QK8_0)) return -1;
84 const size_t block_bytes = packed ? packed_block_bytes(storage) : sizeof(float);
85 const size_t block_elements = packed ? QK8_0 : 1;
86 const size_t gate_row = hidden_dim / block_elements * block_bytes;
87 const size_t down_row = inner / block_elements * block_bytes;
88 const size_t expert_bytes = inner * gate_row;
89 const size_t route_count = rows * top_k;
90 const size_t output_elements = rows * hidden_dim;
91 const size_t row_outputs = route_count * hidden_dim;
92 const size_t packed_elements = quantize ? (gate_row + down_row + sizeof(float) - 1) / sizeof(float) : 0;
93 const size_t row_scratch = 3 * inner + packed_elements;
94 const size_t staged_elements = row_outputs + output_elements;
95 size_t scratch_rows = 1;
96#ifdef _OPENMP
97 if (work_elements >= staged_elements + rows * row_scratch) scratch_rows = rows;
98#endif
99 const size_t scratch_elements = staged_elements + scratch_rows * row_scratch;
100 if (hidden_elements < output_elements || id_elements < route_count || route_elements < route_count ||
101 expert_slot_elements < experts || expert_output_elements < row_outputs ||
102 output_capacity < output_elements || work_elements < scratch_elements) return -1;
103 if (expert_bytes != hidden_dim * down_row || expert_bytes > SIZE_MAX / slots ||
104 matrix_bytes != expert_bytes * slots) return -1;
105 const void *buffers[] = {hidden, ids, route_weights, expert_slots, gate, up, down,
106 expert_output, output, work};
107 const size_t sizes[] = {output_elements * sizeof(float), route_count * sizeof(int32_t),
108 route_count * sizeof(float), experts * sizeof(int32_t),
109 matrix_bytes, matrix_bytes, matrix_bytes, row_outputs * sizeof(float),
110 output_elements * sizeof(float),
111 scratch_elements * sizeof(float)};
112 for (size_t i = 0; i < sizeof(buffers) / sizeof(buffers[0]); ++i) {
113 if (!region(buffers[i], sizes[i])) return -1;
114 if (i < 4 || i >= 7) {
115 if ((uintptr_t)buffers[i] % _Alignof(float)) return -1;
116 }
117 if (i < 7) continue;
118 for (size_t j = 0; j < i; ++j)
119 if (contains_overlap(buffers[i], sizes[i], buffers[j], sizes[j])) return -1;
120 }
121 if (!ck_checked_finite(hidden, output_elements) ||
122 !ck_checked_finite(route_weights, route_count)) return -2;
123
124 unsigned char assigned[1024] = {0};
125 for (size_t id = 0; id < experts; ++id) {
126 const int32_t slot = expert_slots[id];
127 if (slot == -1) continue;
128 if (slot < 0 || (size_t)slot >= slots || assigned[slot]) return -2;
129 assigned[slot] = 1;
130 }
131 for (size_t row = 0; row < rows; ++row) {
132 for (size_t k = 0; k < top_k; ++k) {
133 const size_t route = row * top_k + k;
134 const int32_t id = ids[route];
135 if (id < 0 || (size_t)id >= experts || expert_slots[id] < 0 ||
136 (size_t)expert_slots[id] >= slots || route_weights[route] < 0.0f) return -2;
137 for (size_t prev = 0; prev < k; ++prev)
138 if (id == ids[row * top_k + prev]) return -2;
139 }
140 }
141
142 const float gate_reciprocal = reference_moe ? 1.0f / gate_cap : 0.0f;
143 const float up_reciprocal = reference_moe ? 1.0f / up_cap : 0.0f;
144 if (!isfinite(gate_reciprocal) || !isfinite(up_reciprocal)) return -3;
145
146 float *staged_experts = work;
147 float *staged_output = staged_experts + row_outputs;
148 float *row_work = staged_output + output_elements;
149 memset(staged_output, 0, output_elements * sizeof(float));
150 int failed = 0;
151#ifdef _OPENMP
152#pragma omp parallel for if(scratch_rows > 1) reduction(|:failed)
153#endif
154 for (size_t row = 0; row < rows; ++row) {
155 float *gate_values = row_work + (scratch_rows > 1 ? row * row_scratch : 0);
156 float *up_values = gate_values + inner;
157 float *activation = up_values + inner;
158 unsigned char *packed_hidden = quantize ? (unsigned char *)(activation + inner) : NULL;
159 unsigned char *packed_activation = quantize ? packed_hidden + gate_row : NULL;
160 const float *x = hidden + row * hidden_dim;
161 float *y = staged_output + row * hidden_dim;
162 if (quantize && ck_pack_q8_activation_rne(x, packed_hidden, hidden_dim)) goto row_failed;
163 size_t ordered_slots[1024];
164 if (!reference_moe) {
165 size_t next = 0;
166 for (size_t id = 0; id < experts; ++id)
167 for (size_t slot = 0; slot < top_k; ++slot)
168 if ((size_t)ids[row * top_k + slot] == id) ordered_slots[next++] = slot;
169 }
170 for (size_t index = 0; index < top_k; ++index) {
171 const size_t route = row * top_k + (reference_moe ? index : ordered_slots[index]);
172 const size_t offset = (size_t)expert_slots[ids[route]] * expert_bytes;
173 const unsigned char *g = (const unsigned char *)gate + offset;
174 const unsigned char *u = (const unsigned char *)up + offset;
175 const unsigned char *d = (const unsigned char *)down + offset;
176 if (packed_hidden) {
177 for (size_t i = 0; i < inner; ++i) {
178 ck_q8_activation_dot_pair(g + i * gate_row, u + i * gate_row, packed_hidden,
179 hidden_dim, gate_values + i, up_values + i);
180 }
181 } else {
182 for (size_t i = 0; i < inner; ++i) {
183 gate_values[i] = expert_dot(x, g + i * gate_row, hidden_dim, storage, NULL);
184 up_values[i] = expert_dot(x, u + i * gate_row, hidden_dim, storage, NULL);
185 }
186 }
187 if (reference_moe) {
188 for (size_t i = 0; i < inner; ++i) {
189 const float gate = gate_values[i], up = up_values[i];
190 if (!isfinite(gate) || !isfinite(up)) goto row_failed;
191 const float capped_gate = gate_cap * tanhf(gate * gate_reciprocal);
192 const float capped_up = up_cap * tanhf(up * up_reciprocal);
193 const float sigmoid = 1.0f / (1.0f + expf(-gate));
194 activation[i] = (capped_gate * sigmoid) * capped_up;
195 if (!isfinite(activation[i])) goto row_failed;
196 }
197 } else if (softcapped_swiglu_rows_checked_f32(gate_values, inner, up_values, inner,
198 activation, inner, inner, gate_cap, up_cap)) goto row_failed;
199 if (quantize && ck_pack_q8_activation_rne(activation, packed_activation, inner)) goto row_failed;
200 for (size_t i = 0; i < hidden_dim; ++i) {
201 const float v = expert_dot(activation, d + i * down_row, inner, storage, packed_activation);
202 staged_experts[route * hidden_dim + i] = v;
203 if (reference_moe) {
204 /* A stored FP32 multiply precedes the ordered FP32 addition. */
205 const volatile float weighted = v * route_weights[route];
206 y[i] = index == 0 ? weighted : y[i] + weighted;
207 } else {
208 y[i] += v * route_weights[route];
209 }
210 if (!isfinite(v) || !isfinite(y[i])) goto row_failed;
211 }
212 }
213 continue;
214row_failed:
215 failed = 1;
216 }
217 if (failed) return -3;
218 memcpy(expert_output, staged_experts, row_outputs * sizeof(float));
219 memcpy(output, staged_output, output_elements * sizeof(float));
220 return 0;
221}
