1/**
2 * @file ssm_kernels.c
3 * @brief FP32 SSM causal depthwise convolution kernels for qwen3next/Qwen3.5.
4 *
5 * CK-ENGINE KERNEL RULES:
6 * =======================
7 * 1. NO malloc/free - memory via bump allocator, pointers passed in
8 * 2. NO OpenMP - parallelization at orchestrator/codegen layer
9 * 3. API must define: inputs, outputs, workspace, and memory layouts
10 * 4. Pure computation - deterministic, no side effects
11 *
12 * After changes: make test-ssm-conv && make test-kernels
13 *
14 * This file implements the GGML_OP_SSM_CONV semantics used by qwen3next before
15 * the recurrent DeltaNet update:
16 * out[seq, token, ch] = dot(conv_x[seq, ch, token:token+kernel], kernel[ch, :])
17 *
18 * Memory layouts:
19 * conv_x : [num_seqs, num_channels, kernel_size - 1 + num_tokens]
20 * kernel : [num_channels, kernel_size]
21 * out : [num_seqs, num_tokens, num_channels]
22 * d_out : same as out
23 * d_conv_x : same as conv_x
24 * d_kernel : same as kernel
25 */

38void ssm_conv1d_forward_ref(const float *conv_x,
39 const float *kernel,
40 float *out,
41 int kernel_size,
42 int num_channels,
43 int num_tokens,
44 int num_seqs)
45{
46 if (!conv_x || !kernel || !out) {
47 return;
48 }
49 if (kernel_size <= 0 || num_channels <= 0 || num_tokens < 0 || num_seqs <= 0) {
50 return;
51 }
52
53 const size_t seq_width = (size_t)kernel_size - 1u + (size_t)num_tokens;
54 const size_t conv_seq_stride = (size_t)num_channels * seq_width;
55 const size_t out_seq_stride = (size_t)num_tokens * (size_t)num_channels;
56
57 for (int seq = 0; seq < num_seqs; ++seq) {
58 const float *conv_seq = conv_x + (size_t)seq * conv_seq_stride;
59 float *out_seq = out + (size_t)seq * out_seq_stride;
60
61 for (int tok = 0; tok < num_tokens; ++tok) {
62 float *out_tok = out_seq + (size_t)tok * (size_t)num_channels;
63
64 for (int ch = 0; ch < num_channels; ++ch) {
65 const float *conv_row = conv_seq + (size_t)ch * seq_width + (size_t)tok;
66 const float *kernel_row = kernel + (size_t)ch * (size_t)kernel_size;
67 float sumf = 0.0f;
68 for (int k = 0; k < kernel_size; ++k) {
69 sumf += conv_row[k] * kernel_row[k];
70 }
71 out_tok[ch] = sumf;
72 }
73 }
74 }
75}

77void ssm_conv1d_backward_ref(const float *d_out,
78 const float *conv_x,
79 const float *kernel,
80 float *d_conv_x,
81 float *d_kernel,
82 int kernel_size,
83 int num_channels,
84 int num_tokens,
85 int num_seqs)
86{
87 if (!d_out || !conv_x || !kernel || !d_conv_x || !d_kernel) {
88 return;
89 }
90 if (kernel_size <= 0 || num_channels <= 0 || num_tokens < 0 || num_seqs <= 0) {
91 return;
92 }
93
94 const size_t seq_width = (size_t)kernel_size - 1u + (size_t)num_tokens;
95 const size_t conv_total = (size_t)num_seqs * (size_t)num_channels * seq_width;
96 const size_t kernel_total = (size_t)num_channels * (size_t)kernel_size;
97 const size_t conv_seq_stride = (size_t)num_channels * seq_width;
98 const size_t out_seq_stride = (size_t)num_tokens * (size_t)num_channels;
99
100 memset(d_conv_x, 0, conv_total * sizeof(float));
101 memset(d_kernel, 0, kernel_total * sizeof(float));
102
103 for (int seq = 0; seq < num_seqs; ++seq) {
104 const float *d_out_seq = d_out + (size_t)seq * out_seq_stride;
105 const float *conv_seq = conv_x + (size_t)seq * conv_seq_stride;
106 float *d_conv_seq = d_conv_x + (size_t)seq * conv_seq_stride;
107
108 for (int tok = 0; tok < num_tokens; ++tok) {
109 const float *d_out_tok = d_out_seq + (size_t)tok * (size_t)num_channels;
110
111 for (int ch = 0; ch < num_channels; ++ch) {
112 const float grad = d_out_tok[ch];
113 const float *conv_row = conv_seq + (size_t)ch * seq_width + (size_t)tok;
114 float *d_conv_row = d_conv_seq + (size_t)ch * seq_width + (size_t)tok;
115 const float *kernel_row = kernel + (size_t)ch * (size_t)kernel_size;
116 float *d_kernel_row = d_kernel + (size_t)ch * (size_t)kernel_size;
117
118 for (int k = 0; k < kernel_size; ++k) {
119 d_kernel_row[k] += grad * conv_row[k];
120 d_conv_row[k] += grad * kernel_row[k];
121 }
122 }
123 }
124 }
125}

138/*
139 * llama.cpp's production GGML_OP_SSM_CONV CPU implementation accumulates
140 * every channel/token dot product as an ascending scalar multiply/add chain.
141 * Preserve the rounded product before each addition: contraction or
142 * vectorization changes FP32 rounding boundaries.
143 */
144typedef struct {
145 const float *conv_x;
146 const float *kernel;
147 float *out;
148 int kernel_size;
149 int num_channels;
150 int num_tokens;
151 int num_seqs;
152} ck_ssm_conv1d_llama_args_t;
153
154static void ck_ssm_conv1d_llama_channel_range(int begin, int end, void *opaque)
155{
156 const ck_ssm_conv1d_llama_args_t *args =
157 (const ck_ssm_conv1d_llama_args_t *)opaque;
158 const int kernel_size = args->kernel_size;
159 const int num_channels = args->num_channels;
160 const int num_tokens = args->num_tokens;
161 const size_t seq_width = (size_t)kernel_size - 1u + (size_t)num_tokens;
162 const size_t conv_seq_stride = (size_t)num_channels * seq_width;
163 const size_t out_seq_stride = (size_t)num_tokens * (size_t)num_channels;
164
165 for (int seq = 0; seq < args->num_seqs; ++seq) {
166 const float *conv_seq = args->conv_x + (size_t)seq * conv_seq_stride;
167 float *out_seq = args->out + (size_t)seq * out_seq_stride;
168
169 for (int ch = begin; ch < end; ++ch) {
170 const float *kernel_row =
171 args->kernel + (size_t)ch * (size_t)kernel_size;
172 const float *conv_channel = conv_seq + (size_t)ch * seq_width;
173 for (int tok = 0; tok < num_tokens; ++tok) {
174 const float *conv_row =
175 conv_channel + (size_t)tok;
176 float sum = 0.0f;
177 for (int k = 0; k < kernel_size; ++k) {
178 volatile float product = conv_row[k] * kernel_row[k];
179 volatile float next = sum + product;
180 sum = next;
181 }
182 out_seq[(size_t)tok * (size_t)num_channels + (size_t)ch] = sum;
183 }
184 }
185 }
186}

206void ssm_conv1d_forward_llama_production(const float *conv_x,
207 const float *kernel,
208 float *out,
209 int kernel_size,
210 int num_channels,
211 int num_tokens,
212 int num_seqs)
213{
214 if (!conv_x || !kernel || !out || kernel_size <= 0 || num_channels <= 0 ||
215 num_tokens < 0 || num_seqs <= 0) {
216 return;
217 }
218 ck_ssm_conv1d_llama_args_t args = {
219 conv_x, kernel, out, kernel_size, num_channels, num_tokens, num_seqs,
220 };
221 ck_threadpool_t *pool = ck_threadpool_global();
222 const int workers = pool ? ck_threadpool_n_threads(pool) : 1;
223 const int active = workers < num_channels ? workers : num_channels;
224 if (active > 1 && num_tokens > 1) {
225 ck_threadpool_parallel_for_n(
226 pool, active, 0, num_channels, 32,
227 ck_ssm_conv1d_llama_channel_range, &args);
228 } else {
229 ck_ssm_conv1d_llama_channel_range(0, num_channels, &args);
230 }
231}
