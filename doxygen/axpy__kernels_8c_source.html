42/* =============================================================================
43 * AXPY: y = y + alpha * x
44 *
45 * Core operation for MoE expert gathering:
46 * output = sum_i(weight_i * expert_output_i)
47 *
48 * Implemented as: output += weight * expert_output (called for each expert)
49 * ============================================================================= */
50
51/**
52 * @brief In-place AXPY: y += alpha * x
53 * @test test_axpy.py::TestAXPY::test_axpy_f32
54 * @test test_axpy.py::TestAXPY::test_axpy_vs_naive
55 *
56 * In-place scaled vector addition: y += alpha * x
57 * BLAS-like axpy operation.
58 *
59 * After changes: make test
60 */
61void axpy_f32(float *y,
62 const float *x,
63 float alpha,
64 int n)
65{
66 if (!y || !x || n <= 0) {
67 return;
68 }
69
70 int i = 0;
71
72#ifdef __AVX512F__
73 __m512 valpha = _mm512_set1_ps(alpha);
74 for (; i + 16 <= n; i += 16) {
75 __m512 vy = _mm512_loadu_ps(&y[i]);
76 __m512 vx = _mm512_loadu_ps(&x[i]);
77 vy = _mm512_fmadd_ps(vx, valpha, vy); /* y = y + alpha * x */
78 _mm512_storeu_ps(&y[i], vy);
79 }
80#endif
81
82#ifdef __AVX2__
83 __m256 valpha256 = _mm256_set1_ps(alpha);
84 for (; i + 8 <= n; i += 8) {
85 __m256 vy = _mm256_loadu_ps(&y[i]);
86 __m256 vx = _mm256_loadu_ps(&x[i]);
87 vy = _mm256_fmadd_ps(vx, valpha256, vy);
88 _mm256_storeu_ps(&y[i], vy);
89 }
90#endif
91
92 /* Scalar remainder */
93 for (; i < n; i++) {
94 y[i] += alpha * x[i];
95 }
96}

98/* =============================================================================
99 * Scaled copy: y = alpha * x
100 *
101 * First step when accumulating: initialize output with first expert's result.
102 * ============================================================================= */
103
104/**
105 * @brief Scaled copy: y = alpha * x
106 *
107 * @param y Output vector [n]
108 * @param x Input vector [n]
109 * @param alpha Scalar multiplier
110 * @param n Vector length
111 */
112void scal_copy_f32(float *y,
113 const float *x,
114 float alpha,
115 int n)
116{
117 if (!y || !x || n <= 0) {
118 return;
119 }
120
121 int i = 0;
122
123#ifdef __AVX512F__
124 __m512 valpha = _mm512_set1_ps(alpha);
125 for (; i + 16 <= n; i += 16) {
126 __m512 vx = _mm512_loadu_ps(&x[i]);
127 __m512 vy = _mm512_mul_ps(vx, valpha);
128 _mm512_storeu_ps(&y[i], vy);
129 }
130#endif
131
132#ifdef __AVX2__
133 __m256 valpha256 = _mm256_set1_ps(alpha);
134 for (; i + 8 <= n; i += 8) {
135 __m256 vx = _mm256_loadu_ps(&x[i]);
136 __m256 vy = _mm256_mul_ps(vx, valpha256);
137 _mm256_storeu_ps(&y[i], vy);
138 }
139#endif
140
141 for (; i < n; i++) {
142 y[i] = alpha * x[i];
143 }
144}

146/* =============================================================================
147 * Weighted sum: y = sum_i(weights[i] * x[i])
148 *
149 * Combine multiple expert outputs with their routing weights in one pass.
150 * More efficient than multiple axpy calls when all inputs are available.
151 * ============================================================================= */
152
153/**
154 * @brief Weighted sum of k vectors: y = sum_i(weights[i] * vectors[i])
155 *
156 * @param y Output vector [n]
157 * @param vectors Array of k input vector pointers, each [n]
158 * @param weights Array of k scalar weights
159 * @param k Number of vectors to combine
160 * @param n Vector length
161 */
162void weighted_sum_f32(float *y,
163 const float **vectors,
164 const float *weights,
165 int k,
166 int n)
167{
168 if (!y || !vectors || !weights || k <= 0 || n <= 0) {
169 return;
170 }
171
172 /* Initialize with first vector */
173 scal_copy_f32(y, vectors[0], weights[0], n);
174
175 /* Accumulate rest */
176 for (int i = 1; i < k; i++) {
177 axpy_f32(y, vectors[i], weights[i], n);
178 }
179}

181/* =============================================================================
182 * Zero-initialized AXPY accumulation
183 *
184 * Zero output first, then accumulate. Useful when output may contain garbage.
185 * ============================================================================= */
186
187/**
188 * @brief Zero output then accumulate: y = 0; y += alpha * x
189 *
190 * @param y Output vector [n], zeroed then accumulated
191 * @param x Input vector [n]
192 * @param alpha Scalar multiplier
193 * @param n Vector length
194 */
195void axpy_zero_f32(float *y,
196 const float *x,
197 float alpha,
198 int n)
199{
200 if (!y || n <= 0) {
201 return;
202 }
203
204 memset(y, 0, n * sizeof(float));
205
206 if (x) {
207 axpy_f32(y, x, alpha, n);
208 }
209}

211/* =============================================================================
212 * 2D batched AXPY for [tokens, hidden] shaped tensors
213 *
214 * Process multiple tokens at once, common in transformer inference.
215 * ============================================================================= */
216
217/**
218 * @brief Batched AXPY for 2D tensors: Y[t,:] += alpha * X[t,:]
219 *
220 * @param Y Output tensor [num_tokens, dim]
221 * @param X Input tensor [num_tokens, dim]
222 * @param alpha Scalar multiplier
223 * @param num_tokens Number of tokens
224 * @param dim Hidden dimension
225 * @param y_stride Stride between Y rows (for alignment)
226 * @param x_stride Stride between X rows
227 */
228void axpy_2d_f32(float *Y,
229 const float *X,
230 float alpha,
231 int num_tokens,
232 int dim,
233 int y_stride,
234 int x_stride)
235{
236 if (!Y || !X || num_tokens <= 0 || dim <= 0) {
237 return;
238 }
239
240 /* Default strides if not specified */
241 if (y_stride <= 0) y_stride = dim;
242 if (x_stride <= 0) x_stride = dim;
243
244 for (int t = 0; t < num_tokens; t++) {
245 axpy_f32(Y + t * y_stride, X + t * x_stride, alpha, dim);
246 }
247}

274 *
275 * expert_up layout: [n_experts, intermediate_dim, hidden_dim]

279static inline size_t ck_moe_up_idx(int e, int i, int h, int intermediate_dim, int hidden_dim)

281 return ((size_t)e * (size_t)intermediate_dim + (size_t)i) * (size_t)hidden_dim + (size_t)h;

295static void ck_moe_debug_finite(const char *name, const float *x, size_t n)
296{
297 if (!ck_moe_debug_enabled() || !x) {
298 return;
299 }
300 size_t finite = 0;
301 size_t nan = 0;
302 size_t inf = 0;
303 float min_v = 0.0f;
304 float max_v = 0.0f;
305 int have = 0;
306 for (size_t i = 0; i < n; ++i) {
307 const float v = x[i];
308 if (isnan(v)) {
309 ++nan;
310 } else if (isinf(v)) {
311 ++inf;
312 } else {
313 ++finite;
314 if (!have || v < min_v) min_v = v;
315 if (!have || v > max_v) max_v = v;
316 have = 1;
317 }
318 }
319 fprintf(stderr,
320 "[CK_DEBUG_MOE] %s finite=%zu/%zu nan=%zu inf=%zu min=%g max=%g\n",
321 name,
322 finite,
323 n,
324 nan,
325 inf,
326 have ? min_v : 0.0f,
327 have ? max_v : 0.0f);
328}

330void moe_relu2_expert_forward_f32(const float *hidden,
331 const int *indices,
332 const float *routing_weights,
333 const float *expert_up,
334 const float *expert_down,
335 float *output,
336 int rows,
337 int hidden_dim,
338 int intermediate_dim,
339 int n_experts,
340 int top_k)
341{
342 if (!hidden || !indices || !routing_weights || !expert_up || !expert_down || !output ||
343 rows <= 0 || hidden_dim <= 0 || intermediate_dim <= 0 || n_experts <= 0 || top_k <= 0) {
344 return;
345 }
346
347 const size_t out_count = (size_t)rows * (size_t)hidden_dim;
348 for (size_t p = 0; p < out_count; ++p) output[p] = 0.0f;
349
350 float pre[intermediate_dim];
351 float act[intermediate_dim];
352
353 for (int r = 0; r < rows; ++r) {
354 const float *x = hidden + (size_t)r * (size_t)hidden_dim;
355 float *y = output + (size_t)r * (size_t)hidden_dim;
356 for (int slot = 0; slot < top_k; ++slot) {
357 const int e = indices[(size_t)r * (size_t)top_k + (size_t)slot];
358 if (e < 0 || e >= n_experts) continue;
359 const float route_w = routing_weights[(size_t)r * (size_t)top_k + (size_t)slot];
360
361 for (int i = 0; i < intermediate_dim; ++i) {
362 float v = 0.0f;
363 for (int h = 0; h < hidden_dim; ++h) {
364 v += expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] * x[h];
365 }
366 pre[i] = v;
367 act[i] = (v > 0.0f) ? v * v : 0.0f;
368 }
369
370 for (int h = 0; h < hidden_dim; ++h) {
371 float v = 0.0f;
372 for (int i = 0; i < intermediate_dim; ++i) {
373 v += expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)] * act[i];
374 }
375 y[h] += route_w * v;
376 }
377 }
378 }
379}

383static inline float ck_moe_sigmoid_f32(float x)
384{
385 return 1.0f / (1.0f + expf(-x));
386}
387
388static inline float ck_moe_silu_f32(float x)
389{
390 return x * ck_moe_sigmoid_f32(x);
391}
392
393static inline float ck_moe_dsilu_f32(float x)
394{
395 const float sig = ck_moe_sigmoid_f32(x);
396 return sig + x * sig * (1.0f - sig);
397}

509void moe_swiglu_expert_backward_f32(const float *d_output,
510 const float *hidden,
511 const int *indices,
512 const float *routing_weights,
513 const float *expert_gate,
514 const float *expert_up,
515 const float *expert_down,
516 float *d_hidden,
517 float *d_routing_weights,
518 float *d_expert_gate,
519 float *d_expert_up,
520 float *d_expert_down,
521 int rows,
522 int hidden_dim,
523 int intermediate_dim,
524 int n_experts,
525 int top_k)
526{
527 if (!d_output || !hidden || !indices || !routing_weights || !expert_gate || !expert_up || !expert_down ||
528 !d_hidden || !d_routing_weights || !d_expert_gate || !d_expert_up || !d_expert_down ||
529 rows <= 0 || hidden_dim <= 0 || intermediate_dim <= 0 || n_experts <= 0 || top_k <= 0) {
530 return;
531 }
532
533 for (size_t p = 0; p < (size_t)rows * (size_t)hidden_dim; ++p) d_hidden[p] = 0.0f;
534 for (size_t p = 0; p < (size_t)rows * (size_t)top_k; ++p) d_routing_weights[p] = 0.0f;
535 for (size_t p = 0; p < (size_t)n_experts * (size_t)intermediate_dim * (size_t)hidden_dim; ++p) {
536 d_expert_gate[p] = 0.0f;
537 d_expert_up[p] = 0.0f;
538 }
539 for (size_t p = 0; p < (size_t)n_experts * (size_t)hidden_dim * (size_t)intermediate_dim; ++p) d_expert_down[p] = 0.0f;
540
541 float gate[intermediate_dim];
542 float up[intermediate_dim];
543 float silu_gate[intermediate_dim];
544 float act[intermediate_dim];
545 float d_act[intermediate_dim];
546 float expert_out[hidden_dim];
547
548 for (int r = 0; r < rows; ++r) {
549 const float *x = hidden + (size_t)r * (size_t)hidden_dim;
550 const float *dy = d_output + (size_t)r * (size_t)hidden_dim;
551 float *dx = d_hidden + (size_t)r * (size_t)hidden_dim;
552
553 for (int slot = 0; slot < top_k; ++slot) {
554 const int e = indices[(size_t)r * (size_t)top_k + (size_t)slot];
555 if (e < 0 || e >= n_experts) continue;
556 const float route_w = routing_weights[(size_t)r * (size_t)top_k + (size_t)slot];
557
558 for (int i = 0; i < intermediate_dim; ++i) {
559 float gv = 0.0f;
560 float uv = 0.0f;
561 for (int h = 0; h < hidden_dim; ++h) {
562 gv += expert_gate[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] * x[h];
563 uv += expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] * x[h];
564 }
565 gate[i] = gv;
566 up[i] = uv;
567 silu_gate[i] = ck_moe_silu_f32(gv);
568 act[i] = silu_gate[i] * uv;
569 d_act[i] = 0.0f;
570 }
571
572 for (int h = 0; h < hidden_dim; ++h) {
573 float v = 0.0f;
574 for (int i = 0; i < intermediate_dim; ++i) {
575 v += expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)] * act[i];
576 }
577 expert_out[h] = v;
578 }
579
580 float d_route = 0.0f;
581 for (int h = 0; h < hidden_dim; ++h) {
582 const float d_expert_out = dy[h] * route_w;
583 d_route += dy[h] * expert_out[h];
584 for (int i = 0; i < intermediate_dim; ++i) {
585 d_expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)] += d_expert_out * act[i];
586 d_act[i] += d_expert_out * expert_down[ck_moe_down_idx(e, h, i, hidden_dim, intermediate_dim)];
587 }
588 }
589 d_routing_weights[(size_t)r * (size_t)top_k + (size_t)slot] += d_route;
590
591 for (int i = 0; i < intermediate_dim; ++i) {
592 const float d_up = d_act[i] * silu_gate[i];
593 const float d_gate = d_act[i] * up[i] * ck_moe_dsilu_f32(gate[i]);
594 for (int h = 0; h < hidden_dim; ++h) {
595 d_expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] += d_up * x[h];
596 d_expert_gate[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] += d_gate * x[h];
597 dx[h] += d_up * expert_up[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)] +
598 d_gate * expert_gate[ck_moe_up_idx(e, i, h, intermediate_dim, hidden_dim)];
599 }
600 }
601 }
602 }
603}

605void moe_swiglu_shared_forward_f32(const float *hidden,
606 const float *routed,
607 const float *shared_gate,
608 const float *shared_up,
609 const float *shared_down,
610 float *output,
611 int rows,
612 int hidden_dim,
613 int intermediate_dim)
614{
615 if (!hidden || !shared_gate || !shared_up || !shared_down || !output || rows <= 0 || hidden_dim <= 0 || intermediate_dim <= 0) {
616 return;
617 }
618
619 float gate[intermediate_dim];
620 float up[intermediate_dim];
621 float act[intermediate_dim];
622
623 for (int r = 0; r < rows; ++r) {
624 const float *x = hidden + (size_t)r * (size_t)hidden_dim;
625 const float *route = routed ? (routed + (size_t)r * (size_t)hidden_dim) : NULL;
626 float *y = output + (size_t)r * (size_t)hidden_dim;
627 for (int i = 0; i < intermediate_dim; ++i) {
628 float gv = 0.0f;
629 float uv = 0.0f;
630 for (int h = 0; h < hidden_dim; ++h) {
631 gv += shared_gate[(size_t)i * (size_t)hidden_dim + (size_t)h] * x[h];
632 uv += shared_up[(size_t)i * (size_t)hidden_dim + (size_t)h] * x[h];
