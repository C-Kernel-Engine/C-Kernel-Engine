24#define CK_AUDIO_CONV1D_COMPUTE_BODY \
25 for (size_t oc = job->channel_begin; oc < job->channel_end; ++oc) { \
26 for (size_t frame = 0; frame < job->output_frames; ++frame) { \
27 float sum = job->bias[oc]; \
28 for (size_t ic = 0; ic < job->input_channels; ++ic) { \
29 const float *row = job->weight + \

55/* Output frames are independent. SIMD lanes retain the input-channel/tap
56 * reduction order of the scalar FMA provider; padding frames use its exact
57 * scalar path, so no zero-padded arithmetic is introduced. */
58__attribute__((target("avx2,fma"), noinline))
59void audio_conv1d_compute_avx2(const ck_audio_conv1d_job *job, void *unused) {
60 (void)unused;
61 if (job->stride != 1 || job->output_frames < 8) {
62 audio_conv1d_compute_hwfma(job, NULL);
63 return;
64 }
65 const size_t final_tap = (job->kernel_size - 1) * job->dilation;
66 for (size_t oc = job->channel_begin; oc < job->channel_end; ++oc) {
67 size_t frame = 0;
68 for (; frame < job->output_frames;) {
69 const int64_t first = (int64_t)frame - (int64_t)job->padding;
70 const int64_t last = first + 7 + (int64_t)final_tap;
71 if (frame + 8 <= job->output_frames && first >= 0 &&
72 last < (int64_t)job->input_frames) {
73 __m256 sums = _mm256_set1_ps(job->bias[oc]);
74 for (size_t ic = 0; ic < job->input_channels; ++ic) {
75 const float *row = job->weight +
76 (oc * job->input_channels + ic) * job->kernel_size;
77 const float *input = job->input + ic * job->input_stride +
78 (size_t)first;
79 for (size_t tap = 0; tap < job->kernel_size; ++tap) {
80 const __m256 values = _mm256_loadu_ps(input + tap * job->dilation);
81 sums = _mm256_fmadd_ps(values, _mm256_set1_ps(row[tap]), sums);
82 }
83 }
84 _mm256_storeu_ps(job->staged + oc * job->output_frames + frame, sums);
85 frame += 8;
86 } else {
87 float sum = job->bias[oc];
88 for (size_t ic = 0; ic < job->input_channels; ++ic) {
89 const float *row = job->weight +
90 (oc * job->input_channels + ic) * job->kernel_size;
91 for (size_t tap = 0; tap < job->kernel_size; ++tap) {
92 const int64_t source = (int64_t)frame +
93 (int64_t)tap * (int64_t)job->dilation -
94 (int64_t)job->padding;
95 if (source >= 0 && source < (int64_t)job->input_frames)
96 sum = fmaf(job->input[ic * job->input_stride +
97 (size_t)source], row[tap], sum);
98 }
99 }
100 job->staged[oc * job->output_frames + frame] = sum;
101 ++frame;
102 }
103 }
104 }

127int audio_conv1d_checked_workspace(size_t input_channels, size_t output_channels,
128 size_t input_frames, size_t output_frames, size_t *scratch_elements) {
129 size_t staged;
130 if (!scratch_elements || !input_channels || !output_channels ||
131 !input_frames || !output_frames ||
132 input_channels > SIZE_MAX / input_frames ||
133 output_channels > SIZE_MAX / output_frames)
134 return CK_AUDIO_CONV_CHECKED_OVERFLOW;
135 staged = output_channels * output_frames;
136 if (staged > SIZE_MAX / sizeof(float))
137 return CK_AUDIO_CONV_CHECKED_OVERFLOW;
138 *scratch_elements = staged;
139 return CK_AUDIO_CONV_CHECKED_OK;

142int ck_audio_conv1d_checked_execute(
143 const float *input, size_t input_elements, size_t input_stride,
144 const float *weight, size_t weight_elements,
145 const float *bias, size_t bias_elements,
146 float *output, size_t output_elements, size_t output_stride,
147 float *scratch, size_t scratch_elements,
148 size_t input_channels, size_t output_channels, size_t input_frames,
149 size_t kernel_size, size_t stride, size_t padding, size_t dilation,
150 size_t output_frames,
151 ck_audio_conv1d_compute_fn compute, void *compute_context) {
152 size_t input_span, output_span, weight_count, scratch_need;
153 if (!input_channels || !output_channels || !input_frames ||
154 !kernel_size || !stride || !dilation || !output_frames ||
155 input_channels > INT_MAX || output_channels > INT_MAX ||
156 input_frames > INT_MAX || kernel_size > INT_MAX ||
157 stride > INT_MAX || padding > INT_MAX || dilation > INT_MAX ||
158 output_frames > INT_MAX)
159 return CK_AUDIO_CONV_CHECKED_INVALID;
160 const int64_t effective_kernel = 1 +
161 ((int64_t)kernel_size - 1) * (int64_t)dilation;
162 const int64_t padded = (int64_t)input_frames +
163 2 * (int64_t)padding - effective_kernel;
164 if (padded < 0 || padded / stride + 1 != output_frames)
165 return CK_AUDIO_CONV_CHECKED_INVALID;
166 if (!ck_checked_span((size_t)input_channels, (size_t)input_frames,
167 input_stride, &input_span) ||
168 !ck_checked_span((size_t)output_channels, (size_t)output_frames,
169 output_stride, &output_span) ||
170 audio_conv1d_checked_workspace((size_t)input_channels,
171 (size_t)output_channels, (size_t)input_frames,
172 (size_t)output_frames, &scratch_need))
173 return CK_AUDIO_CONV_CHECKED_OVERFLOW;
174 if ((size_t)output_channels > SIZE_MAX / (size_t)input_channels ||
175 (size_t)output_channels * (size_t)input_channels > SIZE_MAX / (size_t)kernel_size)
176 return CK_AUDIO_CONV_CHECKED_OVERFLOW;
177 weight_count = (size_t)output_channels * (size_t)input_channels * (size_t)kernel_size;
178 if (weight_count > SIZE_MAX / sizeof(float)) return CK_AUDIO_CONV_CHECKED_OVERFLOW;
179 if (input_elements < input_span || weight_elements < weight_count ||
180 bias_elements < (size_t)output_channels || output_elements < output_span ||
181 scratch_elements < scratch_need)
182 return CK_AUDIO_CONV_CHECKED_CAPACITY;
183 if (!ck_checked_region(input, input_span) ||
184 !ck_checked_region(weight, weight_count) ||
185 !ck_checked_region(bias, (size_t)output_channels) ||
186 !ck_checked_region(output, output_span) ||
187 !ck_checked_region(scratch, scratch_need))
188 return CK_AUDIO_CONV_CHECKED_INVALID;
189 if (ck_checked_overlap(output, output_span, input, input_span) ||
190 ck_checked_overlap(output, output_span, weight, weight_count) ||
191 ck_checked_overlap(output, output_span, bias, (size_t)output_channels) ||
192 ck_checked_overlap(output, output_span, scratch, scratch_need) ||
193 ck_checked_overlap(scratch, scratch_need, input, input_span) ||
194 ck_checked_overlap(scratch, scratch_need, weight, weight_count) ||
195 ck_checked_overlap(scratch, scratch_need, bias, (size_t)output_channels))
196 return CK_AUDIO_CONV_CHECKED_INVALID;
197 if (!ck_checked_finite(weight, weight_count) ||
198 !ck_checked_finite(bias, (size_t)output_channels))
199 return CK_AUDIO_CONV_CHECKED_NONFINITE;
200 for (size_t channel = 0; channel < input_channels; ++channel)
201 if (!ck_checked_finite(input + (size_t)channel * input_stride,
202 (size_t)input_frames))
203 return CK_AUDIO_CONV_CHECKED_NONFINITE;
204 /* Validation and staged commit remain shared by both providers. */
205 const ck_audio_conv1d_job job = {input, weight, bias, scratch,
206 input_stride, input_channels, output_channels, input_frames,
207 kernel_size, stride, padding, dilation, output_frames, 0,
208 output_channels};
209 compute(&job, compute_context);
210 if (!ck_checked_finite(scratch, (size_t)output_channels * (size_t)output_frames))
211 return CK_AUDIO_CONV_CHECKED_NONFINITE;
212 for (size_t channel = 0; channel < output_channels; ++channel)
213 memcpy(output + (size_t)channel * output_stride,
214 scratch + (size_t)channel * (size_t)output_frames,
215 (size_t)output_frames * sizeof(float));
216 return CK_AUDIO_CONV_CHECKED_OK;
217}
