25enum { CHUNK = 64, HEAD_WIDTH = 128, MATRIX = CHUNK * HEAD_WIDTH };

50static void bind_math(void)
51{
52 const char *library = getenv("CK_MKL_LIBRARY");
53 void *handle = library && *library ? dlopen(library, RTLD_NOW | RTLD_LOCAL) : RTLD_DEFAULT;
54 if (!handle && library && *library)
55 return;
56 sgemm = (sgemm_fn)dlsym(handle, "sgemm_");
57 vsexp = (vsexp_fn)dlsym(handle, "vsExp");

60static void matmul(int rows, int columns, int depth, const float *a, const float *b,
61 int transpose_a, int transpose_b, float *c)
62{
63 const char ta = transpose_a ? 'T' : 'N';
64 const char tb = transpose_b ? 'T' : 'N';
65 const int lda = transpose_a ? rows : depth;
66 const int ldb = transpose_b ? depth : columns;
67 const float one = 1.0f, zero = 0.0f;
68 /* Fortran BLAS uses column-major storage. Exchange operands and axes. */
69 sgemm(&tb, &ta, &columns, &rows, &depth, &one, b, &ldb, a, &lda,
70 &zero, c, &columns);
71}

78static float reduce_previous_rows(const float *matrix, int row, int column)
79{
80 const float *coefficients = matrix + row * CHUNK;
81 if ((row < 8 && column < (row / 4) * 4) || (row >= 32 && column < 32)) {
82 float blocks = 0.0f, sum = 0.0f;
83 for (int i = 0; i < row; ++i) {
84 if (i > column)
85 sum += coefficients[i] * matrix[i * CHUNK + column];
86 if (i % 16 == 15) {
87 blocks += sum;
88 sum = 0.0f;
89 }
90 }
91 return sum + blocks;
92 }
93
94 /* CPU outer sum uses four interleaved accumulators for the remaining columns. */
95 float partials[4] = {0};
96 const int complete = (row / 4) * 4;
97 for (int i = column + 1; i < complete; ++i)
98 partials[i % 4] += coefficients[i] * matrix[i * CHUNK + column];
99 for (int i = complete > column ? complete : column + 1; i < row; ++i)
100 partials[0] += coefficients[i] * matrix[i * CHUNK + column];
101 return ((partials[0] + partials[1]) + partials[2]) + partials[3];
102}

104static void temporal_transform(chunk_workspace *w)
105{
106 vsexp(CHUNK, w->cumulative_gate, w->gate_exp);
107 for (int row = 0; row < CHUNK; ++row) {
108 w->final_delta[row] = w->cumulative_gate[CHUNK - 1] - w->cumulative_gate[row];
109 for (int column = 0; column < CHUNK; ++column)
110 w->decay_input[row * CHUNK + column] = column <= row
111 ? w->cumulative_gate[row] - w->cumulative_gate[column] : 0.0f;
112 }
113 vsexp(TEMPORAL, w->decay_input, w->decay);
114 vsexp(CHUNK, w->final_delta, w->final_exp);
115 matmul(CHUNK, CHUNK, HEAD_WIDTH, w->key_beta, w->key, 0, 1, w->transform);
116
117 for (int row = 0; row < CHUNK; ++row) {
118 for (int column = 0; column < CHUNK; ++column) {
119 const int offset = row * CHUNK + column;
120 if (column > row)
121 w->decay[offset] = 0.0f;
122 w->transform[offset] = column < row
123 ? -w->transform[offset] * w->decay[offset] : 0.0f;
124 }
125 }
126 /* Updating columns left to right keeps each row coefficient needed for
127 later columns intact until its final use. */
128 for (int row = 1; row < CHUNK; ++row)
129 for (int column = 0; column < row; ++column)
130 w->transform[row * CHUNK + column] += reduce_previous_rows(w->transform, row, column);
131 for (int row = 0; row < CHUNK; ++row)
132 w->transform[row * CHUNK + row] = 1.0f;
133}

135static void apply_chunk(chunk_workspace *w, float *state)
136{
137 temporal_transform(w);
138 for (int row = 0; row < CHUNK; ++row) {
139 for (int d = 0; d < HEAD_WIDTH; ++d) {
140 const int at = row * HEAD_WIDTH + d;
141 w->weighted_key[at] = w->key_beta[at] * w->gate_exp[row];
142 w->weighted_query[at] = w->query[at] * w->gate_exp[row];
143 w->final_key[at] = w->key[at] * w->final_exp[row];
144 }
145 }
146 matmul(CHUNK, HEAD_WIDTH, CHUNK, w->transform, w->value_beta, 0, 0, w->value);
147 matmul(CHUNK, HEAD_WIDTH, CHUNK, w->transform, w->weighted_key, 0, 0, w->cumulative_key);
148 matmul(CHUNK, HEAD_WIDTH, HEAD_WIDTH, w->cumulative_key, state, 0, 0, w->previous_value);
149 for (int i = 0; i < MATRIX; ++i)
150 w->value_new[i] = w->value[i] - w->previous_value[i];
151
152 matmul(CHUNK, HEAD_WIDTH, HEAD_WIDTH, w->weighted_query, state, 0, 0, w->history);
153 matmul(CHUNK, CHUNK, HEAD_WIDTH, w->query, w->key, 0, 1, w->query_key);
154 for (int i = 0; i < TEMPORAL; ++i)
155 w->query_key[i] *= w->decay[i];
156 matmul(CHUNK, HEAD_WIDTH, CHUNK, w->query_key, w->value_new, 0, 0, w->local);
157
158 matmul(HEAD_WIDTH, HEAD_WIDTH, CHUNK, w->final_key, w->value_new, 1, 0, w->update);
159 for (int i = 0; i < STATE; ++i)
160 state[i] = state[i] * w->gate_exp[CHUNK - 1] + w->update[i];
161}

163void gated_deltanet_chunked_bf16_prefill_forward(
164 const float *q, const float *k, const float *v, const float *packed_qkv,
165 const float *g, const float *beta, const float *state_in, float *state_out,
166 float *out, void *workspace, int rows, int heads, int groups, int width, float eps)
167{
168 pthread_once(&bind_once, bind_math);
169 if (!sgemm || !vsexp || !q || !k || !v || !packed_qkv || !g || !beta ||
170 !state_in || !state_out || !out || !workspace || rows < 1 || heads < 1 ||
171 groups < 1 || heads % groups || width != HEAD_WIDTH) {
172 fprintf(stderr, "HARD KERNEL CONTRACT FAULT: chunked BF16 DeltaNet requires MKL sgemm_ and vsExp, 128-wide heads and valid buffers\n");
173 abort();
174 }
175
176 chunk_workspace *w = workspace;
177 const int repeat = heads / groups;
178 const size_t compact_row = (size_t)groups * HEAD_WIDTH;
179 const size_t value_row = (size_t)heads * HEAD_WIDTH;
180 const size_t packed_row = 2 * compact_row + value_row;
181 const size_t vector_limit = (size_t)rows * heads / 32 * 32;
182 const size_t boundary_row = vector_limit / heads;
183 const int boundary_head = (int)(vector_limit % heads);
184 const int boundary_group = boundary_head / repeat;
185 const int mixed_group = boundary_head % repeat;
186 const float scale = (float)(1.0 / sqrt((double)HEAD_WIDTH));
187 float scalar_q[HEAD_WIDTH], scalar_k[HEAD_WIDTH];
188 if (mixed_group) {
189 const float *raw = packed_qkv + boundary_row * packed_row + (size_t)boundary_group * HEAD_WIDTH;
190 memcpy(scalar_q, raw, sizeof(scalar_q));
191 memcpy(scalar_k, raw + compact_row, sizeof(scalar_k));
192 /* The repeated group can straddle the vector end of BF16 rsqrt. */
193 recurrent_qk_l2_norm_pytorch_bf16_storage(
194 scalar_q, scalar_k, 1, HEAD_WIDTH, HEAD_WIDTH, 1, HEAD_WIDTH, eps);
195 }
196
197 for (int head = 0; head < heads; ++head) {
198 const int group = head / repeat;
199 const size_t compact_offset = (size_t)group * HEAD_WIDTH;
200 const size_t head_offset = (size_t)head * HEAD_WIDTH;
201 const float *initial = state_in + (size_t)head * STATE;
202 float *state = state_out + (size_t)head * STATE;
203 if (state != initial)
204 memcpy(state, initial, STATE * sizeof(float));
205 for (int start = 0; start < rows;) {
206 const int valid = rows - start < CHUNK ? rows - start : CHUNK;
207 double cumulative = 0.0;
208 for (int row = 0; row < CHUNK; ++row) {
209 const size_t token = (size_t)start + row;
210 const int present = row < valid;
211 const size_t gate_index = token * heads + head;
212 if (present)
213 cumulative += g[gate_index];
214 w->cumulative_gate[row] = (float)cumulative;
215 const float factor = present
216 ? bf16_round(1.0f / (1.0f + expf(-beta[gate_index]))) : 0.0f;
217 const int scalar = mixed_group && token == boundary_row &&
218 group == boundary_group && head >= boundary_head;
219 for (int d = 0; d < HEAD_WIDTH; ++d) {
220 const int at = row * HEAD_WIDTH + d;
221 const size_t compact = token * compact_row + compact_offset + d;
222 const size_t expanded = token * value_row + head_offset + d;
223 w->query[at] = present ? (scalar ? scalar_q[d] : q[compact]) * scale : 0.0f;
224 w->key[at] = present ? (scalar ? scalar_k[d] : k[compact]) : 0.0f;
225 w->value_beta[at] = present ? v[expanded] * factor : 0.0f;
226 w->key_beta[at] = w->key[at] * factor;
227 }
228 }
229 apply_chunk(w, state);
230 for (int row = 0; row < valid; ++row) {
231 float *destination = out + ((size_t)start + row) * value_row + head_offset;
232 for (int d = 0; d < HEAD_WIDTH; ++d) {
233 const int at = row * HEAD_WIDTH + d;
234 destination[d] = bf16_round(w->history[at] + w->local[at]);
235 }
236 }
237 start += valid;
238 }
239 }
