8/* Compare byte ranges without undefined relational comparisons of pointers. */
9static int overlaps(const void *a, size_t an, const void *b, size_t bn) {
10 uintptr_t x = (uintptr_t)a, y = (uintptr_t)b;
11 if (an > UINTPTR_MAX - x || bn > UINTPTR_MAX - y) return 1;
12 return x < y + bn && y < x + an;
13}

15int ck_rwkv7_token_shift_lerp(const float *x,
16 const float *shift_in,
17 const float *mu,
18 float *xm,
19 float *shift_out,
20 int dim) {
21 if (!x || !shift_in || !mu || !xm || !shift_out || dim <= 0) {
22 return -1;
23 }
24 size_t bytes = (size_t)dim * sizeof(float);
25 if (overlaps(xm, bytes, shift_out, bytes)) return -1;
26 const float *inputs[] = {x, shift_in, mu};
27 for (int j = 0; j < 3; ++j) {
28 if ((xm != inputs[j] && overlaps(xm, bytes, inputs[j], bytes)) ||
29 (shift_out != inputs[j] && overlaps(shift_out, bytes, inputs[j], bytes))) return -1;
30 }
31 for (int i = 0; i < dim; ++i) {
32 const float previous = shift_in[i];
33 const float current = x[i];
34 const float m = mu[i];
35 xm[i] = current + m * (previous - current);
36 shift_out[i] = current;
37 }
38 return 0;
39}

41int ck_rwkv7_norm_kk(const float *kk_in,
42 float *kk_out,
43 int num_heads,
44 int head_dim) {
45 if (!kk_in || !kk_out || num_heads <= 0 || head_dim <= 0 ||
46 num_heads > 256 || head_dim > 512) {
47 return -1;
48 }
49 for (int h = 0; h < num_heads; ++h) {
50 const float *src = kk_in + (size_t)h * (size_t)head_dim;
51 float *dst = kk_out + (size_t)h * (size_t)head_dim;
52 float norm = 0.0f;
53 for (int i = 0; i < head_dim; ++i) {
54 norm += src[i] * src[i];
55 }
56 norm = sqrtf(norm);
57 /* Divisor max(norm,1e-12); an all-zero row stays zero. */
58 if (norm < 1e-12f) {
59 norm = 1e-12f;
60 }
61 const float inv = 1.0f / norm;
62 for (int i = 0; i < head_dim; ++i) {
63 dst[i] = src[i] * inv;
64 }
65 }
66 return 0;
67}

87int ck_rwkv7_lora_gates(const float *xw,
88 const float *xa,
89 const float *xg,
90 const float *w1,
91 const float *w2,
92 const float *w0,
93 const float *a1,
94 const float *a2,
95 const float *a0,
96 const float *g1,
97 const float *g2,
98 int dim,
99 int rank_w,
100 int rank_a,
101 int rank_g,
102 float *w_log,
103 float *a_out,
104 float *g_out) {
105 if (!xw || !xa || !xg || !w1 || !w2 || !w0 || !a1 || !a2 || !a0 || !g1 ||
106 !g2 || !w_log || !a_out || !g_out || dim <= 0 || rank_w <= 0 ||
107 rank_a <= 0 || rank_g <= 0 || rank_w > 512 || rank_a > 512 ||
108 rank_g > 512) {
109 return -1;
110 }
111 /* w: tanh between factors, sigmoid after (+w0), scaled by -e^-0.5. */
112 {
113 float tmp[512];
114 for (int r = 0; r < rank_w; ++r) {
115 float acc = 0.0f;
116 for (int i = 0; i < dim; ++i) {
117 acc += xw[i] * w1[(size_t)i * (size_t)rank_w + (size_t)r];
118 }
119 tmp[r] = tanhf(acc);
120 }
121 for (int i = 0; i < dim; ++i) {
122 float acc = w0[i];
123 for (int r = 0; r < rank_w; ++r) {
124 acc += tmp[r] * w2[(size_t)r * (size_t)dim + (size_t)i];
125 }
126 w_log[i] = -0.6065306597126334f * ck_rwkv7_sigmoid(acc);
127 }
128 }
129 /* a: no activation between, sigmoid after (+a0). */
130 {
131 float tmp[512];
132 for (int r = 0; r < rank_a; ++r) {
133 float acc = 0.0f;
134 for (int i = 0; i < dim; ++i) {
135 acc += xa[i] * a1[(size_t)i * (size_t)rank_a + (size_t)r];
136 }
137 tmp[r] = acc;
138 }
139 for (int i = 0; i < dim; ++i) {
140 float acc = a0[i];
141 for (int r = 0; r < rank_a; ++r) {
142 acc += tmp[r] * a2[(size_t)r * (size_t)dim + (size_t)i];
143 }
144 a_out[i] = ck_rwkv7_sigmoid(acc);
145 }
146 }
147 /* g: sigmoid between factors, nothing after, no bias. */
148 {
149 float tmp[512];
150 for (int r = 0; r < rank_g; ++r) {
151 float acc = 0.0f;
152 for (int i = 0; i < dim; ++i) {
153 acc += xg[i] * g1[(size_t)i * (size_t)rank_g + (size_t)r];
154 }
155 tmp[r] = ck_rwkv7_sigmoid(acc);
156 }
157 for (int i = 0; i < dim; ++i) {
158 float acc = 0.0f;
159 for (int r = 0; r < rank_g; ++r) {
160 acc += tmp[r] * g2[(size_t)r * (size_t)dim + (size_t)i];
161 }
162 g_out[i] = acc;
163 }
164 }
165 return 0;
166}

168int ck_rwkv7_vgate_vmix(const float *v_in,
169 const float *v_first,
170 const float *xv,
171 const float *v1,
172 const float *v2,
173 const float *v0,
174 int dim,
175 int rank_v,
176 float *v_out,
177 float *v_first_out,
178 float *v_gate) {
179 if (!v_in || !v_out || !v_first_out || dim <= 0) {
180 return -1;
181 }
182 if (xv == NULL) {
183 /* Layer 0: produce v_first, no gate read. */
184 for (int i = 0; i < dim; ++i) {
185 v_out[i] = v_in[i];
186 v_first_out[i] = v_in[i];
187 }
188 return 0;
189 }
190 if (!v_first || !v1 || !v2 || !v0 || rank_v <= 0 || rank_v > 512) {
191 return -1;
192 }
193 float tmp[512];
194 for (int r = 0; r < rank_v; ++r) {
195 float acc = 0.0f;
196 for (int i = 0; i < dim; ++i) {
197 acc += xv[i] * v1[(size_t)i * (size_t)rank_v + (size_t)r];
198 }
199 tmp[r] = acc;
200 }
201 for (int i = 0; i < dim; ++i) {
202 float acc = v0[i];
203 for (int r = 0; r < rank_v; ++r) {
204 acc += tmp[r] * v2[(size_t)r * (size_t)dim + (size_t)i];
205 }
206 const float gate = ck_rwkv7_sigmoid(acc);
207 if (v_gate) {
208 v_gate[i] = gate;
209 }
210 v_out[i] = v_in[i] + (v_first[i] - v_in[i]) * gate;
211 v_first_out[i] = v_first[i];
212 }
213 return 0;
214}

216int ck_rwkv7_state_step(const float *w_log,
217 const float *k,
218 const float *v,
219 const float *kk,
220 const float *a,
221 const float *state_in,
222 float *state_out,
223 int head_dim) {
224 if (!w_log || !k || !v || !kk || !a || !state_in || !state_out ||
225 head_dim <= 0 || head_dim > 512) {
226 return -1;
227 }
228 float decay[512];
229 float bvec[512];
230 for (int j = 0; j < head_dim; ++j) {
231 decay[j] = expf(w_log[j]);
232 bvec[j] = kk[j] * a[j];
233 }
234 for (int i = 0; i < head_dim; ++i) {
235 float t = 0.0f;
236 for (int l = 0; l < head_dim; ++l) {
237 t += state_in[(size_t)i * (size_t)head_dim + (size_t)l] * kk[l];
238 }
239 for (int j = 0; j < head_dim; ++j) {
240 state_out[(size_t)i * (size_t)head_dim + (size_t)j] =
241 state_in[(size_t)i * (size_t)head_dim + (size_t)j] * decay[j] -
242 t * bvec[j] + v[i] * k[j];
243 }
244 }
245 return 0;
246}

265int ck_rwkv7_wkv_decode(const float *r,
266 const float *w_log,
267 const float *k,
268 const float *v,
269 const float *kk,
270 const float *a,
271 const float *g,
272 const float *rk,
273 const float *ln_w,
274 const float *ln_b,
275 float norm_eps,
276 const float *state_in,
277 float *state_out,
278 float *y_out,
279 int num_heads,
280 int head_dim) {
281 if (!r || !w_log || !k || !v || !kk || !a || !g || !rk || !ln_w || !ln_b ||
282 !state_in || !state_out || !y_out || num_heads <= 0 || head_dim <= 0 ||
283 !(norm_eps > 0.0f) || !isfinite(norm_eps) ||
284 !isfinite(norm_eps * (float)head_dim)) {
285 return -1;
286 }
287 if (num_heads > 256 || head_dim > 512) {
288 return -1;
289 }
290
291 /* Per-head generalised delta-rule step + raw readout (pre-norm). */
292 for (int h = 0; h < num_heads; ++h) {
293 const float *rh = r + (size_t)h * (size_t)head_dim;
294 const float *kh = k + (size_t)h * (size_t)head_dim;
295 const float *vh = v + (size_t)h * (size_t)head_dim;
296 const float *wh = w_log + (size_t)h * (size_t)head_dim;
297 const float *khh = kk + (size_t)h * (size_t)head_dim;
298 const float *ah = a + (size_t)h * (size_t)head_dim;
299 const float *sin = state_in + (size_t)h * (size_t)head_dim * (size_t)head_dim;
300 float *sout = state_out + (size_t)h * (size_t)head_dim * (size_t)head_dim;
301 float *yh = y_out + (size_t)h * (size_t)head_dim;
302
303 if (ck_rwkv7_state_step(wh, kh, vh, khh, ah, sin, sout, head_dim) != 0 ||
304 ck_rwkv7_state_readout(sout, rh, yh, head_dim) != 0) {
305 return -1;
306 }
307 }
308
309 /* GroupNorm over C with H groups, eps = norm_eps*head_dim. */
310 const float eps = norm_eps * (float)head_dim;
311 for (int h = 0; h < num_heads; ++h) {
312 float *yh = y_out + (size_t)h * (size_t)head_dim;
313 const float *lwh = ln_w + (size_t)h * (size_t)head_dim;
314 const float *lbh = ln_b + (size_t)h * (size_t)head_dim;
315 float mean = 0.0f;
316 for (int i = 0; i < head_dim; ++i) {
317 mean += yh[i];
318 }
319 mean /= (float)head_dim;
320 float var = 0.0f;
321 for (int i = 0; i < head_dim; ++i) {
322 const float d = yh[i] - mean;
323 var += d * d;
324 }
325 var /= (float)head_dim;
326 const float inv = 1.0f / sqrtf(var + eps);
327 for (int i = 0; i < head_dim; ++i) {
328 yh[i] = (yh[i] - mean) * inv * lwh[i] + lbh[i];
329 }
330 }
331
332 /* Per-head bonus s_h = sum(r_h*k_h*rk_h); y += s_h*v; then y *= g. */
333 const int dim = num_heads * head_dim;
334 for (int h = 0; h < num_heads; ++h) {
335 const float *rh = r + (size_t)h * (size_t)head_dim;
336 const float *kh = k + (size_t)h * (size_t)head_dim;
337 const float *vh = v + (size_t)h * (size_t)head_dim;
338 const float *rkh = rk + (size_t)h * (size_t)head_dim;
339 float *yh = y_out + (size_t)h * (size_t)head_dim;
340 float s = 0.0f;
341 for (int i = 0; i < head_dim; ++i) {
342 s += rh[i] * kh[i] * rkh[i];
343 }
344 for (int i = 0; i < head_dim; ++i) {
345 yh[i] += s * vh[i];
346 }
347 }
348 for (int i = 0; i < dim; ++i) {
349 y_out[i] *= g[i];
350 }
351 return 0;
352}

361int ck_rwkv7_channelmix_decode(const float *x,
362 const float *shift_in,
363 const float *mu,
364 const float *Wk,
365 const float *Wv,
366 float *shift_out,
367 float *out,
368 int dim,
369 int hidden_dim,
370 float *scratch,
371 size_t scratch_elems) {
372 if (!x || !shift_in || !mu || !Wk || !Wv || !shift_out || !out || dim <= 0 ||
373 hidden_dim <= 0 || hidden_dim > 8192) {
374 return -1;
375 }
376 if (shift_out == shift_in) {
377 return -1;
378 }
379 if (!scratch ||
380 scratch_elems < (size_t)dim + (size_t)hidden_dim ||
381 scratch == (const float *)x || scratch == (const float *)shift_in ||
382 scratch == (const float *)mu || scratch == (const float *)Wk ||
383 scratch == (const float *)Wv || scratch == (const float *)shift_out ||
384 scratch == (const float *)out) {
385 return -1;
386 }
387 const size_t bytes = (size_t)dim * sizeof(float);
388 if ((size_t)hidden_dim > SIZE_MAX / bytes) return -1;
389 const size_t weight_bytes = bytes * (size_t)hidden_dim;
390 const size_t scratch_bytes = ((size_t)dim + (size_t)hidden_dim) * sizeof(float);
391 if (overlaps(shift_out, bytes, out, bytes)) return -1;
392 const float *inputs[] = {x, shift_in, mu, Wk, Wv};
393 const size_t sizes[] = {bytes, bytes, bytes, weight_bytes, weight_bytes};
394 for (int j = 0; j < 5; ++j) {
395 if (overlaps(scratch, scratch_bytes, inputs[j], sizes[j])) return -1;
396 if (!(j == 0 && shift_out == x) &&
397 overlaps(shift_out, bytes, inputs[j], sizes[j])) return -1;
398 if (!(j == 0 && out == x) &&
399 overlaps(out, bytes, inputs[j], sizes[j])) return -1;
400 }
401 if (overlaps(scratch, scratch_bytes, out, bytes) ||
402 overlaps(scratch, scratch_bytes, shift_out, bytes)) return -1;
403 /* Hoisted token-shift mix; per-index read-before-write keeps
404 * shift_out == x aliasing safe. */
405 float *xk = scratch;
406 float *hidden = scratch + dim;
407 for (int i = 0; i < dim; ++i) {
408 xk[i] = x[i] + mu[i] * (shift_in[i] - x[i]);
409 shift_out[i] = x[i];
410 }
411 /* hidden[j] = relu(Wk[j,:] @ xk)^2, Wk [K,C] row-major. */
412 for (int j = 0; j < hidden_dim; ++j) {
413 const float *row = Wk + (size_t)j * (size_t)dim;
414 float acc = 0.0f;
415 for (int i = 0; i < dim; ++i) {
416 acc += row[i] * xk[i];
417 }
418 const float rel = acc > 0.0f ? acc : 0.0f;
419 hidden[j] = rel * rel;
420 }
421 /* out = Wv @ hidden, Wv [C,K] row-major; runs after the last xk read,
422 * so out == x aliasing is safe. */
423 for (int i = 0; i < dim; ++i) {
424 const float *row = Wv + (size_t)i * (size_t)hidden_dim;
425 float acc = 0.0f;
426 for (int j = 0; j < hidden_dim; ++j) {
427 acc += row[j] * hidden[j];
428 }
429 out[i] = acc;
430 }
431 return 0;
432}
