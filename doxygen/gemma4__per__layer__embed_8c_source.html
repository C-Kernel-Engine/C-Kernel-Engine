24static inline float ck_gemma4_gelu(float x)
25{
26 const float c0 = 0.7978845608028654f;
27 const float c1 = 0.044715f;
28 return 0.5f * x * (1.0f + tanhf(c0 * x * (1.0f + c1 * x * x)));
29}

31static inline void ck_gemma4_unpack_q5_k_scales(const uint8_t *scales, uint8_t *sc, uint8_t *m)
32{
33 sc[0] = scales[0] & 0x3F;
34 sc[1] = scales[1] & 0x3F;
35 sc[2] = scales[2] & 0x3F;
36 sc[3] = scales[3] & 0x3F;
37
38 m[0] = scales[4] & 0x3F;
39 m[1] = scales[5] & 0x3F;
40 m[2] = scales[6] & 0x3F;
41 m[3] = scales[7] & 0x3F;
42
43 sc[4] = (scales[8] & 0x0F) | ((scales[0] >> 6) << 4);
44 sc[5] = (scales[9] & 0x0F) | ((scales[1] >> 6) << 4);
45 sc[6] = (scales[10] & 0x0F) | ((scales[2] >> 6) << 4);
46 sc[7] = (scales[11] & 0x0F) | ((scales[3] >> 6) << 4);
47
48 m[4] = (scales[8] >> 4) | ((scales[4] >> 6) << 4);
49 m[5] = (scales[9] >> 4) | ((scales[5] >> 6) << 4);
50 m[6] = (scales[10] >> 4) | ((scales[6] >> 6) << 4);
51 m[7] = (scales[11] >> 4) | ((scales[7] >> 6) << 4);
52}

54static inline uint8_t ck_gemma4_q5_k_value(const ck_gemma4_block_q5_K *block, int subblock, int i)
55{
56 const uint8_t *ql = block->qs + (subblock / 2) * 32;
57 const uint8_t low = (subblock & 1) ? (uint8_t)(ql[i] >> 4) : (uint8_t)(ql[i] & 0x0F);
58 const uint8_t high = (block->qh[i] & (uint8_t)(1u << subblock)) ? 16u : 0u;
59 return (uint8_t)(low | high);
60}

62static void ck_gemma4_dequant_q5_k_block(const ck_gemma4_block_q5_K *block, float *out)
63{
64 uint8_t sc[8];
65 uint8_t m[8];
66 ck_gemma4_unpack_q5_k_scales(block->scales, sc, m);
67 const float d = CK_FP16_TO_FP32(block->d);
68 const float dmin = CK_FP16_TO_FP32(block->dmin);
69 for (int s = 0; s < 8; ++s) {
70 const float scale = d * (float)sc[s];
71 const float minv = dmin * (float)m[s];
72 for (int i = 0; i < 32; ++i) {
73 out[s * 32 + i] = scale * (float)ck_gemma4_q5_k_value(block, s, i) - minv;
74 }
75 }
76}

78static void ck_gemma4_rmsnorm_tmp(const float *x, const float *gamma, float *out, int n, float eps)
79{
80 double ss = 0.0;
81 for (int i = 0; i < n; ++i) {
82 ss += (double)x[i] * (double)x[i];
83 }
84 const float scale = 1.0f / sqrtf((float)(ss / (double)n) + eps);
85 for (int i = 0; i < n; ++i) {
86 out[i] = x[i] * scale * gamma[i];
87 }
88}

212void gemma4_per_layer_embed_forward(float *hidden,
213 const float *per_layer_input,
214 const float *inp_gate,
215 const float *proj,
216 const float *post_norm,
217 const float *out_scale,
218 int tokens,
219 int layer,
220 int num_layers,
221 int embed_dim,
222 int per_layer_dim,
223 float eps)
224{
225 if (!hidden || !per_layer_input || !inp_gate || !proj || !post_norm ||
226 tokens <= 0 || layer < 0 || layer >= num_layers || embed_dim <= 0 || per_layer_dim != QK_K) {
227 return;
228 }
229
230 float gate_vec[QK_K];
231 float branch[4096];
232 float branch_normed[4096];
233 if (embed_dim > (int)(sizeof(branch) / sizeof(branch[0]))) {
234 return;
235 }
236
237 for (int t = 0; t < tokens; ++t) {
238 float *h = hidden + (size_t)t * (size_t)embed_dim;
239 const float *inp_vec = per_layer_input + ((size_t)t * (size_t)num_layers + (size_t)layer) * (size_t)per_layer_dim;
240
241 for (int i = 0; i < per_layer_dim; ++i) {
242 const float *row = inp_gate + (size_t)i * (size_t)embed_dim;
243 float acc = 0.0f;
244 for (int j = 0; j < embed_dim; ++j) {
245 acc += row[j] * h[j];
246 }
247 gate_vec[i] = ck_gemma4_gelu(acc) * inp_vec[i];
248 }
249
250 for (int j = 0; j < embed_dim; ++j) {
251 const float *row = proj + (size_t)j * (size_t)per_layer_dim;
252 float acc = 0.0f;
253 for (int i = 0; i < per_layer_dim; ++i) {
254 acc += row[i] * gate_vec[i];
255 }
256 branch[j] = acc;
257 }
258 ck_gemma4_rmsnorm_tmp(branch, post_norm, branch_normed, embed_dim, eps);
259 const float layer_scale = out_scale ? out_scale[0] : 1.0f;
260 for (int j = 0; j < embed_dim; ++j) {
261 h[j] = (h[j] + branch_normed[j]) * layer_scale;
262 }
263 }
