39static float ck_gemma4_bf16_dot(const uint16_t *x, const uint16_t *y, int n)
40{
41 int i = 0;
42 double sum = 0.0;
43#if defined(__AVX2__)
44 // Match ggml-cpu's BF16 AVX2 reduction for this selected numerical contract.
45 __m256 c1 = _mm256_setzero_ps();
46 __m256 c2 = _mm256_setzero_ps();
47 __m256 c3 = _mm256_setzero_ps();
48 __m256 c4 = _mm256_setzero_ps();
49#define CK_GEMMA4_LOAD_BF16(p) _mm256_castsi256_ps(_mm256_slli_epi32( \
50 _mm256_cvtepu16_epi32(_mm_loadu_si128((const __m128i *)(p))), 16))
51 for (; i + 32 <= n; i += 32) {
52 c1 = _mm256_add_ps(_mm256_mul_ps(CK_GEMMA4_LOAD_BF16(x + i),
53 CK_GEMMA4_LOAD_BF16(y + i)), c1);
54 c2 = _mm256_add_ps(_mm256_mul_ps(CK_GEMMA4_LOAD_BF16(x + i + 8),
55 CK_GEMMA4_LOAD_BF16(y + i + 8)), c2);
56 c3 = _mm256_add_ps(_mm256_mul_ps(CK_GEMMA4_LOAD_BF16(x + i + 16),
57 CK_GEMMA4_LOAD_BF16(y + i + 16)), c3);
58 c4 = _mm256_add_ps(_mm256_mul_ps(CK_GEMMA4_LOAD_BF16(x + i + 24),
59 CK_GEMMA4_LOAD_BF16(y + i + 24)), c4);
60 }
61#undef CK_GEMMA4_LOAD_BF16
62 c1 = _mm256_add_ps(_mm256_add_ps(c1, c3), _mm256_add_ps(c2, c4));
63 __m128 g = _mm_add_ps(_mm256_extractf128_ps(c1, 1), _mm256_castps256_ps128(c1));
64 g = _mm_add_ps(g, _mm_movehl_ps(g, g));
65 g = _mm_add_ss(g, _mm_movehdup_ps(g));
66 sum = (double)_mm_cvtss_f32(g);
67#endif
68 for (; i < n; ++i) {
69 sum += (double)(ck_bf16_to_f32(x[i]) * ck_bf16_to_f32(y[i]));
70 }
71 return (float)sum;
72}

74static inline float ck_gemma4_gelu(float x)
75{
76 if (x <= -10.0f) return 0.0f;
77 if (x >= 10.0f) return x;
78 x = CK_FP16_TO_FP32(CK_FP32_TO_FP16(x));
79 const float c0 = 0.7978845608028654f;
80 const float c1 = 0.044715f;
81 const float y = 0.5f * x * (1.0f + tanhf(c0 * x * (1.0f + c1 * x * x)));
82 return CK_FP16_TO_FP32(CK_FP32_TO_FP16(y));
83}

85static inline void ck_gemma4_unpack_q5_k_scales(const uint8_t *scales, uint8_t *sc, uint8_t *m)
86{
87 sc[0] = scales[0] & 0x3F;
88 sc[1] = scales[1] & 0x3F;
89 sc[2] = scales[2] & 0x3F;
90 sc[3] = scales[3] & 0x3F;
91
92 m[0] = scales[4] & 0x3F;
93 m[1] = scales[5] & 0x3F;
94 m[2] = scales[6] & 0x3F;
95 m[3] = scales[7] & 0x3F;
96
97 sc[4] = (scales[8] & 0x0F) | ((scales[0] >> 6) << 4);
98 sc[5] = (scales[9] & 0x0F) | ((scales[1] >> 6) << 4);
99 sc[6] = (scales[10] & 0x0F) | ((scales[2] >> 6) << 4);
100 sc[7] = (scales[11] & 0x0F) | ((scales[3] >> 6) << 4);
101
102 m[4] = (scales[8] >> 4) | ((scales[4] >> 6) << 4);
103 m[5] = (scales[9] >> 4) | ((scales[5] >> 6) << 4);
104 m[6] = (scales[10] >> 4) | ((scales[6] >> 6) << 4);
105 m[7] = (scales[11] >> 4) | ((scales[7] >> 6) << 4);
106}

108static inline uint8_t ck_gemma4_q5_k_value(const ck_gemma4_block_q5_K *block, int subblock, int i)
109{
110 const uint8_t *ql = block->qs + (subblock / 2) * 32;
111 const uint8_t low = (subblock & 1) ? (uint8_t)(ql[i] >> 4) : (uint8_t)(ql[i] & 0x0F);
112 const uint8_t high = (block->qh[i] & (uint8_t)(1u << subblock)) ? 16u : 0u;
113 return (uint8_t)(low | high);
114}

116static void ck_gemma4_dequant_q5_k_block(const ck_gemma4_block_q5_K *block, float *out)
117{
118 uint8_t sc[8];
119 uint8_t m[8];
120 ck_gemma4_unpack_q5_k_scales(block->scales, sc, m);
121 const float d = CK_FP16_TO_FP32(block->d);
122 const float dmin = CK_FP16_TO_FP32(block->dmin);
123 for (int s = 0; s < 8; ++s) {
124 const float scale = d * (float)sc[s];
125 const float minv = dmin * (float)m[s];
126 for (int i = 0; i < 32; ++i) {
127 out[s * 32 + i] = scale * (float)ck_gemma4_q5_k_value(block, s, i) - minv;
128 }
129 }

132static void ck_gemma4_rmsnorm_tmp(const float *x, const float *gamma, float *out, int n, float eps)
133{
134 double ss = 0.0;
135 for (int i = 0; i < n; ++i) {
136 ss += (double)x[i] * (double)x[i];
137 }
138 const float scale = 1.0f / sqrtf((float)(ss / (double)n) + eps);
139 for (int i = 0; i < n; ++i) {
140 out[i] = x[i] * scale * gamma[i];
141 }
142}

158static void ck_gemma4_prepare_q5_range(int begin, int end, void *opaque)
159{
160 const ck_gemma4_prepare_args_t *args =
161 (const ck_gemma4_prepare_args_t *)opaque;
162 const ck_gemma4_block_q5_K *token_blocks =
163 (const ck_gemma4_block_q5_K *)args->per_layer_token_emb;
164 const size_t token_blocks_per_row = (size_t)args->num_layers;
165 const float token_scale = sqrtf((float)args->per_layer_dim);
166 const float model_scale = 1.0f / sqrtf((float)args->embed_dim);
167 const float mix_scale = 0.7071067811865475f;
168 float token_vec[QK_K];
169 float proj_vec[QK_K];
170 float proj_normed[QK_K];
171
172 for (int t = begin; t < end; ++t) {
173 const int token = args->token_ids[t];
174 if (token < 0 || token >= args->vocab_size) continue;
175 const float *h = args->hidden + (size_t)t * (size_t)args->embed_dim;
176 uint16_t rounded_hidden[args->embed_dim];
177 for (int j = 0; j < args->embed_dim; ++j) {
178 rounded_hidden[j] = float_to_bf16(h[j]);
179 }
180 for (int layer = 0; layer < args->num_layers; ++layer) {
181 float *dst = args->per_layer_input +
182 ((size_t)t * (size_t)args->num_layers + (size_t)layer) *
183 (size_t)args->per_layer_dim;
184 const ck_gemma4_block_q5_K *tok_block = token_blocks +
185 (size_t)token * token_blocks_per_row + (size_t)layer;
186 ck_gemma4_dequant_q5_k_block(tok_block, token_vec);
187 for (int i = 0; i < args->per_layer_dim; ++i) {
188 token_vec[i] *= token_scale;
189 }
190
191 const uint16_t *model_proj_base = args->per_layer_model_proj +
192 (size_t)layer * (size_t)args->per_layer_dim *
193 (size_t)args->embed_dim;
194 for (int i = 0; i < args->per_layer_dim; ++i) {
195 const uint16_t *row = model_proj_base +
196 (size_t)i * (size_t)args->embed_dim;
197 const float acc = ck_gemma4_bf16_dot(row, rounded_hidden, args->embed_dim);
198 proj_vec[i] = acc * model_scale;
199 }
200 ck_gemma4_rmsnorm_tmp(
201 proj_vec, args->per_layer_proj_norm, proj_normed,
202 args->per_layer_dim, args->eps);
203 for (int i = 0; i < args->per_layer_dim; ++i) {
204 dst[i] = (token_vec[i] + proj_normed[i]) * mix_scale;
205 }
206 }
207 }

366static int ck_gemma4_embed_row_group(const ck_gemma4_embed_args_t *args, int token, int tokens)
367{
368 if (!args->segment_lengths) return tokens > 1 ? 2 : 1;
369 int start = 0;
370 for (int segment = 0; segment < args->num_segments; ++segment) {
371 const int length = args->segment_lengths[segment];
372 if (token < start + length) {
373 if (length == 1 || (args->singleton_tail && segment == args->num_segments - 1)) {
374 return 1;

376 return 2;

383static void ck_gemma4_embed_range(int begin, int end, void *opaque)
384{
385 const ck_gemma4_embed_args_t *args =
386 (const ck_gemma4_embed_args_t *)opaque;
387 float gate_vec[QK_K];
388 float branch[4096];
389 float branch_normed[4096];
390 for (int t = begin; t < end; ++t) {
391 float *h = args->hidden + (size_t)t * (size_t)args->embed_dim;
392 const float *inp_vec = args->per_layer_input +
393 ((size_t)t * (size_t)args->num_layers + (size_t)args->layer) *
394 (size_t)args->per_layer_dim;
395
396 const int row_group = ck_gemma4_embed_row_group(args, t, args->tokens);
397 gemm_nt_f32_llama_production_output_range(
398 h, args->inp_gate, NULL, gate_vec, row_group,
399 args->per_layer_dim, args->embed_dim, 0, args->per_layer_dim);
400 for (int i = 0; i < args->per_layer_dim; ++i) {
401 gate_vec[i] = ck_gemma4_gelu(gate_vec[i]) * inp_vec[i];
402 }
403
404 gemm_nt_f32_llama_production_output_range(
405 gate_vec, args->proj, NULL, branch, row_group,
406 args->embed_dim, args->per_layer_dim, 0, args->embed_dim);
407 rmsnorm_forward_llama_production(
408 branch, args->post_norm, branch_normed, NULL,
409 1, args->embed_dim, args->embed_dim, args->eps);
410 const float layer_scale = args->out_scale ? args->out_scale[0] : 1.0f;
411 for (int j = 0; j < args->embed_dim; ++j) {
412 h[j] = (h[j] + branch_normed[j]) * layer_scale;
413 }
414 }
