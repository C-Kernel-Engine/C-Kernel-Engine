5/**
6 * @file ck_threadpool.c
7 * @brief Persistent pthread thread pool for CK-Engine inference
8 *
9 * Architecture:
10 * - N-1 worker pthreads created at startup, main thread is thread 0
11 * - Workers spin on atomic dispatch counter waiting for work
12 * - Barriers use atomic counter + spin-wait with _mm_pause()
13 * - Hybrid polling: spin CK_THREADPOOL_SPIN_COUNT rounds, then condvar
14 * - All atomics on separate cache lines to avoid false sharing
15 *
16 * Based on the ggml_threadpool design from llama.cpp, adapted for
17 * CK-Engine's kernel dispatch model.
18 */
19

124static void barrier_init(ck_barrier_t *b, int n_threads)
125{
126 atomic_store(&b->n_arrived, 0);
127 atomic_store(&b->n_phase, 0);
128 b->n_threads = n_threads;
129}

131/**
132 * Spin-wait barrier. All threads must call this.
133 * Uses phase counter to allow re-use without reset.
134 */
135static void barrier_wait(ck_barrier_t *b)
136{
137 const int n = b->n_threads;
138 const int phase = atomic_load_explicit(&b->n_phase, memory_order_relaxed);
139
140 if (atomic_fetch_add_explicit(&b->n_arrived, 1, memory_order_acq_rel) == n - 1) {
141 /* Last thread to arrive — reset and advance phase */
142 atomic_store_explicit(&b->n_arrived, 0, memory_order_relaxed);
143 atomic_store_explicit(&b->n_phase, phase + 1, memory_order_release);
144 } else {
145 /* Spin until phase advances */
146 int spins = 0;
147 while (atomic_load_explicit(&b->n_phase, memory_order_acquire) == phase) {
148 CK_SPIN_PAUSE();
149 spins++;
150 /* After many spins, yield to avoid wasting CPU on oversubscribed systems */
151 if (spins > CK_THREADPOOL_SPIN_COUNT * 16) {
152 sched_yield();
153 spins = 0;
154 }
155 }
156 }
157}

159/* ============================================================================
160 * Worker Thread
161 * ============================================================================ */
162
163static void *worker_main(void *arg)
164{
165 ck_worker_t *w = (ck_worker_t *)arg;
166 ck_threadpool_t *pool = w->pool;
167 const int ith = w->id;
168 int last_dispatch = 0;
169
170 for (;;) {
171 /* Spin-wait for new dispatch */
172 int spins = 0;
173 int active = 0;
174 ck_work_fn_t fn = NULL;
175 void *args = NULL;
176 for (;;) {
177 /* Check shutdown */
178 if (atomic_load_explicit(&pool->stop, memory_order_acquire)) {
179 return NULL;
180 }
181
182 /* Check for new work */
183 int current = atomic_load_explicit(&pool->n_dispatch, memory_order_acquire);
184 active = atomic_load_explicit(&pool->active_threads, memory_order_acquire);
185 if (current != last_dispatch) {
186 last_dispatch = current;
187 if (ith < active) {
188 break;
189 }
190 spins = 0;
191 }
192
193 /* Threads outside the active subset sleep instead of spinning. */
194 if (ith >= active || spins >= CK_THREADPOOL_SPIN_COUNT) {
195 pthread_mutex_lock(&pool->mutex);
196 for (;;) {
197 if (atomic_load_explicit(&pool->stop, memory_order_acquire)) {
198 pthread_mutex_unlock(&pool->mutex);
199 return NULL;
200 }
201 current = atomic_load_explicit(&pool->n_dispatch, memory_order_acquire);
202 active = atomic_load_explicit(&pool->active_threads, memory_order_acquire);
203 if (current != last_dispatch) {
204 last_dispatch = current;
205 if (ith < active) {
206 pthread_mutex_unlock(&pool->mutex);
207 goto worker_have_work;
208 }
209 }
210 pthread_cond_wait(&pool->cond_dispatch, &pool->mutex);
211 }
212 }
213
214 CK_SPIN_PAUSE();
215 spins++;
216 }
217
218worker_have_work:
219 /* Execute work */
220 fn = pool->work_fn;
221 args = pool->work_args;
222 if (fn) {
223 fn(ith, active, args);
224 }
225
226 /* Signal completion */
227 if (atomic_fetch_add_explicit(&pool->n_complete, 1, memory_order_acq_rel)
228 == active - 2) {
229 /* Last worker done — wake main thread if it's waiting */
230 pthread_mutex_lock(&pool->mutex);
231 pthread_cond_signal(&pool->cond_done);
232 pthread_mutex_unlock(&pool->mutex);
233 }
234 }
235
236 return NULL;
237}

245int ck_threadpool_bounded_capacity(int default_threads, int logical_threads)
246{
247 if (default_threads < 1) default_threads = 1;
248 if (default_threads > CK_THREADPOOL_MAX_THREADS) {
249 default_threads = CK_THREADPOOL_MAX_THREADS;
250 }
251 if (logical_threads <= default_threads) return default_threads;
252
253 int capacity = default_threads + (logical_threads - default_threads) / 2;
254 if (capacity > CK_THREADPOOL_MAX_THREADS) {
255 capacity = CK_THREADPOOL_MAX_THREADS;
256 }
257 return capacity;
258}

260ck_threadpool_t *ck_threadpool_create_capacity(int default_threads,
261 int capacity_threads)
262{
263 if (default_threads <= 0) {
264 default_threads = ck_get_physical_cores();
265 if (default_threads <= 0) default_threads = 1;
266 /* Cap at reasonable default for memory-bound workloads */
267 if (default_threads > 8) default_threads = 8;
268 }
269 if (capacity_threads < default_threads) {
270 capacity_threads = default_threads;
271 }
272 if (capacity_threads > CK_THREADPOOL_MAX_THREADS) {
273 capacity_threads = CK_THREADPOOL_MAX_THREADS;
274 }
275 if (default_threads > capacity_threads) {
276 default_threads = capacity_threads;
277 }
278
279 ck_threadpool_t *pool = aligned_alloc(CK_CACHE_LINE, sizeof(ck_threadpool_t));
280 if (!pool) return NULL;
281 memset(pool, 0, sizeof(*pool));
282
283 pool->n_threads = capacity_threads;
284 pool->default_threads = default_threads;
285 atomic_store(&pool->n_dispatch, 0);
286 atomic_store(&pool->n_complete, 0);
287 atomic_store(&pool->active_threads, default_threads);
288 atomic_store(&pool->stop, 0);
289 atomic_store(&pool->paused, 0);
290 atomic_store(&pool->profile_enabled, 0);
291 pool->work_fn = NULL;
292 pool->work_args = NULL;
293
294 barrier_init(&pool->barrier, default_threads);
295
296 pthread_mutex_init(&pool->mutex, NULL);
297 pthread_cond_init(&pool->cond_dispatch, NULL);
298 pthread_cond_init(&pool->cond_done, NULL);
299
300 /* Thread 0 = main thread (no pthread created) */
301 pool->workers[0].id = 0;
302 pool->workers[0].pool = pool;
303 pool->workers[0].thread = pthread_self();
304
305 /* Spawn N-1 worker threads */
306 for (int i = 1; i < capacity_threads; i++) {
307 pool->workers[i].id = i;
308 pool->workers[i].pool = pool;
309
310 int rc = pthread_create(&pool->workers[i].thread, NULL,
311 worker_main, &pool->workers[i]);
312 if (rc != 0) {
313 fprintf(stderr, "[CK threadpool] Failed to create worker %d: %s\n",
314 i, strerror(rc));
315 /* Reduce thread count to what we managed to create */
316 pool->n_threads = i;
317 barrier_init(&pool->barrier, i);
318 break;
319 }
320 }
321
322 if (pool->n_threads > 1) {
323 fprintf(stderr,
324 "[CK threadpool] Created %d threads (default=%d, 1 main + %d workers)\n",
325 pool->n_threads, pool->default_threads, pool->n_threads - 1);
326 }
327
328 return pool;
329}

336void ck_threadpool_destroy(ck_threadpool_t *pool)
337{
338 if (!pool) return;
339
340 /* Signal shutdown */
341 atomic_store_explicit(&pool->stop, 1, memory_order_release);
342
343 /* Wake all sleeping workers */
344 pthread_mutex_lock(&pool->mutex);
345 pthread_cond_broadcast(&pool->cond_dispatch);
346 pthread_mutex_unlock(&pool->mutex);
347
348 /* Join all worker threads */
349 for (int i = 1; i < pool->n_threads; i++) {
350 pthread_join(pool->workers[i].thread, NULL);
351 }
352
353 pthread_cond_destroy(&pool->cond_dispatch);
354 pthread_cond_destroy(&pool->cond_done);
355 pthread_mutex_destroy(&pool->mutex);
356
357 free(pool);
358}

364void ck_threadpool_dispatch_n(ck_threadpool_t *pool, int active_threads, ck_work_fn_t fn, void *args)
365{
366 if (!pool || !fn) return;
367 if (active_threads <= 0) {
368 active_threads = 1;
369 }
370 if (active_threads > pool->n_threads) {
371 active_threads = pool->n_threads;
372 }
373
374 const int profile = atomic_load_explicit(
375 &pool->profile_enabled, memory_order_relaxed);
376 const uint64_t dispatch_start = profile ? monotonic_ns() : 0;
377
378 /* Single-thread fast path: just call directly */
379 if (active_threads == 1 || pool->n_threads == 1) {
380 fn(0, 1, args);
381 if (profile) {
382 const uint64_t dispatch_end = monotonic_ns();
383 atomic_fetch_add_explicit(&pool->profile_dispatch_count, 1, memory_order_relaxed);
384 atomic_fetch_add_explicit(
385 &pool->profile_dispatch_total_ns,
386 dispatch_end - dispatch_start,
387 memory_order_relaxed);
388 atomic_fetch_add_explicit(
389 &pool->profile_main_work_ns,
390 dispatch_end - dispatch_start,
391 memory_order_relaxed);
392 }
393 return;
394 }
395
396 /* Reset barrier phase for this dispatch */
397 barrier_init(&pool->barrier, active_threads);
398
399 /* Set work descriptor */
400 pool->work_fn = fn;
401 pool->work_args = args;
402 atomic_store_explicit(&pool->active_threads, active_threads, memory_order_release);
403 atomic_store_explicit(&pool->n_complete, 0, memory_order_release);
404
405 /* Wake workers by bumping dispatch counter */
406 atomic_fetch_add_explicit(&pool->n_dispatch, 1, memory_order_release);
407
408 /* Also signal condvar for sleeping workers */
409 pthread_mutex_lock(&pool->mutex);
410 pthread_cond_broadcast(&pool->cond_dispatch);
411 pthread_mutex_unlock(&pool->mutex);
412
413 /* Main thread (ith=0) does its share */
414 const uint64_t main_start = profile ? monotonic_ns() : 0;
415 fn(0, active_threads, args);
416 const uint64_t main_end = profile ? monotonic_ns() : 0;
417
418 /* Wait for all workers to complete */
419 if (active_threads > 1) {
420 int spins = 0;
421 while (atomic_load_explicit(&pool->n_complete, memory_order_acquire)
422 < active_threads - 1) {
423 CK_SPIN_PAUSE();
424 spins++;
425 if (spins >= CK_THREADPOOL_SPIN_COUNT) {
426 pthread_mutex_lock(&pool->mutex);
427 if (atomic_load_explicit(&pool->n_complete, memory_order_acquire)
428 < active_threads - 1) {
429 pthread_cond_wait(&pool->cond_done, &pool->mutex);
430 }
431 pthread_mutex_unlock(&pool->mutex);
432 spins = 0;
433 }
434 }
435 }
436 if (profile) {
437 const uint64_t dispatch_end = monotonic_ns();
438 atomic_fetch_add_explicit(&pool->profile_dispatch_count, 1, memory_order_relaxed);
439 atomic_fetch_add_explicit(
440 &pool->profile_dispatch_total_ns,
441 dispatch_end - dispatch_start,
442 memory_order_relaxed);
443 atomic_fetch_add_explicit(
444 &pool->profile_main_work_ns,
445 main_end - main_start,
446 memory_order_relaxed);
447 atomic_fetch_add_explicit(
448 &pool->profile_completion_wait_ns,
449 dispatch_end - main_end,
450 memory_order_relaxed);
451 }
452}

460typedef struct {
461 _Alignas(CK_CACHE_LINE) atomic_int next;
462 int end;
463 int grain_size;
464 ck_range_fn_t fn;
465 void *args;
466} ck_parallel_for_work_t;

468static void ck_parallel_for_worker(int ith, int nth, void *opaque)
469{
470 (void)ith;
471 (void)nth;
472 ck_parallel_for_work_t *work = (ck_parallel_for_work_t *)opaque;
473 for (;;) {
474 const int begin = atomic_fetch_add_explicit(
475 &work->next, work->grain_size, memory_order_relaxed);
476 if (begin >= work->end) break;
477 int end = begin + work->grain_size;
478 if (end > work->end) end = work->end;
479 work->fn(begin, end, work->args);
480 }
481}

483void ck_threadpool_parallel_for_n(ck_threadpool_t *pool,
484 int active_threads,
485 int begin,
486 int end,
487 int grain_size,
488 ck_range_fn_t fn,
489 void *args)
490{
491 if (!fn || begin >= end) return;
492 if (grain_size <= 0) grain_size = 1;
493 if (!pool || active_threads <= 1) {
494 fn(begin, end, args);
495 return;
496 }
497
498 ck_parallel_for_work_t work = {
499 .end = end,
500 .grain_size = grain_size,
501 .fn = fn,
502 .args = args,
503 };
504 atomic_init(&work.next, begin);
505 ck_threadpool_dispatch_n(
506 pool, active_threads, ck_parallel_for_worker, &work);
507}

613static void global_pool_init(void)
614{
615 const int available_threads = ck_available_logical_cpus();
616 int physical_threads = ck_get_physical_cores();
617 if (physical_threads > available_threads) physical_threads = available_threads;
618 int default_threads = ck_get_num_threads();
619 if (default_threads > available_threads) {
620 default_threads = available_threads;
621 }
622 int capacity_threads = default_threads;
623 const char *capacity_env = getenv("CK_THREADPOOL_CAPACITY");
624 if (capacity_env && atoi(capacity_env) > 0) {
625 capacity_threads = atoi(capacity_env);
626 } else if (!getenv("CK_NUM_THREADS") &&
627 default_threads == physical_threads) {
628 /* Generated runtimes set OMP_NUM_THREADS=1 to keep OpenMP dormant and
629 * configure the CK pool separately. Do not mistake that isolation
630 * setting for a CK capacity cap. A non-physical default remains an
631 * explicit width and does not gain automatic SMT workers. */
632 capacity_threads = ck_threadpool_bounded_capacity(
633 default_threads, available_threads);
634 }
635 if (capacity_threads > available_threads) capacity_threads = available_threads;
636 g_threadpool = ck_threadpool_create_capacity(
637 default_threads, capacity_threads);
638}
