1/**
2 * @file gemm_kernels_q4k_q8k_vnni.c
3 * @brief VNNI Q4_K x Q8_K matvec kernel (inference only)
4 *
5 * CK-ENGINE KERNEL RULES:
6 * =======================
7 * 1. NO malloc/free - memory via bump allocator, pointers passed in
8 * 2. NO OpenMP - parallelization at orchestrator/codegen layer
9 * 3. API must define: inputs, outputs, workspace, and memory layouts
10 * 4. Pure computation - deterministic, no side effects
11 *
12 * After changes: make test && make llamacpp-parity-full
13 *
14 * Requires AVX512-VNNI for vpdpbusd instruction.
15 *
16 * Packed-meta status:
17 * The packed-meta Q4_K helpers in this file are real kernel experiments, but
18 * they are not the default production layout yet. The current v8 integration
19 * is env-gated because the dispatcher still owns temporary packed-weight
20 * lifetime and shape policy. The final production design should prepack Q4_K
21 * weights at model-load/conversion time, record the extra memory in the model
22 * layout, free it with the model runtime, and choose this kernel only through
23 * hardware/shape dispatch after sweep data confirms it wins. Until then, keep
24 * the canonical GGUF-layout Q4_K path as the parity fallback.
25 */

51static inline int32_t hsum256_epi32(__m256i v) {
52 __m128i lo = _mm256_castsi256_si128(v);
53 __m128i hi = _mm256_extracti128_si256(v, 1);
54 __m128i sum = _mm_add_epi32(lo, hi);
55 sum = _mm_hadd_epi32(sum, sum);
56 sum = _mm_hadd_epi32(sum, sum);
57 return _mm_cvtsi128_si32(sum);
58}

60static inline int32_t dot_q4_k_q8_k_32_vnni(const uint8_t *q4_packed_32,
61 const int8_t *q8_32,
62 int high_nibble) {
63 const __m256i packed = _mm256_loadu_si256((const __m256i *)q4_packed_32);
64 const __m256i mask4 = _mm256_set1_epi8(0x0F);
65 const __m256i q4_bytes = high_nibble
66 ? _mm256_and_si256(_mm256_srli_epi16(packed, 4), mask4)
67 : _mm256_and_si256(packed, mask4);
68 const __m256i q8_bytes = _mm256_loadu_si256((const __m256i *)q8_32);
69 __m256i acc = _mm256_setzero_si256();
70 acc = _mm256_dpbusd_epi32(acc, q4_bytes, q8_bytes);
71 return hsum256_epi32(acc);
72}

74static inline float dot_q4_k_q8_k_vnni_block(const block_q4_K *w,
75 const block_q8_K *x) {
76 uint8_t sc[8], m_val[8];
77 unpack_q4_k_scales(w->scales, sc, m_val);
78
79 const float d = CK_FP16_TO_FP32(w->d) * x->d;
80 const float dmin = CK_FP16_TO_FP32(w->dmin) * x->d;
81 float sumf = 0.0f;
82
83 for (int j = 0, is = 0, q_offset = 0; j < QK_K; j += 64, is += 2, q_offset += 32) {
84 const uint8_t *qs = &w->qs[q_offset];
85 const int8_t *q8_lo = &x->qs[j];
86 const int8_t *q8_hi = &x->qs[j + 32];
87 if (j + 128 < QK_K) {
88 __builtin_prefetch(&w->qs[q_offset + 64], 0, 1);
89 __builtin_prefetch(&x->qs[j + 128], 0, 1);
90 }
91
92 const int32_t sum_lo = dot_q4_k_q8_k_32_vnni(qs, q8_lo, 0);
93 const int32_t sum_hi = dot_q4_k_q8_k_32_vnni(qs, q8_hi, 1);
94 const int32_t bsum_lo = (int32_t)x->bsums[j / 16] +
95 (int32_t)x->bsums[j / 16 + 1];
96 const int32_t bsum_hi = (int32_t)x->bsums[(j + 32) / 16] +
97 (int32_t)x->bsums[(j + 32) / 16 + 1];
98
99 sumf += d * (float)sc[is] * (float)sum_lo;
100 sumf -= dmin * (float)m_val[is] * (float)bsum_lo;
101 sumf += d * (float)sc[is + 1] * (float)sum_hi;
102 sumf -= dmin * (float)m_val[is + 1] * (float)bsum_hi;
103 }
104
105 return sumf;
106}

136void pack_q4_k_to_packed_u8(const void *src, void *dst, int N, int K)
137{
138 if (!src || !dst || N <= 0 || K <= 0 || (K % QK_K) != 0) {
139 return;
140 }
141 const block_q4_K *in = (const block_q4_K *)src;
142 block_q4_K_packed_u8 *out = (block_q4_K_packed_u8 *)dst;
143 const int blocks_per_row = K / QK_K;
144 for (int n = 0; n < N; ++n) {
145 for (int b = 0; b < blocks_per_row; ++b) {
146 const block_q4_K *sb = in + (size_t)n * (size_t)blocks_per_row + (size_t)b;
147 block_q4_K_packed_u8 *pb = out + (size_t)n * (size_t)blocks_per_row + (size_t)b;
148 pb->d = sb->d;
149 pb->dmin = sb->dmin;
150 unpack_q4_k_scales(sb->scales, pb->sc, pb->m);
151 for (int j = 0, q_offset = 0; j < QK_K; j += 64, q_offset += 32) {
152 const uint8_t *qs = &sb->qs[q_offset];
153 for (int l = 0; l < 32; ++l) {
154 pb->qs[j + l] = qs[l] & 0x0F;
155 pb->qs[j + 32 + l] = qs[l] >> 4;
156 }
157 }
158 }
159 }
160}

183static inline int32_t dot_q4_packed_u8_q8_32_vnni(const uint8_t *q4_32,
184 const int8_t *q8_32)
185{
186 const __m256i q4 = _mm256_loadu_si256((const __m256i *)q4_32);
187 const __m256i q8 = _mm256_loadu_si256((const __m256i *)q8_32);
188 __m256i acc = _mm256_setzero_si256();
189 acc = _mm256_dpbusd_epi32(acc, q4, q8);

206static inline float dot_q4_k_packed_u8_q8_k_block(const block_q4_K_packed_u8 *w,
207 const block_q8_K *x)
208{
209 const float d = CK_FP16_TO_FP32(w->d) * x->d;
210 const float dmin = CK_FP16_TO_FP32(w->dmin) * x->d;
211 float sumf = 0.0f;
212 for (int j = 0; j < QK_K; j += 32) {
213 const int is = j / 32;
214#if defined(__AVX512VNNI__) && defined(__AVX512VL__)
215 const int32_t sumi = dot_q4_packed_u8_q8_32_vnni(&w->qs[j], &x->qs[j]);
216#else
217 const int32_t sumi = dot_q4_packed_u8_q8_32_ref(&w->qs[j], &x->qs[j]);
218#endif
219 const int32_t bsum = (int32_t)x->bsums[j / 16] + (int32_t)x->bsums[j / 16 + 1];
220 sumf += d * (float)w->sc[is] * (float)sumi;
221 sumf -= dmin * (float)w->m[is] * (float)bsum;
222 }
223 return sumf;
224}

226static inline float dot_q4_k_packed_meta_q8_k_block(const block_q4_K_packed_meta *w,
227 const block_q8_K *x)
228{
229 const float d = CK_FP16_TO_FP32(w->d) * x->d;
230 const float dmin = CK_FP16_TO_FP32(w->dmin) * x->d;
231 float sumf = 0.0f;
232 for (int j = 0, is = 0, q_offset = 0; j < QK_K; j += 64, is += 2, q_offset += 32) {
233 const uint8_t *qs = &w->qs[q_offset];
234 const int8_t *q8_lo = &x->qs[j];
235 const int8_t *q8_hi = &x->qs[j + 32];
236
237#if defined(__AVX512VNNI__) && defined(__AVX512VL__)
238 const int32_t sum_lo = dot_q4_k_q8_k_32_vnni(qs, q8_lo, 0);
239 const int32_t sum_hi = dot_q4_k_q8_k_32_vnni(qs, q8_hi, 1);
240#else
241 int32_t sum_lo = 0;
242 int32_t sum_hi = 0;
243 for (int l = 0; l < 32; ++l) {
244 sum_lo += (int32_t)(qs[l] & 0x0F) * (int32_t)q8_lo[l];
245 sum_hi += (int32_t)(qs[l] >> 4) * (int32_t)q8_hi[l];
246 }
247#endif
248 const int32_t bsum_lo = (int32_t)x->bsums[j / 16] +
249 (int32_t)x->bsums[j / 16 + 1];
250 const int32_t bsum_hi = (int32_t)x->bsums[(j + 32) / 16] +
251 (int32_t)x->bsums[(j + 32) / 16 + 1];
252
253 sumf += d * (float)w->sc[is] * (float)sum_lo;
254 sumf -= dmin * (float)w->m[is] * (float)bsum_lo;
255 sumf += d * (float)w->sc[is + 1] * (float)sum_hi;
256 sumf -= dmin * (float)w->m[is + 1] * (float)bsum_hi;
257 }
258 return sumf;
259}

261void gemm_nt_q4_k_packed_u8_q8_k(const void *A_q8,
262 const void *B_packed,
263 const float *bias,
264 float *C,
265 int M, int N, int K)
266{
267 if (!A_q8 || !B_packed || !C || M <= 0 || N <= 0 || K <= 0 || (K % QK_K) != 0) {
268 return;
269 }
270 const block_q8_K *A = (const block_q8_K *)A_q8;
271 const block_q4_K_packed_u8 *W = (const block_q4_K_packed_u8 *)B_packed;
272 const int blocks_per_vec = K / QK_K;
273 const int blocks_per_row = K / QK_K;
274 for (int m = 0; m < M; ++m) {
275 const block_q8_K *a_row = A + (size_t)m * (size_t)blocks_per_vec;
276 float *c_row = C + (size_t)m * (size_t)N;
277 for (int n = 0; n < N; ++n) {
278 const block_q4_K_packed_u8 *w_row = W + (size_t)n * (size_t)blocks_per_row;
279 float sum = bias ? bias[n] : 0.0f;
280 for (int b = 0; b < blocks_per_row; ++b) {
281 sum += dot_q4_k_packed_u8_q8_k_block(&w_row[b], &a_row[b]);
282 }
283 c_row[n] = sum;
284 }
285 }
286}

329static void gemm_q4_packed_meta_thread_fn(int ith, int nth, void *args)
330{
331 gemm_q4_packed_meta_thread_work_t *a = (gemm_q4_packed_meta_thread_work_t *)args;
332 if (!a || ith < 0 || nth <= 0 || ith >= nth) {
333 return;
334 }
335 const int dm = (a->M + nth - 1) / nth;
336 const int m0 = dm * ith;
337 const int m1 = (m0 + dm < a->M) ? (m0 + dm) : a->M;
338 if (m0 >= a->M) {
339 return;
340 }
341 for (int m = m0; m < m1; ++m) {
342 const block_q8_K *a_row = a->A + (size_t)m * (size_t)a->blocks_per_vec;
343 float *c_row = a->C + (size_t)m * (size_t)a->N;

345 const block_q4_K_packed_meta *w_row = a->W + (size_t)n * (size_t)a->blocks_per_row;
346 float sum = a->bias ? a->bias[n] : 0.0f;
347 for (int b = 0; b < a->blocks_per_row; ++b) {
348 sum += dot_q4_k_packed_meta_q8_k_block(&w_row[b], &a_row[b]);

455void gemv_q4_k_q8_k_vnni(float *y,
456 const void *W,
457 const void *x_q8,
458 int M, int K)
459{
460#if defined(__AVX512VNNI__) && defined(__AVX512VL__)
461 const char *fast_env = getenv("CK_ENABLE_Q4K_Q8K_VNNI_FAST");
462 const int fast_disabled = fast_env && fast_env[0] && fast_env[0] == '0';
463 if (!fast_disabled && !ck_strict_parity_enabled()) {
464 if (!y || !W || !x_q8 || M <= 0 || K <= 0) {
465 return;
466 }
467
468 const block_q4_K *blocks = (const block_q4_K *)W;
469 const block_q8_K *x = (const block_q8_K *)x_q8;
470 const int blocks_per_row = K / QK_K;
471
472 for (int row = 0; row < M; ++row) {
473 const block_q4_K *w_row = blocks + (size_t)row * (size_t)blocks_per_row;
474 float sum = 0.0f;
475 for (int b = 0; b < blocks_per_row; ++b) {
476 sum += dot_q4_k_q8_k_vnni_block(&w_row[b], &x[b]);
477 }
478 y[row] = sum;
479 }
480 return;
481 }
482#endif
483
484 /* Strict/debug parity keeps the llama-style scalar accumulation path.
485 * Production AVX-512 hosts use VNNI by default; set
486 * CK_ENABLE_Q4K_Q8K_VNNI_FAST=0 or CK_DEBUG_Q4K_Q8_REF=1 when attributing
487 * borderline logit movement against scalar/reference behavior.
488 */
489 gemv_q4_k_q8_k_ref(y, W, x_q8, M, K);
490}

493void gemv_q4_k_q8_k_parallel_vnni(float *y,
494 const void *W,
495 const void *x_q8,
496 int M, int K,
497 int ith, int nth)
498{
499#if defined(__AVX512VNNI__) && defined(__AVX512VL__)
500 if (!y || !W || !x_q8 || M <= 0 || K <= 0) {
501 return;
502 }
503 if (ith < 0 || nth <= 0 || ith >= nth) {
504 return;
505 }
506
507 const int dr = (M + nth - 1) / nth;
508 const int r0 = dr * ith;
509 const int r1 = (r0 + dr < M) ? (r0 + dr) : M;
510 if (r0 >= M) {
511 return;
512 }
513
514 const block_q4_K *blocks = (const block_q4_K *)W;
515 const block_q8_K *x = (const block_q8_K *)x_q8;
516 const int blocks_per_row = K / QK_K;
