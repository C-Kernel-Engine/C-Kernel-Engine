3/* Caller-supplied inverse frequencies make their construction/storage an
4 * explicit model artifact. Phase is two FP32 multiplies, without FMA. */
5static int tables(const float *positions, size_t position_elements,
6 const float *frequencies, size_t frequency_elements, float scale,

9{
10 size_t n;
11 if (!rows || !columns || position_elements < rows || frequency_elements < columns ||
12 !ck_checked_span(rows, columns, stride, &n) || sine_elements < n || cosine_elements < n)
13 return -2;
14 if (!isfinite(scale) || !ck_checked_region(positions, rows) ||
15 !ck_checked_region(frequencies, columns) || !ck_checked_region(sine, n) ||
16 !ck_checked_region(cosine, n) ||
17 ck_checked_overlap(positions, rows, sine, n) || ck_checked_overlap(positions, rows, cosine, n) ||
18 ck_checked_overlap(frequencies, columns, sine, n) || ck_checked_overlap(frequencies, columns, cosine, n) ||
19 (!combined && ck_checked_overlap(sine, n, cosine, n))) return -1;
20 if (!ck_checked_finite(positions, rows) || !ck_checked_finite(frequencies, columns)) return -3;
21 for (size_t row = 0; row < rows; ++row) {
22 float position = positions[row] * scale;
23 if (!isfinite(position)) return -3;
24 for (size_t col = 0; col < columns; ++col)
25 if (!isfinite(position * frequencies[col])) return -3;
26 }
27 for (size_t row = 0; row < rows; ++row) {
28 float position = positions[row] * scale;
29 for (size_t col = 0; col < columns; ++col) {
30 float phase = position * frequencies[col];
31 sine[row * stride + col] = sinf(phase);
32 cosine[row * stride + col] = cosf(phase);
33 }
34 }
35 return 0;
36}
