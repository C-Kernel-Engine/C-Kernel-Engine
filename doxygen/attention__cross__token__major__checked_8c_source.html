31static int attention_scores(const float *q, const float *key, size_t memory_tokens,
32 size_t width, size_t head, size_t dim, float *scores, double *denominator) {
33 double scale = 1.0 / sqrt((double)dim), maximum = -INFINITY;
34 for (size_t k = 0; k < memory_tokens; ++k) {
35 double value = dot(q, key + k * width + head * dim, dim) * scale;
36 if (!isfinite(value) || fabs(value) > FLT_MAX) return -1;
37 scores[k] = (float)value;
38 if (scores[k] > maximum) maximum = scores[k];
39 }
40 *denominator = 0.0;
41 for (size_t k = 0; k < memory_tokens; ++k) {
42 scores[k] = (float)exp((double)scores[k] - maximum);
43 *denominator += (double)scores[k];
44 }
45 return !isfinite(*denominator) || *denominator <= 0.0 ? -1 : 0;
46}
