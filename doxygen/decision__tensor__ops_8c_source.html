7static int region(const void *p, size_t n) {

11static int overlap(const void *a, size_t n, const void *b, size_t m) {
12 return (uintptr_t)a < (uintptr_t)b + m * 4 &&
13 (uintptr_t)b < (uintptr_t)a + n * 4;
14}
15static int product(size_t a, size_t b, size_t *n) {
16 if (!a || !b || a > SIZE_MAX / b) return 0;

20static int buffers(const void **inputs, const size_t *sizes, size_t count,
21 float *out, size_t on, float *scratch, size_t sn, size_t needed) {
22 if (on < needed || sn < needed || !region(out, on) || !region(scratch, sn)) return 0;
23 if (overlap(out, on, scratch, sn)) return 0;
24 for (size_t i=0; i<count; ++i) {
25 if (!region(inputs[i], sizes[i]) || overlap(out,on,inputs[i],sizes[i]) ||
26 overlap(scratch,sn,inputs[i],sizes[i])) return 0;
27 }
28 return 1;
29}

38int ck_broadcast_add_rows_f32(const float *left, size_t left_elements,
39 const float *right, size_t right_elements, float *output, size_t output_elements,
40 float *scratch, size_t scratch_elements, size_t rows, size_t channels) {
41 size_t n;
42 if (!product(rows,channels,&n) || left_elements<n || right_elements<channels) return -2;
43 const void *in[]={left,right}; size_t sizes[]={left_elements,right_elements};
44 if (!buffers(in,sizes,2,output,output_elements,scratch,scratch_elements,n) ||
45 !all_finite(left,n) || !all_finite(right,channels)) return -1;
46 for (size_t i=0; i<n; ++i) scratch[i]=left[i]+right[i%channels];
47 return publish(output,scratch,n);

49static int pair_check(const float *left,size_t ln,const float *right,size_t rn,
50 const int32_t *indices,size_t in,float *out,size_t on,float *scratch,size_t sn,
51 size_t groups,size_t rows,size_t channels,size_t need) {
52 size_t l,r;
53 if (!product(groups,channels,&l) || !product(rows,channels,&r) ||
54 ln<l || rn<r || in<rows) return 0;
55 const void *inputs[]={left,right,indices}; size_t sizes[]={ln,rn,in};
56 if (!buffers(inputs,sizes,3,out,on,scratch,sn,need) || !all_finite(left,l) || !all_finite(right,r)) return 0;
57 for (size_t i=0; i<rows; ++i) if (indices[i]<0 || (size_t)indices[i]>=groups) return 0;
58 return 1;
59}
60int ck_indexed_pair_features_f32(const float *left,size_t left_elements,
61 const float *right,size_t right_elements,const int32_t *indices,size_t indices_elements,
62 float *output,size_t output_elements,float *scratch,size_t scratch_elements,
63 size_t groups,size_t rows,size_t channels) {
64 size_t n;
65 if (!product(rows,channels,&n) || n>SIZE_MAX/4) return -2;
66 if (!pair_check(left,left_elements,right,right_elements,indices,indices_elements,
67 output,output_elements,scratch,scratch_elements,groups,rows,channels,n*4)) return -1;
68 for (size_t r=0;r<rows;++r) for (size_t c=0;c<channels;++c) {
69 float a=left[(size_t)indices[r]*channels+c], b=right[r*channels+c];
70 size_t k=r*channels*4+c;
71 scratch[k]=a; scratch[k+channels]=b;
72 scratch[k+2*channels]=a*b; scratch[k+3*channels]=fabsf(a-b);
73 }
74 return publish(output,scratch,n*4);
75}
76int ck_indexed_cosine_rows_f32(const float *left,size_t left_elements,
77 const float *right,size_t right_elements,const int32_t *indices,size_t indices_elements,
78 float *output,size_t output_elements,float *scratch,size_t scratch_elements,
79 size_t groups,size_t rows,size_t channels,float epsilon) {
80 if (!isfinite(epsilon) || epsilon<=0 || !pair_check(left,left_elements,right,right_elements,
81 indices,indices_elements,output,output_elements,scratch,scratch_elements,
82 groups,rows,channels,rows)) return -1;
83 for (size_t r=0;r<rows;++r) {
84 const float *a=left+(size_t)indices[r]*channels, *b=right+r*channels;
85 double aa=0,bb=0;
86 for(size_t c=0;c<channels;++c) {
87 volatile double x=(double)a[c]*a[c], y=(double)b[c]*b[c];
88 aa+=x; bb+=y;
89 }
90 double na=sqrt(aa),nb=sqrt(bb);
91 if (!isfinite(na)||!isfinite(nb)||na>FLT_MAX||nb>FLT_MAX) return -1;
92 float fa=fmaxf((float)na,epsilon), fb=fmaxf((float)nb,epsilon);
93 double sum=0;
94 for(size_t c=0;c<channels;++c) {
95 float x=a[c]/fa,y=b[c]/fb;
96 volatile float term=x*y;
97 sum+=(double)term;
98 }
99 scratch[r]=(float)sum;
100 }
101 return publish(output,scratch,rows);
102}

106int ck_scalar_exp_clamped_f32(const float *input,size_t input_elements,
107 float *output,size_t output_elements,float maximum) {
108 if (!scalar_check(input,input_elements,output,output_elements)||!isfinite(maximum)) return -1;
109 float value=expf(fminf(input[0],maximum));

113int ck_scalar_sigmoid_f32(const float *input,size_t input_elements,
114 float *output,size_t output_elements) {
115 if (!scalar_check(input,input_elements,output,output_elements)) return -1;
116 float x=input[0],e=expf(-fabsf(x));
117 float value=x>=0 ? 1.0f/(1.0f+e) : e/(1.0f+e);
118 output[0]=value; return 0;
119}
