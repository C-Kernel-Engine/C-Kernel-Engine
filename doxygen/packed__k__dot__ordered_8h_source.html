9/* Decode one packed block at a time. FP32 operands are accumulated in channel
10 * order in FP64, without requiring a dense weight buffer. */
11#define CK_DEFINE_PACKED_ORDERED_DOT(name, type, block_width, decode) \
12static inline float name(const float *input, const unsigned char *weight, size_t width) \
13{ \
14 double sum = 0.0; \
15 for (size_t first = 0; first < width; first += block_width, weight += sizeof(type)) { \
16 type block; \
17 float values[block_width]; \
18 memcpy(&block, weight, sizeof(block)); \
19 decode(&block, values, block_width); \
20 for (size_t channel = 0; channel < block_width; ++channel) \
21 sum += (double)input[first + channel] * (double)values[channel]; \
22 } \
23 return (float)sum; \
24}

41static inline float ck_packed_q4_k_dot_ordered_avx2(
42 const float *input, const unsigned char *weight, size_t width)
43{
44#if defined(__clang__)
45#pragma clang fp contract(off)
46#endif
47 double sum = 0.0;
48 const __m256i mask = _mm256_set1_epi32(15);
49 for (size_t first = 0; first < width; first += QK_K, weight += sizeof(block_q4_K)) {
50 block_q4_K block;
51 memcpy(&block, weight, sizeof(block));
52 const float scale = _cvtsh_ss(block.d), minimum = _cvtsh_ss(block.dmin);
53 uint8_t scales[8], mins[8];
54 unpack_q4_k_scales(block.scales, scales, mins);
55 for (int group = 0; group < 8; ++group) {
56 const __m256 d = _mm256_set1_ps(scale * (float)scales[group]);
57 const __m256 m = _mm256_set1_ps(minimum * (float)mins[group]);
58 const uint8_t *qs = block.qs + (group / 2) * 32;
59 const float *row = input + first + group * 32;
60 for (int j = 0; j < 32; j += 8) {
61 const __m128i bytes = _mm_loadl_epi64((const __m128i *)(qs + j));
62 __m256i codes = _mm256_cvtepu8_epi32(bytes);
63 if (group & 1) codes = _mm256_srli_epi32(codes, 4);
64 codes = _mm256_and_si256(codes, mask);
65 const __m256 expanded = _mm256_sub_ps(_mm256_mul_ps(d, _mm256_cvtepi32_ps(codes)), m);
66 const __m256 x = _mm256_loadu_ps(row + j);
67 double products[8];
68 _mm256_storeu_pd(products, _mm256_mul_pd(
69 _mm256_cvtps_pd(_mm256_castps256_ps128(x)),
70 _mm256_cvtps_pd(_mm256_castps256_ps128(expanded))));
71 _mm256_storeu_pd(products + 4, _mm256_mul_pd(
72 _mm256_cvtps_pd(_mm256_extractf128_ps(x, 1)),
73 _mm256_cvtps_pd(_mm256_extractf128_ps(expanded, 1))));
74 for (int k = 0; k < 8; ++k) sum += products[k];
75 }
76 }
77 }
78 return (float)sum;
79}
