41void split_q_gate_forward(const float *packed_qg,
42 float *q,
43 float *gate,
44 int rows,
45 int q_dim,
46 int gate_dim,
47 int group_dim) {
48 const int packed_dim = q_dim + gate_dim;
49 if (!packed_qg || !q || !gate || rows <= 0 || q_dim <= 0 || gate_dim <= 0) {
50 return;
51 }
52 if (group_dim <= 0) {
53 group_dim = q_dim;
54 }
55 const int q_groups = q_dim / group_dim;
56 const int gate_group_dim = (q_groups > 0 && gate_dim % q_groups == 0) ? (gate_dim / q_groups) : gate_dim;
57 for (int row = 0; row < rows; ++row) {
58 const float *src = packed_qg + (size_t) row * (size_t) packed_dim;
59 float *q_dst = q + (size_t) row * (size_t) q_dim;
60 float *gate_dst = gate + (size_t) row * (size_t) gate_dim;
61 if (q_groups > 0 && q_groups * group_dim == q_dim && q_groups * gate_group_dim == gate_dim) {
62 for (int group = 0; group < q_groups; ++group) {
63 const size_t src_group_off = (size_t) group * (size_t) (group_dim + gate_group_dim);
64 memcpy(
65 q_dst + (size_t) group * (size_t) group_dim,
66 src + src_group_off,
67 (size_t) group_dim * sizeof(float));
68 memcpy(
69 gate_dst + (size_t) group * (size_t) gate_group_dim,
70 src + src_group_off + (size_t) group_dim,
71 (size_t) gate_group_dim * sizeof(float));
72 }
73 } else {
74 memcpy(q_dst, src, (size_t) q_dim * sizeof(float));
75 memcpy(gate_dst, src + q_dim, (size_t) gate_dim * sizeof(float));
76 }
77 }
78}

145static int hybrid_ranges_overlap(const float *lhs,
146 const float *rhs,
147 size_t elements) {
148 const uintptr_t lhs_begin = (uintptr_t)lhs;
149 const uintptr_t rhs_begin = (uintptr_t)rhs;
150 const size_t bytes = elements * sizeof(float);
151 return lhs_begin < rhs_begin + bytes && rhs_begin < lhs_begin + bytes;

154void attn_gate_sigmoid_mul_forward(const float *x,
155 const float *gate,
156 float *out,
157 int rows,
158 int num_heads,
159 int state_dim) {
160 if (!x || !gate || !out || rows <= 0 || num_heads <= 0 ||
161 state_dim <= 0 || num_heads > INT_MAX / state_dim) {
162 return;
163 }
164 const int dim = num_heads * state_dim;
165 const size_t elements = (size_t)rows * (size_t)dim;
166 ck_attn_gate_sigmoid_mul_args_t args = {x, gate, out, dim};
167 const size_t minimum_elements_per_thread = 32768u;
168
169 const int partial_x_overlap = out != x &&
170 hybrid_ranges_overlap(out, x, elements);
171 const int gate_overlap = hybrid_ranges_overlap(out, gate, elements);
172 if (partial_x_overlap || gate_overlap ||
173 elements < 2u * minimum_elements_per_thread) {
174 attn_gate_sigmoid_mul_row_range(0, rows, &args);
175 return;
176 }
177
178 ck_threadpool_t *pool = ck_threadpool_global();
179 int active = pool ? ck_threadpool_n_threads(pool) : 1;
180 size_t useful_threads =
181 (elements + minimum_elements_per_thread - 1u) /
182 minimum_elements_per_thread;
183 if ((size_t)active > useful_threads) {
184 active = (int)useful_threads;
185 }
186 if (active > rows) {
187 active = rows;
188 }
189
190 if (active > 1 && ck_threadpool_thread_id(pool) <= 0) {
191 ck_threadpool_parallel_for_n(
192 pool, active, 0, rows, 4,
193 attn_gate_sigmoid_mul_row_range, &args);
194 } else {
195 attn_gate_sigmoid_mul_row_range(0, rows, &args);
196 }
197}

235 for (int row = 0; row < rows; ++row) {
236 const float *x_row = x + (size_t) row * (size_t) dim;
237 const float *gate_row = gate + (size_t) row * (size_t) num_heads;
238 float *out_row = out + (size_t) row * (size_t) dim;
239 for (int head = 0; head < num_heads; ++head) {
240 const float value = gate_row[head];
241 const float scale = value > 20.0f

244 const size_t base = (size_t) head * (size_t) state_dim;
245 for (int col = 0; col < state_dim; ++col) {
246 out_row[base + (size_t) col] =
247 x_row[base + (size_t) col] * scale;
248 }

270 d_x_row[col] = d_out_row[col] * sig;
