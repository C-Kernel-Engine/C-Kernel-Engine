21static int binary_bf16(const uint16_t *left, const uint16_t *right, size_t rhs_count,
22 uint16_t *out, uint16_t *stage, size_t n, int multiply)
23{
24 if (!n || !rhs_count || n % rhs_count ||

26 !valid_span(out, n, 2) || !valid_span(stage, n, 2))
27 return -2;
28 const size_t bytes = n * 2;
29 if (overlaps(stage, bytes, left, bytes) || overlaps(stage, bytes, right, rhs_count * 2) ||
30 overlaps(stage, bytes, out, bytes) || overlaps(out, bytes, left, bytes) ||
31 overlaps(out, bytes, right, rhs_count * 2))
32 return -2;
33
34 size_t column = 0;
35 for (size_t i = 0; i < n; ++i) {
36 const float a = bf16_to_float(left[i]), b = bf16_to_float(right[column]);
37 if (++column == rhs_count)
38 column = 0;
39 const uint16_t value = float_to_bf16(multiply ? a * b : a + b);
40 if (!isfinite(a) || !isfinite(b) || !isfinite(bf16_to_float(value)))
41 return -3;
42 stage[i] = value;
43 }
