9static int ck_mul_size(size_t a, size_t b, size_t *out) {
10 if (a && b > SIZE_MAX / a) return 0;
11 *out = a * b;

15static int ck_overlap(const void *a, size_t na, const void *b, size_t nb) {
16 uintptr_t x = (uintptr_t)a, y = (uintptr_t)b;
17 if (na > UINTPTR_MAX - x || nb > UINTPTR_MAX - y) return 1;
18 return x < y + nb && y < x + na;
19}
