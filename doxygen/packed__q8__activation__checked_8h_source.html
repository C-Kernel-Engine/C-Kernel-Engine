7/* Input rows are finite. Emit little-endian Q8_0 blocks with nearest-even codes. */
8static inline int ck_pack_q8_activation_rne(const float *input, unsigned char *packed, size_t width)
9{
10 for (size_t i = 0; i < width; i += QK8_0, packed += sizeof(block_q8_0)) {
11 float peak = 0.0f;
12 for (size_t j = 0; j < QK8_0; ++j)
13 peak = fmaxf(peak, fabsf(input[i + j]));
14 const uint16_t scale = ck_fp32_to_fp16_soft(peak / 127.0f);
15 if ((scale & UINT16_C(0x7c00)) == UINT16_C(0x7c00)) return -1;
16 packed[0] = (unsigned char)scale;
17 packed[1] = (unsigned char)(scale >> 8);
18 const float inverse = scale ? 127.0f / peak : 0.0f;
19 for (size_t j = 0; j < QK8_0; ++j) {
20 const float value = input[i + j] * inverse;
21 const float magnitude = fminf(127.0f, fabsf(value));
22 int rounded = (int)magnitude;
23 const float fraction = magnitude - (float)rounded;
24 if (fraction > 0.5f || (fraction == 0.5f && (rounded & 1))) ++rounded;
25 packed[2 + j] = (unsigned char)(value < 0.0f ? -rounded : rounded);
26 }
27 }
28 return 0;
29}

31static inline float ck_q8_activation_dot(const unsigned char *weight, const unsigned char *input,
32 size_t width)
33{
34 /* The x86 GGML reduction groups four signed products into each of
35 * eight FP32 FMA lanes, then combines lanes in a fixed tree. */
36 float lanes[8] = {0};
37 for (size_t i = 0; i < width; i += QK8_0, weight += sizeof(block_q8_0), input += sizeof(block_q8_0)) {
38 const float dw = ck_fp16_to_fp32_soft((uint16_t)(weight[0] | ((uint16_t)weight[1] << 8)));
39 const float dx = ck_fp16_to_fp32_soft((uint16_t)(input[0] | ((uint16_t)input[1] << 8)));
40 const float scale = dw * dx;
41 for (size_t lane = 0; lane < 8; ++lane) {
42 int dot = 0;
43 for (size_t j = 2 + 4 * lane; j < 6 + 4 * lane; ++j) {
44 const int w = weight[j], x = input[j];
45 dot += (w < 128 ? w : w - 256) * (x < 128 ? x : x - 256);
46 }
47 lanes[lane] = fmaf(scale, (float)dot, lanes[lane]);
48 }
49 }
50 return ((lanes[0] + lanes[4]) + (lanes[2] + lanes[6]))
51 + ((lanes[1] + lanes[5]) + (lanes[3] + lanes[7]));
52}
