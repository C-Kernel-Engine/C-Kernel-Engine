7/* Counter-based excitation. Each element depends only on its index, seed and
8 * stream, so requests cannot inherit RNG state from previous calls. The
9 * Box-Muller transform uses FP64 libm and stores FP32 samples. It does not
10 * promise the same random stream as PyTorch. */
11static uint64_t mix64(uint64_t value) {
12 value += UINT64_C(0x9e3779b97f4a7c15);
13 value = (value ^ (value >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
14 value = (value ^ (value >> 27)) * UINT64_C(0x94d049bb133111eb);
15 return value ^ (value >> 31);
16}
