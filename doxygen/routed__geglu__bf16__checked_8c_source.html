47int ck_routed_geglu_bf16_checked(
48 const uint16_t *input, const int32_t *ids, const float *scores,
49 const uint16_t *gate_up, const uint16_t *down, uint16_t *out, uint16_t *routes,
50 void *workspace, size_t workspace_bytes, size_t rows, size_t width,
51 size_t inner, size_t experts, size_t top_k, int use_blas,
52 size_t hidden_elements, size_t selected_elements, size_t route_elements,
53 size_t gate_elements, size_t down_elements)
54{
55 // Each sized domain fits a signed 32-bit BLAS extent. Cap products before
56 // multiplying so all subsequent byte calculations also fit size_t.
57 if (!rows || !width || !inner || !experts || !top_k || top_k > experts ||
58 rows > INT_MAX || width > INT_MAX || inner > INT_MAX / 2 || experts > INT32_MAX ||
59 width > SIZE_MAX / 64 || inner > SIZE_MAX / 64 || rows > SIZE_MAX / (64 * (width + inner)) ||
60 experts > SIZE_MAX / (64 * width) / inner || top_k > SIZE_MAX / (64 * rows) / width)
61 return -2;
62 const size_t rh = rows * width, ri = rows * inner;
63 const size_t rk = rows * top_k, routed = rk * width;
64 const size_t hi = width * inner, blas_elements = rh + 2 * hi + 2 * ri;
65 const size_t work_bytes = 8 * blas_elements + 4 * inner + 2 * (4 * rh + 5 * ri + routed);
66 if (workspace_bytes < work_bytes || (use_blas != 0 && use_blas != 1) ||
67 hidden_elements != rh || selected_elements != rk || route_elements != routed ||
68 gate_elements != experts * 2 * hi || down_elements != experts * hi)
69 return -2;
70 const void *inputs[] = {input, ids, scores, gate_up, down};
71 const size_t sizes[] = {rh * 2, rk * 4, rk * 4, experts * hi * 4, experts * hi * 2};
72 const size_t aligns[] = {2, 4, 4, 2, 2};
73 if (!span(out, rh * 2, 2) || !span(routes, routed * 2, 2) ||
74 !span(workspace, work_bytes, _Alignof(double)) ||
75 overlaps(out, rh * 2, routes, routed * 2) ||
76 overlaps(out, rh * 2, workspace, work_bytes) ||
77 overlaps(routes, routed * 2, workspace, work_bytes))
78 return -2;
79 for (size_t i = 0; i < sizeof(inputs) / sizeof(inputs[0]); ++i)
80 if (!span(inputs[i], sizes[i], aligns[i]) ||
81 overlaps(inputs[i], sizes[i], out, rh * 2) ||
82 overlaps(inputs[i], sizes[i], routes, routed * 2) ||
83 overlaps(inputs[i], sizes[i], workspace, work_bytes))
84 return -2;
85
86 for (size_t row = 0; row < rows; ++row) {
87 for (size_t k = 0; k < top_k; ++k) {
88 const size_t index = row * top_k + k;
89 if (ids[index] < 0 || (size_t)ids[index] >= experts || !isfinite(scores[index]))
90 return -3;
91 for (size_t previous = 0; previous < k; ++previous)
92 if (ids[index] == ids[row * top_k + previous])
93 return -3;
94 }
95 }
96
97 double *blas = workspace;
98 float *gelu = (float *)(blas + blas_elements);
99 uint16_t *gathered = (uint16_t *)(gelu + inner);
100 uint16_t *gate = gathered + rh;
101 uint16_t *active = gate + 2 * ri;
102 uint16_t *projected = active + ri;
103 uint16_t *stage = projected + rh;
104 uint16_t *accum = stage + rh + 2 * ri;
105 uint16_t *route_stage = accum + rh;
106 memset(accum, 0, rh * sizeof(*accum));
107 const projection_workspace projection = {blas, blas_elements, stage, rh + 2 * ri, use_blas};
108
109 // Match expert-major index_add order. The route slot is used only to
110 // associate the input row and scaled probability with this expert.
111 for (size_t expert = 0; expert < experts; ++expert) {
112 size_t count = 0;
113 for (size_t index = 0; index < rk; ++index)
114 if ((size_t)ids[index] == expert) {
115 memcpy(gathered + count * width, input + (index / top_k) * width, width * 2);
116 ++count;
117 }
118 if (!count)
119 continue;
120 int status = project(gathered, gate_up + expert * 2 * hi, gate, count, width, 2 * inner, projection);
121 if (status)
122 return status;
123 for (size_t row = 0; row < count; ++row) {
124 status = ck_geglu_tanh_bf16_checked(gate + row * 2 * inner,
125 gate + row * 2 * inner + inner, active + row * inner, gelu, inner);
126 if (status)
127 return status;
128 }
129 status = project(active, down + expert * hi, projected, count, inner, width, projection);
130 if (status)
131 return status;
132 size_t row = 0;
133 for (size_t index = 0; index < rk; ++index) {
134 if ((size_t)ids[index] != expert)
135 continue;
136 uint16_t *destination = accum + (index / top_k) * width;
137 const uint16_t *unweighted = projected + row * width;
138 memcpy(route_stage + index * width, unweighted, width * 2);
139 for (size_t col = 0; col < width; ++col) {
140 const uint16_t scaled = float_to_bf16(bf16_to_float(unweighted[col]) * scores[index]);
141 const uint16_t sum = float_to_bf16(bf16_to_float(destination[col]) + bf16_to_float(scaled));
142 if (!isfinite(bf16_to_float(sum)))
143 return -3;
144 destination[col] = sum;
145 }
146 ++row;
147 }
148 }
149 memcpy(out, accum, rh * 2);
150 memcpy(routes, route_stage, routed * 2);
151 return 0;
152}
