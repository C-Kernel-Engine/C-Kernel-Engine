32static inline int32_t hsum256_epi32(__m256i v) {
33 __m128i lo = _mm256_castsi256_si128(v);
34 __m128i hi = _mm256_extracti128_si256(v, 1);
35 __m128i sum = _mm_add_epi32(lo, hi);
36 sum = _mm_hadd_epi32(sum, sum);
37 sum = _mm_hadd_epi32(sum, sum);
38 return _mm_cvtsi128_si32(sum);
39}

41static inline int32_t dot_q4_q8_32_avx2(const uint8_t *q4,
42 const int8_t *q8,
43 int high_nibble) {
44 const __m256i packed = _mm256_loadu_si256((const __m256i *)q4);
45 const __m256i mask4 = _mm256_set1_epi8(0x0F);
46 const __m256i q4_bytes = high_nibble
47 ? _mm256_and_si256(_mm256_srli_epi16(packed, 4), mask4)
48 : _mm256_and_si256(packed, mask4);
49 const __m256i q8_bytes = _mm256_loadu_si256((const __m256i *)q8);
50 const __m256i pair_sums_i16 = _mm256_maddubs_epi16(q4_bytes, q8_bytes);
51 const __m256i sums_i32 = _mm256_madd_epi16(pair_sums_i16, _mm256_set1_epi16(1));
52 return hsum256_epi32(sums_i32);
53}

55static float dot_q4_k_q8_k_avx2(const block_q4_K *w,
56 const block_q8_K *x,
57 int k)
58{
59 const int nb = k / QK_K;
60 float sumf = 0.0f;
61
62 for (int i = 0; i < nb; ++i) {
63 uint8_t sc[8], m_val[8];
64 unpack_q4_k_scales(w[i].scales, sc, m_val);
65
66 const float d = CK_FP16_TO_FP32(w[i].d) * x[i].d;
67 const float dmin = CK_FP16_TO_FP32(w[i].dmin) * x[i].d;
68
69 int sumi = 0;
70 for (int j = 0; j < QK_K / 16; ++j) {
71 sumi += (int)x[i].bsums[j] * (int)m_val[j / 2];
72 }
73
74 int32_t sumi_block = 0;
75 int is = 0;
76 int q_offset = 0;
77
78 for (int j = 0; j < QK_K; j += 64) {
79 const uint8_t *qs = &w[i].qs[q_offset];
80 const int8_t *q8_lo = &x[i].qs[j];
81 const int8_t *q8_hi = &x[i].qs[j + 32];
82
83 const int32_t lo = dot_q4_q8_32_avx2(qs, q8_lo, 0);
84 const int32_t hi = dot_q4_q8_32_avx2(qs, q8_hi, 1);
85 sumi_block += (int32_t)sc[is] * lo + (int32_t)sc[is + 1] * hi;
86
87 q_offset += 32;
88 is += 2;
89 }
90
91 sumf += d * (float)sumi_block - dmin * (float)sumi;
92 }
93
94 return sumf;
95}

98void gemv_q4_k_q8_k_avx2(float *y,
99 const void *W,
100 const void *x_q8,
101 int M, int K)
102{
103#if defined(__AVX2__)
104 if (!y || !W || !x_q8 || M <= 0 || K <= 0) {
105 return;
106 }
107
108 const block_q4_K *blocks = (const block_q4_K *)W;
109 const block_q8_K *x = (const block_q8_K *)x_q8;
110 const int blocks_per_row = K / QK_K;
111
112 for (int row = 0; row < M; ++row) {
113 const block_q4_K *w_row = blocks + (size_t)row * (size_t)blocks_per_row;
114 y[row] = dot_q4_k_q8_k_avx2(w_row, x, K);
115 }
116#else
117 gemv_q4_k_q8_k_ref(y, W, x_q8, M, K);
118#endif
119}
