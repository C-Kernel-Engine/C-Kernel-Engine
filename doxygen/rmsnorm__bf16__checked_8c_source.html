25static int rmsnorm_rows_bf16_checked(
26 const uint16_t *input, size_t input_elements, const uint16_t *gamma,
27 size_t gamma_elements, uint16_t *output, size_t output_elements,
28 float *work, size_t work_elements, size_t rows, size_t width, float epsilon,
29 rmsnorm_forward_fn normalize)
30{
31 if (!rows || !width || rows > INT_MAX || width > INT_MAX ||
32 rows > (SIZE_MAX - width) / width || !isfinite(epsilon) || epsilon <= 0.0f) {
33 return -2;
34 }
35 const size_t count = rows * width;
36 const size_t required_work = count + width;
37 if (input_elements < count || gamma_elements < width ||
38 output_elements < count || work_elements < required_work) {
39 return -2;
40 }
41 if (!valid_span(input, count, sizeof(*input)) ||
42 !valid_span(gamma, width, sizeof(*gamma)) ||
43 !valid_span(output, count, sizeof(*output)) ||
44 !valid_span(work, required_work, sizeof(*work))) {
45 return -1;
46 }
47 const size_t work_bytes = required_work * sizeof(*work);
48 const size_t row_bytes = count * sizeof(*output);
49 const size_t gamma_bytes = width * sizeof(*gamma);
50 if (overlap(work, work_bytes, input, row_bytes) ||
51 overlap(work, work_bytes, gamma, gamma_bytes) ||
52 overlap(work, work_bytes, output, row_bytes) ||
53 overlap(output, row_bytes, input, row_bytes) ||
54 overlap(output, row_bytes, gamma, gamma_bytes)) {
55 return -1;
56 }
57 float *expanded_gamma = work + count;
58 for (size_t i = 0; i < width; ++i) {
59 expanded_gamma[i] = bf16_to_float(gamma[i]);
60 if (!isfinite(expanded_gamma[i])) {
61 return -3;
62 }
63 }
64 for (size_t i = 0; i < count; ++i) {
65 work[i] = bf16_to_float(input[i]);
66 if (!isfinite(work[i])) {
67 return -3;
68 }
69 }
70 normalize(
71 work, expanded_gamma, work, NULL, (int)rows, (int)width, (int)width, epsilon);
72 for (size_t i = 0; i < count; ++i) {
73 if (!isfinite(work[i])) {
74 return -3;
75 }
76 }
77 for (size_t i = 0; i < count; ++i) {
78 output[i] = float_to_bf16(work[i]);
79 }
80 return 0;
81}
