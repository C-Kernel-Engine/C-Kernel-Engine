32/* Bound on discovered CPUs; CPU_SETSIZE is 1024 on Linux. */
33#define CK_TOPOLOGY_MAX_CPUS 1024

35/* Capacity ratio above which the allowed set counts as heterogeneous
36 * (GB10 Cortex-X925 vs Cortex-A725 is ~1.4; SMT siblings are ~1.0). */
37#define CK_TOPOLOGY_HETEROGENEOUS_RATIO_X100 115
38
39/* A CPU belongs to the top tier when its capacity is within 90% of the max
40 * (per-core capacity values differ slightly between otherwise identical
41 * big cores, e.g. 997 vs 1024). */
42#define CK_TOPOLOGY_TOP_TIER_PCT 90

55static long read_cpu_capacity(int cpu)
56{
57 char path[128];
58 snprintf(path, sizeof(path),
59 "/sys/devices/system/cpu/cpu%d/cpu_capacity", cpu);
60 FILE *f = fopen(path, "r");
61 if (f) {
62 long v = -1;
63 if (fscanf(f, "%ld", &v) != 1) v = -1;
64 fclose(f);
65 if (v > 0) return v;
66 }
67 /* Fallback: cpufreq maximum frequency in kHz. */
68 snprintf(path, sizeof(path),
69 "/sys/devices/system/cpu/cpu%d/cpufreq/cpuinfo_max_freq", cpu);
70 f = fopen(path, "r");
71 if (!f) return -1;
72 long v = -1;
73 if (fscanf(f, "%ld", &v) != 1) v = -1;
74 fclose(f);
75 return v;
76}
77#endif

79static void topology_discover(void)
80{
81 ck_topology_state_t *t = &g_topology;
82 t->n_allowed = 1;
83 t->heterogeneous = 0;
84 t->n_big = 1;
85 t->big_cpus[0] = 0;
86
87#ifdef __linux__
88 cpu_set_t allowed;
89 CPU_ZERO(&allowed);
90 if (sched_getaffinity(0, sizeof(allowed), &allowed) != 0) {
91 return; /* keep the conservative single-CPU default */
92 }
93
94 long capacities[CK_TOPOLOGY_MAX_CPUS];
95 int ids[CK_TOPOLOGY_MAX_CPUS];
96 int n = 0;
97 long cap_min = 0, cap_max = 0;
98 for (int cpu = 0; cpu < CPU_SETSIZE && n < CK_TOPOLOGY_MAX_CPUS; ++cpu) {
99 if (!CPU_ISSET(cpu, &allowed)) continue;
100 ids[n] = cpu;
101 capacities[n] = read_cpu_capacity(cpu);
102 if (capacities[n] > 0) {
103 if (cap_max == 0 || capacities[n] > cap_max) cap_max = capacities[n];
104 if (cap_min == 0 || capacities[n] < cap_min) cap_min = capacities[n];
105 }
106 ++n;
107 }
108 if (n == 0) {
109 return;
110 }
111 t->n_allowed = n;
112
113 if (cap_min <= 0 || cap_max * 100 <= cap_min * CK_TOPOLOGY_HETEROGENEOUS_RATIO_X100) {
114 /* Homogeneous or capacity unknown: every allowed CPU is top tier. */
115 t->heterogeneous = 0;
116 t->n_big = n;
117 for (int i = 0; i < n; ++i) t->big_cpus[i] = ids[i];
118 return;
119 }
120
121 t->heterogeneous = 1;
122 int nb = 0;
123 for (int i = 0; i < n && nb < CK_TOPOLOGY_MAX_CPUS; ++i) {
124 if (capacities[i] > 0 && capacities[i] * 100 >= cap_max * CK_TOPOLOGY_TOP_TIER_PCT) {
125 t->big_cpus[nb++] = ids[i];
126 }
127 }
128 if (nb == 0) {
129 t->n_big = n;
130 for (int i = 0; i < n; ++i) t->big_cpus[i] = ids[i];
131 return;
132 }
133 t->n_big = nb;
134#endif
135}

154int ck_topology_big_cpus(int *cpus, int max_cpus)
155{
156 pthread_once(&g_topology_once, topology_init);
157 if (!cpus || max_cpus <= 0) return 0;
158 int n = g_topology.n_big;
159 if (n > max_cpus) n = max_cpus;
160 for (int i = 0; i < n; ++i) {
161 cpus[i] = g_topology.big_cpus[i];
162 }
163 return n;
164}
