7static int span_fits(size_t rows, size_t stride, size_t width, size_t elements)
8{
9 return rows && width && stride >= width &&
10 rows - 1 <= (SIZE_MAX - width) / stride &&
11 (rows - 1) * stride + width <= elements;

14static double dot_bias(const float *input, const float *weight,
15 size_t width, float bias)
16{
17 double sum = (double)bias;
18 for (size_t c = 0; c < width; ++c) {
19 volatile double product = (double)input[c] * (double)weight[c];
20 sum += product;
21 }
22 return sum;
23}

38 * Each lane preserves ordered FP64
39 * multiplication and addition. No horizontal reduction or FMA is introduced. */
40/* Disable contraction without removing FMA from the ISA target. GCC 15

46static void compute_independent_lanes(const float *input, size_t input_stride,
47 const float *weight, size_t weight_stride,

49 size_t rows, size_t width, size_t channels)

52#pragma clang fp contract(off)
53#pragma clang fp exceptions(strict)

55 size_t r = 0;

57 for (size_t o = 0; o < channels; ++o) {

59 size_t c = 0;

61 __m128 a = _mm_loadu_ps(input + r * input_stride + c);
62 __m128 b = _mm_loadu_ps(input + (r + 1) * input_stride + c);
63 __m128 d = _mm_loadu_ps(input + (r + 2) * input_stride + c);
64 __m128 e = _mm_loadu_ps(input + (r + 3) * input_stride + c);
65 _MM_TRANSPOSE4_PS(a, b, d, e);

67 _mm256_set1_pd((double)weight[o * weight_stride + c])));
68 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(b),
69 _mm256_set1_pd((double)weight[o * weight_stride + c + 1])));
70 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(d),
71 _mm256_set1_pd((double)weight[o * weight_stride + c + 2])));
72 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(e),
73 _mm256_set1_pd((double)weight[o * weight_stride + c + 3])));
74 }

77 input[(r+2)*input_stride+c], input[(r+1)*input_stride+c],
78 input[r*input_stride+c]);
79 __m256d w = _mm256_set1_pd((double)weight[o*weight_stride+c]);
80 sum = _mm256_add_pd(sum, _mm256_mul_pd(values, w));
81 }

84 for (size_t lane = 0; lane < 4; ++lane)
85 output[(r+lane)*output_stride+o] = (float)lanes[lane];
86 }
87 }
88 for (; r < rows; ++r) {
89 size_t o = 0;
90 for (; o + 4 <= channels; o += 4) {
91 __m256d sum = bias ? _mm256_cvtps_pd(_mm_loadu_ps(bias + o)) :
92 _mm256_setzero_pd();

94 for (; c + 4 <= width; c += 4) {
95 __m128 a = _mm_loadu_ps(weight + o * weight_stride + c);
96 __m128 b = _mm_loadu_ps(weight + (o + 1) * weight_stride + c);
97 __m128 d = _mm_loadu_ps(weight + (o + 2) * weight_stride + c);
98 __m128 e = _mm_loadu_ps(weight + (o + 3) * weight_stride + c);

100 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(a),
101 _mm256_set1_pd((double)input[r * input_stride + c])));
102 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(b),
103 _mm256_set1_pd((double)input[r * input_stride + c + 1])));
104 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(d),
105 _mm256_set1_pd((double)input[r * input_stride + c + 2])));
106 sum = _mm256_add_pd(sum, _mm256_mul_pd(_mm256_cvtps_pd(e),
107 _mm256_set1_pd((double)input[r * input_stride + c + 3])));
108 }
109 for (; c < width; ++c) {
110 const __m256d w = _mm256_set_pd(weight[(o+3)*weight_stride+c],
111 weight[(o+2)*weight_stride+c], weight[(o+1)*weight_stride+c],
112 weight[o*weight_stride+c]);
113 const __m256d value = _mm256_set1_pd((double)input[r*input_stride+c]);
114 sum = _mm256_add_pd(sum, _mm256_mul_pd(value, w));
115 }
116 _mm_storeu_ps(output+r*output_stride+o, _mm256_cvtpd_ps(sum));
117 }
118 for (; o < channels; ++o)
119 output[r*output_stride+o] = (float)dot_bias(input+r*input_stride,
120 weight+o*weight_stride, width, bias ? bias[o] : 0.0f);
121 }
122}

124/* Validate four independent weight rows with the original per-row L1 order.
125 * No output is published here; a late nonfinite value still rejects atomically. */
126__attribute__((target("avx2"), noinline))
127static int validate_weight_rows(const float *weight, size_t stride,
128 const float *bias, size_t width, size_t channels,
129 double *max_l1, double *max_bias)
130{
131 size_t o = 0;
132 const __m256d sign = _mm256_set1_pd(-0.0);
133 const __m256d limit = _mm256_set1_pd((double)FLT_MAX);
134 for (; o + 4 <= channels; o += 4) {
135 __m256d sums = _mm256_setzero_pd();
136 size_t c = 0;
137 for (; c + 4 <= width; c += 4) {
138 const __m128 sign32 = _mm_set1_ps(-0.0f);
139 const __m128 limit32 = _mm_set1_ps(FLT_MAX);
140 __m128 a = _mm_andnot_ps(sign32, _mm_loadu_ps(weight + o * stride + c));
141 __m128 b = _mm_andnot_ps(sign32, _mm_loadu_ps(weight + (o + 1) * stride + c));
142 __m128 d = _mm_andnot_ps(sign32, _mm_loadu_ps(weight + (o + 2) * stride + c));
143 __m128 e = _mm_andnot_ps(sign32, _mm_loadu_ps(weight + (o + 3) * stride + c));
144 const __m128 finite = _mm_and_ps(_mm_and_ps(_mm_cmple_ps(a, limit32),
145 _mm_cmple_ps(b, limit32)), _mm_and_ps(_mm_cmple_ps(d, limit32),
146 _mm_cmple_ps(e, limit32)));
147 if (_mm_movemask_ps(finite) != 15) return -1;
148 _MM_TRANSPOSE4_PS(a, b, d, e);
149 sums = _mm256_add_pd(sums, _mm256_cvtps_pd(a));
150 sums = _mm256_add_pd(sums, _mm256_cvtps_pd(b));
151 sums = _mm256_add_pd(sums, _mm256_cvtps_pd(d));
152 sums = _mm256_add_pd(sums, _mm256_cvtps_pd(e));
153 }
154 for (; c < width; ++c) {
155 __m256d v = _mm256_set_pd(weight[(o+3)*stride+c], weight[(o+2)*stride+c],
156 weight[(o+1)*stride+c], weight[o*stride+c]);
157 v = _mm256_andnot_pd(sign, v);
158 if (_mm256_movemask_pd(_mm256_cmp_pd(v, limit, _CMP_LE_OQ)) != 15)
159 return -1;
160 sums = _mm256_add_pd(sums, v);
161 }
162 double lanes[4];
163 _mm256_storeu_pd(lanes, sums);
164 for (size_t lane = 0; lane < 4; ++lane) {
165 if (lanes[lane] > *max_l1) *max_l1 = lanes[lane];
166 if (bias) {
167 if (!isfinite(bias[o+lane])) return -1;
168 const double b = fabs((double)bias[o+lane]);
169 if (b > *max_bias) *max_bias = b;
170 }
171 }
172 }
173 for (; o < channels; ++o) {
174 double sum = 0.0;
175 for (size_t c = 0; c < width; ++c) {
176 const float v = weight[o*stride+c];
177 if (!isfinite(v)) return -1;
178 sum += fabs((double)v);
179 }
180 if (sum > *max_l1) *max_l1 = sum;
181 if (bias) {
182 if (!isfinite(bias[o])) return -1;
183 const double b = fabs((double)bias[o]);
184 if (b > *max_bias) *max_bias = b;
185 }
186 }
187 return 0;
188}

200static void compute_projection_tiles(size_t begin, size_t end, void *opaque)
201{
202 const struct projection_tiles *p = opaque;
203 for (size_t tile = begin; tile < end; ++tile) {
204 const size_t row = tile / p->channel_tiles * p->row_tile;
205 const size_t first = tile % p->channel_tiles * projection_tile_channels;
206 const size_t channels_left = p->channels - first, rows_left = p->rows - row;
207 const size_t channels = channels_left < projection_tile_channels ? channels_left : projection_tile_channels;
208 const size_t rows = rows_left < p->row_tile ? rows_left : p->row_tile;
209 compute_independent_lanes(p->input + row * p->input_stride, p->input_stride,
210 p->weight + first * p->weight_stride, p->weight_stride, p->bias ? p->bias + first : NULL,
211 p->output + row * p->output_stride + first, p->output_stride, rows, p->width, channels);
212 }

216static int disjoint(const float *a, size_t na, const float *b, size_t nb)
217{
218 if (!b || !nb) return 1;
219 uintptr_t x = (uintptr_t)a, y = (uintptr_t)b;
220 if (na > (UINTPTR_MAX-x)/sizeof(float) || nb > (UINTPTR_MAX-y)/sizeof(float))
221 return 0;
222 return x + na*sizeof(float) <= y || y + nb*sizeof(float) <= x;
223}

226int linear_rows_checked_f32(
227 const float *input, size_t input_elements, size_t input_stride,
228 const float *weight, size_t weight_elements, size_t weight_stride,
229 const float *bias, size_t bias_elements,
230 float *output, size_t output_elements, size_t output_stride,
231 size_t rows, size_t input_channels, size_t output_channels)
232{
233 if (!input || !weight || !output ||
234 !span_fits(rows, input_stride, input_channels, input_elements) ||
235 !span_fits(output_channels, weight_stride, input_channels, weight_elements) ||
236 !span_fits(rows, output_stride, output_channels, output_elements) ||
237 (bias ? bias_elements < output_channels : bias_elements != 0))
238 return -1;
239
240 /* Validate every accessed value before the first write. A conservative
241 * absolute bound permits one compute pass for ordinary model values.
242 * Values near the FP32 limit retain the exact two-pass staging path. */
243 double max_input = 0.0;
244 double max_weight_l1 = 0.0;
245 double max_bias = 0.0;
246 for (size_t r = 0; r < rows; ++r)
247 for (size_t c = 0; c < input_channels; ++c)
248 {
249 const float value = input[r * input_stride + c];
250 if (!isfinite(value)) return -1;
251 const double absolute = fabs((double)value);
252 if (absolute > max_input) max_input = absolute;
253 }
254#if (defined(__x86_64__) || defined(__i386__)) && (defined(__GNUC__) || defined(__clang__))
255 __builtin_cpu_init();
256 if (output_channels >= 4 && __builtin_cpu_supports("avx2")) {
257 if (validate_weight_rows(weight, weight_stride, bias, input_channels,
258 output_channels, &max_weight_l1, &max_bias))
259 return -1;
260 } else
261#endif
262 for (size_t o = 0; o < output_channels; ++o) {
263 if (bias && !isfinite(bias[o])) return -1;
264 if (bias && fabs((double)bias[o]) > max_bias)
265 max_bias = fabs((double)bias[o]);
266 double weight_l1 = 0.0;
267 for (size_t c = 0; c < input_channels; ++c)
268 {
269 const float value = weight[o * weight_stride + c];
270 if (!isfinite(value)) return -1;
271 weight_l1 += fabs((double)value);
272 }
273 if (weight_l1 > max_weight_l1) max_weight_l1 = weight_l1;
274 }
275 /* At most UINT32_MAX positive terms keep FP64 summation error far below
276 * the 4x guard, including error in the L1 bound. Every result therefore
277 * fits FP32 and publishing cannot fail after the first output write. */
278 if (input_channels <= (SIZE_MAX / 4) && input_channels <= UINT32_MAX &&
279 max_input * max_weight_l1 + max_bias <= (double)FLT_MAX / 4.0) {
280#if (defined(__x86_64__) || defined(__i386__)) && (defined(__GNUC__) || defined(__clang__))
281 __builtin_cpu_init();
282 if ((rows >= 4 || output_channels >= 4) && __builtin_cpu_supports("avx2") &&
283 disjoint(output, output_elements, input, input_elements) &&
284 disjoint(output, output_elements, weight, weight_elements) &&
285 disjoint(output, output_elements, bias, bias_elements)) {
286#if defined(__ELF__)
287 const int narrow = output_channels < 128 && output_channels >= 16 && input_channels >= 256;
288 if (rows >= 16 && input_channels >= 16 && (output_channels >= 128 || narrow) &&
289 ck_get_num_threads && ck_threadpool_global && ck_threadpool_in_dispatch &&
290 ck_threadpool_n_threads && ck_threadpool_parallel_for_n) {
291 struct projection_tiles work = {
292 .input = input, .weight = weight, .bias = bias, .output = output,
293 .input_stride = input_stride, .weight_stride = weight_stride,
294 .output_stride = output_stride, .rows = rows, .width = input_channels,
295 .channels = output_channels, .row_tile = narrow ? 4 : rows,
296 .channel_tiles = (output_channels - 1) / projection_tile_channels + 1,
297 };
298 const size_t row_tiles = (rows - 1) / work.row_tile + 1;
299 ck_checked_parallel_for(row_tiles * work.channel_tiles, 1, compute_projection_tiles, &work);
300 return 0;
301 }
302#endif
303 compute_independent_lanes(input, input_stride, weight, weight_stride, bias,
304 output, output_stride, rows, input_channels,
305 output_channels);
306 return 0;
307 }
308#endif
309 for (size_t r = 0; r < rows; ++r)
310 for (size_t o = 0; o < output_channels; ++o)
311 output[r * output_stride + o] = (float)dot_bias(
312 input + r * input_stride, weight + o * weight_stride,
313 input_channels, bias ? bias[o] : 0.0f);
314 return 0;
315 }
316 for (size_t r = 0; r < rows; ++r)
317 for (size_t o = 0; o < output_channels; ++o) {
318 const double result = dot_bias(input + r * input_stride,
319 weight + o * weight_stride,
320 input_channels, bias ? bias[o] : 0.0f);
321 if (!isfinite(result) || fabs(result) > FLT_MAX) return -1;
322 }
323 for (size_t r = 0; r < rows; ++r)
324 for (size_t o = 0; o < output_channels; ++o)
325 output[r * output_stride + o] = (float)dot_bias(
326 input + r * input_stride, weight + o * weight_stride,
327 input_channels, bias ? bias[o] : 0.0f);
328 return 0;
329}
