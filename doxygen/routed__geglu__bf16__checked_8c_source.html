53static int routed_geglu_bf16_checked(
54 const uint16_t *input, const int32_t *ids, const float *scores,
55 const uint16_t *gate_up, const uint16_t *down, uint16_t *out, uint16_t *routes,
56 void *workspace, size_t workspace_bytes, size_t rows, size_t width,
57 size_t inner, size_t experts, size_t top_k, int projection_mode,
58 size_t hidden_elements, size_t selected_elements, size_t route_elements,
59 size_t gate_elements, size_t down_elements)
60{
61 // Each sized domain fits a signed 32-bit BLAS extent. Cap products before
62 // multiplying so all subsequent byte calculations also fit size_t.
63 if (!rows || !width || !inner || !experts || !top_k || top_k > experts ||
64 rows > INT_MAX || width > INT_MAX || inner > INT_MAX / 2 || experts > INT32_MAX ||
65 width > SIZE_MAX / 64 || inner > SIZE_MAX / 64 || rows > SIZE_MAX / (64 * (width + inner)) ||
66 experts > SIZE_MAX / (64 * width) / inner || top_k > SIZE_MAX / (64 * rows) / width)
67 return -2;
68 const size_t rh = rows * width, ri = rows * inner;
69 const size_t rk = rows * top_k, routed = rk * width;
70 const size_t hi = width * inner;
71 const size_t blas_elements = rh + 2 * ri + (projection_mode == 2 ? 0 : 2 * hi);
72 const size_t blas_bytes = blas_elements * (projection_mode == 2 ? sizeof(float) : sizeof(double));
73 const size_t work_bytes = blas_bytes + 4 * inner + 2 * (4 * rh + 5 * ri + routed);
74 if (workspace_bytes < work_bytes ||
75 hidden_elements != rh || selected_elements != rk || route_elements != routed ||
76 gate_elements != experts * 2 * hi || down_elements != experts * hi)
77 return -2;
78 const void *inputs[] = {input, ids, scores, gate_up, down};
79 const size_t sizes[] = {rh * 2, rk * 4, rk * 4, experts * hi * 4, experts * hi * 2};
80 const size_t aligns[] = {2, 4, 4, 2, 2};
81 if (!span(out, rh * 2, 2) || !span(routes, routed * 2, 2) ||
82 !span(workspace, work_bytes, _Alignof(double)) ||
83 overlaps(out, rh * 2, routes, routed * 2) ||
84 overlaps(out, rh * 2, workspace, work_bytes) ||
85 overlaps(routes, routed * 2, workspace, work_bytes))
86 return -2;
87 for (size_t i = 0; i < sizeof(inputs) / sizeof(inputs[0]); ++i)
88 if (!span(inputs[i], sizes[i], aligns[i]) ||
89 overlaps(inputs[i], sizes[i], out, rh * 2) ||
90 overlaps(inputs[i], sizes[i], routes, routed * 2) ||
91 overlaps(inputs[i], sizes[i], workspace, work_bytes))
92 return -2;
93
94 for (size_t row = 0; row < rows; ++row) {
95 for (size_t k = 0; k < top_k; ++k) {
96 const size_t index = row * top_k + k;
97 if (ids[index] < 0 || (size_t)ids[index] >= experts || !isfinite(scores[index]))
98 return -3;
99 for (size_t previous = 0; previous < k; ++previous)
100 if (ids[index] == ids[row * top_k + previous])
101 return -3;
102 }
103 }
104
105 double *blas = workspace;
106 float *gelu = (float *)((uint8_t *)workspace + blas_bytes);
107 uint16_t *gathered = (uint16_t *)(gelu + inner);
108 uint16_t *gate = gathered + rh;
109 uint16_t *active = gate + 2 * ri;
110 uint16_t *projected = active + ri;
111 uint16_t *stage = projected + rh;
112 uint16_t *accum = stage + rh + 2 * ri;
113 uint16_t *route_stage = accum + rh;
114 memset(accum, 0, rh * sizeof(*accum));
115 const projection_workspace projection = {blas, blas_elements, stage, rh + 2 * ri, projection_mode};
116
117 // Match expert-major index_add order. The route slot is used only to
118 // associate the input row and scaled probability with this expert.
119 for (size_t expert = 0; expert < experts; ++expert) {
120 size_t count = 0;
121 for (size_t index = 0; index < rk; ++index)
122 if ((size_t)ids[index] == expert) {
123 memcpy(gathered + count * width, input + (index / top_k) * width, width * 2);
124 ++count;
125 }
126 if (!count)
127 continue;
128 int status = project(gathered, gate_up + expert * 2 * hi, gate, count, width, 2 * inner, projection);
129 if (status)
130 return status;
131 for (size_t row = 0; row < count; ++row) {
132 status = ck_geglu_tanh_bf16_checked(gate + row * 2 * inner,
133 gate + row * 2 * inner + inner, active + row * inner, gelu, inner);
134 if (status)
135 return status;
136 }
137 status = project(active, down + expert * hi, projected, count, inner, width, projection);
138 if (status)
139 return status;
140 size_t row = 0;
141 for (size_t index = 0; index < rk; ++index) {
142 if ((size_t)ids[index] != expert)
143 continue;
144 uint16_t *destination = accum + (index / top_k) * width;
145 const uint16_t *unweighted = projected + row * width;
146 memcpy(route_stage + index * width, unweighted, width * 2);
147 for (size_t col = 0; col < width; ++col) {
148 const uint16_t scaled = float_to_bf16(bf16_to_float(unweighted[col]) * scores[index]);
149 const uint16_t sum = float_to_bf16(bf16_to_float(destination[col]) + bf16_to_float(scaled));
150 if (!isfinite(bf16_to_float(sum)))
151 return -3;
152 destination[col] = sum;
153 }
154 ++row;
155 }
156 }
157 memcpy(out, accum, rh * 2);
158 memcpy(routes, route_stage, routed * 2);
159 return 0;
160}
