13/* Input rows are finite. Emit little-endian Q8_0 blocks with nearest-even codes. */
14static inline int ck_pack_q8_activation_rne(const float *input, unsigned char *packed, size_t width)
15{
16 for (size_t i = 0; i < width; i += QK8_0, packed += sizeof(block_q8_0)) {
17 float peak = 0.0f;
18 for (size_t j = 0; j < QK8_0; ++j)
19 peak = fmaxf(peak, fabsf(input[i + j]));
20 const uint16_t scale = ck_fp32_to_fp16_soft(peak / 127.0f);
21 if ((scale & UINT16_C(0x7c00)) == UINT16_C(0x7c00)) return -1;
22 packed[0] = (unsigned char)scale;
23 packed[1] = (unsigned char)(scale >> 8);
24 const float inverse = scale ? 127.0f / peak : 0.0f;
25 for (size_t j = 0; j < QK8_0; ++j) {
26 const float value = input[i + j] * inverse;
27 const float magnitude = fminf(127.0f, fabsf(value));
28 int rounded = (int)magnitude;
29 const float fraction = magnitude - (float)rounded;
30 if (fraction > 0.5f || (fraction == 0.5f && (rounded & 1))) ++rounded;
31 packed[2 + j] = (unsigned char)(value < 0.0f ? -rounded : rounded);
32 }
33 }
34 return 0;
35}

37static inline float ck_q8_activation_dot_scalar(const unsigned char *weight, const unsigned char *input,
38 size_t width)
39{
40 /* The x86 GGML reduction groups four signed products into each of
41 * eight FP32 FMA lanes, then combines lanes in a fixed tree. */
42 float lanes[8] = {0};
43 for (size_t i = 0; i < width; i += QK8_0, weight += sizeof(block_q8_0), input += sizeof(block_q8_0)) {
44 const float dw = ck_fp16_to_fp32_soft((uint16_t)(weight[0] | ((uint16_t)weight[1] << 8)));
45 const float dx = ck_fp16_to_fp32_soft((uint16_t)(input[0] | ((uint16_t)input[1] << 8)));
46 const float scale = dw * dx;
47 for (size_t lane = 0; lane < 8; ++lane) {
48 int dot = 0;
49 for (size_t j = 2 + 4 * lane; j < 6 + 4 * lane; ++j) {
50 const int w = weight[j], x = input[j];
51 dot += (w < 128 ? w : w - 256) * (x < 128 ? x : x - 256);
52 }
53 lanes[lane] = fmaf(scale, (float)dot, lanes[lane]);
54 }
55 }
56 return ((lanes[0] + lanes[4]) + (lanes[2] + lanes[6]))
57 + ((lanes[1] + lanes[5]) + (lanes[3] + lanes[7]));
58}

61static inline __attribute__((target("avx2"))) __m256
62ck_q8_activation_products(const unsigned char *weight, __m256i activation)
63{
64 const __m256i w = _mm256_loadu_si256((const __m256i *)(weight + 2));
65 /* Packed activations are in [-127, 127], so the signed transfer and
66 * each pair sum are exact even for a weight code of -128. */
67 const __m256i pairs = _mm256_maddubs_epi16(_mm256_abs_epi8(w), _mm256_sign_epi8(activation, w));
68 return _mm256_cvtepi32_ps(_mm256_madd_epi16(pairs, _mm256_set1_epi16(1)));
69}

71static inline __attribute__((target("avx2"))) float
72ck_q8_activation_reduce(__m256 sum)
73{
74 const __m128 halves = _mm_add_ps(_mm256_castps256_ps128(sum), _mm256_extractf128_ps(sum, 1));
75 const __m128 pairs = _mm_add_ps(halves, _mm_movehl_ps(halves, halves));
76 return _mm_cvtss_f32(_mm_add_ss(pairs, _mm_movehdup_ps(pairs)));
77}

86static __attribute__((target("avx2,fma,f16c"))) float
87ck_q8_activation_dot_avx2_f16c(const unsigned char *weight, const unsigned char *input, size_t width)
88{
89 __m256 sum = _mm256_setzero_ps();
90 for (size_t i = 0; i < width; i += QK8_0, weight += sizeof(block_q8_0), input += sizeof(block_q8_0)) {
91 const __m256i x = _mm256_loadu_si256((const __m256i *)(input + 2));
92 const uint32_t scales = ck_q8_x86_scale_bits(weight) | ((uint32_t)ck_q8_x86_scale_bits(input) << 16);
93 const __m128 decoded = _mm_cvtph_ps(_mm_cvtsi32_si128((int)scales));
94 const __m128 scale = _mm_mul_ss(decoded, _mm_shuffle_ps(decoded, decoded, _MM_SHUFFLE(1, 1, 1, 1)));
95 sum = _mm256_fmadd_ps(_mm256_broadcastss_ps(scale), ck_q8_activation_products(weight, x), sum);
96 }
97 return ck_q8_activation_reduce(sum);
98}

100static __attribute__((target("avx2,fma,f16c"))) void
101ck_q8_activation_dot_pair_avx2_f16c(const unsigned char *gate, const unsigned char *up,
102 const unsigned char *input, size_t width, float *g, float *u)
103{
104 __m256 gate_sum = _mm256_setzero_ps(), up_sum = _mm256_setzero_ps();
105 for (size_t i = 0; i < width; i += QK8_0) {
106 const __m256i x = _mm256_loadu_si256((const __m256i *)(input + 2));
107 const uint32_t weights = ck_q8_x86_scale_bits(gate) | ((uint32_t)ck_q8_x86_scale_bits(up) << 16);
108 const __m128i halves = _mm_set_epi32(0, 0, ck_q8_x86_scale_bits(input), (int)weights);
109 const __m128 decoded = _mm_cvtph_ps(halves);
110 const __m128 scales = _mm_mul_ps(decoded, _mm_shuffle_ps(decoded, decoded, _MM_SHUFFLE(2, 2, 2, 2)));
111 const __m128 up_scale = _mm_shuffle_ps(scales, scales, _MM_SHUFFLE(1, 1, 1, 1));
112 gate_sum = _mm256_fmadd_ps(_mm256_broadcastss_ps(scales), ck_q8_activation_products(gate, x), gate_sum);
113 up_sum = _mm256_fmadd_ps(_mm256_broadcastss_ps(up_scale), ck_q8_activation_products(up, x), up_sum);
114 gate += sizeof(block_q8_0);
115 up += sizeof(block_q8_0);
116 input += sizeof(block_q8_0);
117 }
118 *g = ck_q8_activation_reduce(gate_sum);
119 *u = ck_q8_activation_reduce(up_sum);
120}
