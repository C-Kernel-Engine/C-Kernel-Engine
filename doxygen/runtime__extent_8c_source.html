13 int32_t *valid_extent) {
14 if (!valid_extent || (count && !values) || min_value < 0 ||
15 max_value < min_value) return CK_RUNTIME_EXTENT_INVALID;
16 if (count > value_elements || initial_extent > capacity)
17 return CK_RUNTIME_EXTENT_LIMIT;
18 if (initial_extent > INT32_MAX || capacity > INT32_MAX)
19 return CK_RUNTIME_EXTENT_OVERFLOW;
20 if (count > SIZE_MAX / sizeof(int32_t))
21 return CK_RUNTIME_EXTENT_OVERFLOW;
22
23 int32_t total = (int32_t)initial_extent;
24 for (size_t i = 0; i < count; ++i) {
25 const int32_t item = values[i];
26 if (item < min_value || item > max_value)
27 return CK_RUNTIME_EXTENT_INVALID;
28 if (item > INT32_MAX - total)
29 return CK_RUNTIME_EXTENT_OVERFLOW;
30 total += item;
31 if ((size_t)total > capacity)
32 return CK_RUNTIME_EXTENT_LIMIT;
33 }
34 *valid_extent = total;
35 return CK_RUNTIME_EXTENT_OK;
36}

52 channels - 1 > (SIZE_MAX - valid_frames) / output_stride)
53 return CK_RUNTIME_EXTENT_OVERFLOW;
54 const size_t input_required = (channels - 1) * input_stride + valid_frames;
55 const size_t output_required = (channels - 1) * output_stride + valid_frames;
56 if (input_required > SIZE_MAX / sizeof(float) ||
57 output_required > SIZE_MAX / sizeof(float))
58 return CK_RUNTIME_EXTENT_OVERFLOW;
59 if (input_elements < input_required || output_elements < output_required)
60 return CK_RUNTIME_EXTENT_LIMIT;
61 for (size_t channel = 0; channel < channels; ++channel) {
62 for (size_t frame = 0; frame < valid_frames; ++frame) {
63 output[channel * output_stride + frame] =

65 }
66 }
