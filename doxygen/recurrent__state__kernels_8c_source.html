25static void ck_recurrent_conv_state_channel_range(int begin,
26 int end,
27 void *opaque) {
28 const ck_recurrent_conv_state_args_t *args =
29 (const ck_recurrent_conv_state_args_t *)opaque;
30
31 const int first_seq = begin / args->channels;
32 const int last_seq = (end - 1) / args->channels;
33 for (int seq = first_seq; seq <= last_seq; ++seq) {
34 int ch_begin = begin - seq * args->channels;
35 int ch_end = end - seq * args->channels;
36 if (ch_begin < 0) ch_begin = 0;
37 if (ch_end > args->channels) ch_end = args->channels;
38
39 const size_t state_seq_offset =
40 (size_t)seq * (size_t)args->channels *
41 (size_t)args->history_len;
42 const size_t conv_seq_offset =
43 (size_t)seq * (size_t)args->channels *
44 (size_t)args->total_len;
45 for (int ch = ch_begin; ch < ch_end; ++ch) {
46 memcpy(
47 args->conv_x + conv_seq_offset +
48 (size_t)ch * (size_t)args->total_len,
49 args->state_in + state_seq_offset +
50 (size_t)ch * (size_t)args->history_len,
51 (size_t)args->history_len * sizeof(float));
52 }
53
54 const int q_begin = ch_begin;
55 const int q_end = ch_end < args->q_dim ? ch_end : args->q_dim;
56 const int k_begin = ch_begin > args->q_dim ?
57 ch_begin - args->q_dim : 0;
58 const int k_end = ch_end < args->q_dim + args->k_dim ?
59 ch_end - args->q_dim : args->k_dim;
60 const int v_begin = ch_begin > args->q_dim + args->k_dim ?
61 ch_begin - args->q_dim - args->k_dim : 0;
62 const int v_end = ch_end - args->q_dim - args->k_dim < args->v_dim ?
63 ch_end - args->q_dim - args->k_dim : args->v_dim;
64
65 for (int tok = 0; tok < args->num_tokens; ++tok) {
66 const int row = seq * args->num_tokens + tok;
67 for (int col = q_begin; col < q_end; ++col) {
68 args->conv_x[conv_seq_offset +
69 (size_t)col * (size_t)args->total_len +
70 (size_t)(args->history_len + tok)] =
71 args->q[(size_t)row * (size_t)args->q_dim + (size_t)col];
72 }
73 for (int col = k_begin; col < k_end; ++col) {
74 args->conv_x[conv_seq_offset +
75 (size_t)(args->q_dim + col) * (size_t)args->total_len +
76 (size_t)(args->history_len + tok)] =
77 args->k[(size_t)row * (size_t)args->k_dim + (size_t)col];
78 }
79 for (int col = v_begin; col < v_end; ++col) {
80 args->conv_x[conv_seq_offset +
81 (size_t)(args->q_dim + args->k_dim + col) *
82 (size_t)args->total_len +
83 (size_t)(args->history_len + tok)] =
84 args->v[(size_t)row * (size_t)args->v_dim + (size_t)col];
85 }
86 }
87
88 for (int ch = ch_begin; ch < ch_end; ++ch) {
89 memcpy(
90 args->state_out + state_seq_offset +
91 (size_t)ch * (size_t)args->history_len,
92 args->conv_x + conv_seq_offset +
93 (size_t)ch * (size_t)args->total_len +
94 (size_t)args->num_tokens,
95 (size_t)args->history_len * sizeof(float));
96 }
97 }

100void recurrent_conv_state_update_forward(const float *state_in,
101 const float *q,
102 const float *k,
103 const float *v,
104 float *conv_x,
105 float *state_out,
106 int history_len,
107 int num_seqs,
108 int num_tokens,
109 int q_dim,
110 int k_dim,
111 int v_dim) {
112 if (!state_in || !q || !k || !v || !conv_x || !state_out ||
113 history_len < 0 || num_seqs <= 0 || num_tokens < 0 ||
114 q_dim < 0 || k_dim < 0 || v_dim < 0 ||
115 q_dim > INT_MAX - k_dim || q_dim + k_dim > INT_MAX - v_dim ||
116 history_len > INT_MAX - num_tokens) {
117 return;
118 }
119 const int channels = q_dim + k_dim + v_dim;
120 const int total_len = history_len + num_tokens;
121 if (channels <= 0 || total_len <= 0 || num_seqs > INT_MAX / channels ||
122 (num_tokens > 0 && num_seqs > INT_MAX / num_tokens)) {
123 return;
124 }
125
126 ck_recurrent_conv_state_args_t args = {
127 state_in, q, k, v, conv_x, state_out,
128 history_len, num_tokens, q_dim, k_dim, v_dim, channels, total_len,
129 };
130 const int jobs = num_seqs * channels;
131 const size_t elements = (size_t)jobs * (size_t)num_tokens;
132 const size_t minimum_elements_per_thread = 32768u;
133 if (num_tokens <= 1 || elements < 2u * minimum_elements_per_thread) {
134 ck_recurrent_conv_state_channel_range(0, jobs, &args);
135 return;
136 }
137
138 ck_threadpool_t *pool = ck_threadpool_global();
139 int active = pool ? ck_threadpool_n_threads(pool) : 1;
140 size_t useful_threads =
141 (elements + minimum_elements_per_thread - 1u) /
142 minimum_elements_per_thread;
143 if ((size_t)active > useful_threads) active = (int)useful_threads;
144 if (active > jobs) active = jobs;
145
146 if (active > 1 && ck_threadpool_thread_id(pool) <= 0) {
147 ck_threadpool_parallel_for_n(
148 pool, active, 0, jobs, 32,
149 ck_recurrent_conv_state_channel_range, &args);
150 } else {
151 ck_recurrent_conv_state_channel_range(0, jobs, &args);
152 }
153}

155static int recurrent_conv_backward_extents(int history_len,
156 int num_seqs,
157 int num_tokens,
158 int q_dim,
159 int k_dim,
160 int v_dim,
161 int *channels_out,
162 int *total_len_out,
163 size_t *elements_out) {
164 if (history_len < 0 || num_seqs <= 0 || num_tokens < 0 || q_dim < 0 ||
165 k_dim < 0 || v_dim < 0 || q_dim > INT_MAX - k_dim ||
166 q_dim + k_dim > INT_MAX - v_dim || history_len > INT_MAX - num_tokens) {
167 return 0;
168 }
169 const int channels = q_dim + k_dim + v_dim;
170 const int total_len = history_len + num_tokens;
171 if (channels == 0 || total_len == 0) {
172 return 0;
173 }
174 size_t elements = (size_t)num_seqs;
175 if ((size_t)total_len > SIZE_MAX / elements) {
176 return 0;
177 }
178 elements *= (size_t)total_len;
179 if ((size_t)channels > SIZE_MAX / elements) {
180 return 0;
181 }
182 *channels_out = channels;
183 *total_len_out = total_len;
184 *elements_out = elements * (size_t)channels;
185 return 1;
186}
