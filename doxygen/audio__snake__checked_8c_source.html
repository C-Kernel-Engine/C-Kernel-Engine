6static int snake_value(float x, float alpha, float *value) {
7 const float argument = alpha * x;
8 if (!isfinite(argument)) return 0;
9 const float sine = sinf(argument);
10 const float inverse = 1.0f / alpha;
11 const float square = sine * sine;
12 /* Preserve the FP32 multiplication before addition even when the model
13 * library is built with FMA enabled. */
14 volatile float scaled = inverse * square;
15 const float result = x + scaled;
16 if (!isfinite(sine) || !isfinite(inverse) || !isfinite(result)) return 0;
17 *value = result;
18 return 1;
19}
