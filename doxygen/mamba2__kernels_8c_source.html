9static int ck_mamba_debug_enabled(void) {
10 static int cached = -1;
11 if (cached < 0) {
12 const char *v = getenv("CK_DEBUG_MAMBA");
13 cached = (v && v[0] && v[0] != '0') ? 1 : 0;
14 }

18static void ck_mamba_debug_finite(const char *name, const float *x, size_t n) {
19 if (!ck_mamba_debug_enabled() || !x || n == 0) {
20 return;
21 }
22 size_t finite = 0, nan = 0, inf = 0;
23 float min_v = 0.0f, max_v = 0.0f;
24 for (size_t i = 0; i < n; ++i) {
25 const float v = x[i];
26 if (isnan(v)) {
27 ++nan;
28 } else if (!isfinite(v)) {
29 ++inf;
30 } else {
31 if (finite == 0 || v < min_v) min_v = v;
32 if (finite == 0 || v > max_v) max_v = v;
33 ++finite;
34 }
35 }
36 fprintf(stderr, "[CK_DEBUG_MAMBA] %s finite=%zu/%zu nan=%zu inf=%zu min=%g max=%g\n",
37 name, finite, n, nan, inf, finite ? (double)min_v : 0.0, finite ? (double)max_v : 0.0);
38}

40static inline float mamba2_sigmoid_f32(float x) {
41 if (x >= 0.0f) {
42 const float z = expf(-x);
43 return 1.0f / (1.0f + z);
44 }
45 {
46 const float z = expf(x);
47 return z / (1.0f + z);
48 }
49}

51static inline float mamba2_silu_f32(float x) {
52 return x * mamba2_sigmoid_f32(x);
53}

55static inline float mamba2_softplus_f32(float x) {
56 if (x > 20.0f) {
57 return x;
58 }
59 if (x < -20.0f) {
60 return expf(x);
61 }
62 return log1pf(expf(x));
63}

102void mamba2_conv1d_decode_f32(const float *state_in,
103 const float *x,
104 const float *weight,
105 const float *bias,
106 float *conv_out,
107 float *state_out,
108 int rows,
109 int conv_dim,
110 int kernel_size) {
111 if (!state_in || !x || !weight || !conv_out || !state_out ||
112 rows <= 0 || conv_dim <= 0 || kernel_size <= 0) {
113 return;
114 }
115
116 ck_mamba_debug_finite("conv.state_in", state_in, (size_t)conv_dim * (size_t)kernel_size);
117 ck_mamba_debug_finite("conv.x[0]", x, (size_t)conv_dim);
118
119 /* State layout is [conv_dim, kernel_size]. Rows are time steps, not
120 * independent batch entries. Keep a local rolling state so prefill scans
121 * the prompt in order and decode (rows=1) is the same contract.
122 */
123 float state_work[(size_t)conv_dim * (size_t)kernel_size];
124 memcpy(state_work, state_in, (size_t)conv_dim * (size_t)kernel_size * sizeof(float));
125
126 for (int row = 0; row < rows; ++row) {
127 for (int ch = 0; ch < conv_dim; ++ch) {
128 const size_t base = (size_t)ch * (size_t)kernel_size;
129 for (int k = 0; k < kernel_size - 1; ++k) {
130 state_work[base + (size_t)k] = state_work[base + (size_t)k + 1u];
131 }
132 state_work[base + (size_t)kernel_size - 1u] =
133 x[(size_t)row * (size_t)conv_dim + (size_t)ch];
134
135 float acc = bias ? bias[ch] : 0.0f;
136 for (int k = 0; k < kernel_size; ++k) {
137 acc += state_work[base + (size_t)k] *
138 weight[(size_t)ch * (size_t)kernel_size + (size_t)k];
139 }
140 conv_out[(size_t)row * (size_t)conv_dim + (size_t)ch] = mamba2_silu_f32(acc);
141 }
142 }
143
144 memcpy(state_out, state_work, (size_t)conv_dim * (size_t)kernel_size * sizeof(float));
145 ck_mamba_debug_finite("conv.out[0]", conv_out, (size_t)conv_dim);
146 ck_mamba_debug_finite("conv.state_out", state_out, (size_t)conv_dim * (size_t)kernel_size);
147}

149void mamba2_dt_softplus_f32(const float *dt,
150 const float *dt_bias,
151 float *dt_out,
152 int rows,
153 int num_heads,
154 float dt_min,
155 float dt_max) {
156 if (!dt || !dt_out || rows <= 0 || num_heads <= 0) {
157 return;
158 }
159
160 ck_mamba_debug_finite("dt.in[0]", dt, (size_t)num_heads);
161 ck_mamba_debug_finite("dt.bias", dt_bias, (size_t)num_heads);
162 for (int row = 0; row < rows; ++row) {
163 for (int h = 0; h < num_heads; ++h) {
164 float v = dt[(size_t)row * (size_t)num_heads + (size_t)h];
165 if (dt_bias) {
166 v += dt_bias[h];
167 }
168 v = mamba2_softplus_f32(v);
169 if (dt_min < dt_max) {
170 if (v < dt_min) {
171 v = dt_min;
172 } else if (v > dt_max) {
173 v = dt_max;
174 }
175 }
176 dt_out[(size_t)row * (size_t)num_heads + (size_t)h] = v;
177 }
178 }
179 ck_mamba_debug_finite("dt.out[0]", dt_out, (size_t)num_heads);
180}

182void mamba2_selective_state_update_decode_f32(const float *state_in,
183 const float *x,
184 const float *dt,
185 const float *a,

187 const float *c,
188 const float *d,
189 float *state_out,

192 int num_heads,
193 int head_dim,
194 int state_dim,
195 int num_groups) {
196 if (!state_in || !x || !dt || !a || !b || !c || !d || !state_out || !y ||
197 rows <= 0 || num_heads <= 0 || head_dim <= 0 || state_dim <= 0 || num_groups <= 0) {
198 return;
199 }

201 const int packed_xbc = (x == b && b == c);
202 const size_t inner_dim = (size_t)num_heads * (size_t)head_dim;
203 const size_t bc_dim = (size_t)num_groups * (size_t)state_dim;
204 const size_t packed_stride = inner_dim + 2u * bc_dim;
205
206 for (int row = 0; row < rows; ++row) {
207 const float *packed_row = packed_xbc ? x + (size_t)row * packed_stride : NULL;
208 for (int h = 0; h < num_heads; ++h) {
209 const int heads_per_group = (num_heads + num_groups - 1) / num_groups;
210 int group = h / heads_per_group;
211 if (group >= num_groups) group = num_groups - 1;
212 const float dt_h = dt[(size_t)row * (size_t)num_heads + (size_t)h];
213 const float d_a = expf(dt_h * a[h]);
214 const float d_h = d[h];
215 const float *b_row = packed_xbc
216 ? (packed_row + inner_dim + (size_t)group * (size_t)state_dim)
217 : (b + ((size_t)row * (size_t)num_groups + (size_t)group) * (size_t)state_dim);
218 const float *c_row = packed_xbc
219 ? (packed_row + inner_dim + bc_dim + (size_t)group * (size_t)state_dim)
220 : (c + ((size_t)row * (size_t)num_groups + (size_t)group) * (size_t)state_dim);
221
222 for (int hd = 0; hd < head_dim; ++hd) {
223 const size_t x_idx = ((size_t)row * (size_t)num_heads + (size_t)h) * (size_t)head_dim + (size_t)hd;
224 const float x_val = packed_xbc ? packed_row[(size_t)h * (size_t)head_dim + (size_t)hd] : x[x_idx];
225 const size_t state_base =
226 (((size_t)row * (size_t)num_heads + (size_t)h) * (size_t)head_dim + (size_t)hd) *
227 (size_t)state_dim;
228 float acc = 0.0f;
229 for (int s = 0; s < state_dim; ++s) {
230 const size_t si = state_base + (size_t)s;
231 const float new_state = state_in[si] * d_a + dt_h * b_row[s] * x_val;
232 state_out[si] = new_state;
233 acc += new_state * c_row[s];
234 }
235 y[x_idx] = acc + d_h * x_val;
236 }
237 }
238 }

317void mamba2_rmsnorm_gate_f32(const float *x,
318 const float *gate,
319 const float *weight,
320 float *out,
321 int rows,
322 int inner_dim,
323 int group_size,
324 float eps) {
325 if (!x || !gate || !weight || !out || rows <= 0 || inner_dim <= 0 || group_size <= 0) {
326 return;
327 }
328
329 ck_mamba_debug_finite("rmsgate.x[0]", x, (size_t)inner_dim);
330 ck_mamba_debug_finite("rmsgate.gate[0]", gate, (size_t)inner_dim);
331 ck_mamba_debug_finite("rmsgate.weight", weight, (size_t)inner_dim);
332 for (int row = 0; row < rows; ++row) {
333 const float *x_row = x + (size_t)row * (size_t)inner_dim;
334 const float *gate_row = gate + (size_t)row * (size_t)inner_dim;
335 float *out_row = out + (size_t)row * (size_t)inner_dim;
336
337 for (int start = 0; start < inner_dim; start += group_size) {
338 int end = start + group_size;
339 if (end > inner_dim) {
340 end = inner_dim;
341 }
342 const int count = end - start;
343 float ms = 0.0f;
344 for (int col = start; col < end; ++col) {
345 const float gated = x_row[col] * mamba2_silu_f32(gate_row[col]);
346 ms += gated * gated;
347 }
348 const float inv_rms = 1.0f / sqrtf(ms / (float)count + eps);
349 for (int col = start; col < end; ++col) {
350 const float gated = x_row[col] * mamba2_silu_f32(gate_row[col]);
351 out_row[col] = gated * inv_rms * weight[col];
352 }
353 }
354 }
355 ck_mamba_debug_finite("rmsgate.out[0]", out, (size_t)inner_dim);
356}
