11{

47static int valid_session(const ck_diffusion_session *s)
48{
49 if (!s || (uintptr_t)s % _Alignof(ck_diffusion_session) || !s->arena || !s->planes ||
50 (uintptr_t)s->planes % _Alignof(ck_diffusion_cache_plane) ||
51 !s->prefill.entry || !s->append.entry || !s->step || !s->plane_count ||
52 s->plane_count > SIZE_MAX / sizeof(*s->planes) ||
53 !s->kv_width || !s->latent_elements || !s->prompt_rows || !s->block_rows ||
54 !s->vocabulary || s->vocabulary > INT32_MAX || s->capacity > INT32_MAX ||
55 s->block_rows > (size_t)INT32_MAX - s->capacity || s->prompt_rows > s->capacity ||
56 s->block_rows > s->capacity - s->prompt_rows || s->capacity > SIZE_MAX / s->kv_width ||
57 s->prompt_rows > SIZE_MAX / 4 || s->block_rows > SIZE_MAX / 4 ||
58 s->valid_rows > s->capacity ||
59 (s->valid_rows && (s->valid_rows < s->prompt_rows ||
60 (s->valid_rows - s->prompt_rows) % s->block_rows)))
61 return 0;
62
63 span arena, bindings, state;
64 if (!region(&arena, s->arena, s->arena_bytes, 1) ||
65 !region(&bindings, s->planes, s->plane_count * sizeof(*s->planes), 1) ||
66 !region(&state, s, sizeof(*s), 1) ||
67 !disjoint(arena, state) || !disjoint(bindings, state) || !disjoint(bindings, arena))
68 return 0;
69
70 const size_t p = s->prompt_rows, b = s->block_rows, n = s->latent_elements;
71 const ck_diffusion_step_bindings *d = &s->step_bindings;
72 const size_t offsets[] = {s->prefill.token_ids, s->prefill.positions, s->prefill.ranges,
73 s->append.token_ids, s->append.positions, s->append.ranges,
74 s->positions, s->conditional_ranges, s->unconditional_ranges, s->guidance,
75 d->latent, d->next_latent, d->noise, d->gamma, d->self_condition, d->conditional_mean, d->token_ids};
76 const size_t counts[] = {p,p,4*p,b,b,4*b,b,4*b,4*b,1,n,n,n,2,n,n,b};
77 const size_t widths[] = {4,4,4,4,4,4,4,4,4,4,8,8,8,8,4,4,4};
78 span fixed[sizeof offsets / sizeof offsets[0]];
79 for (size_t i = 0; i < sizeof fixed / sizeof fixed[0]; ++i) {
80 if (!slot(s, &fixed[i], offsets[i], counts[i], widths[i])) return 0;
81 for (size_t j = 0; j < i; ++j)
82 if (!disjoint(fixed[i], fixed[j])) return 0;
83 }
84
85 for (size_t i = 0; i < s->plane_count; ++i) {
86 span here[3];
87 if (!plane_slots(s, i, here)) return 0;
88 for (size_t k = 0; k < 3; ++k) {
89 for (size_t j = 0; j < sizeof fixed / sizeof fixed[0]; ++j)
90 if (!disjoint(here[k], fixed[j])) return 0;
91 for (size_t j = 0; j < k; ++j)
92 if (!disjoint(here[k], here[j])) return 0;
93 }
94 for (size_t j = 0; j < i; ++j) {
95 span previous[3];
96 if (!plane_slots(s, j, previous)) return 0;
97 for (size_t k = 0; k < 3; ++k)
98 for (size_t l = 0; l < 3; ++l)
99 if (!disjoint(here[k], previous[l])) return 0;
100 }
101 }
102 return 1;
103}

110static void bind_visibility(uint8_t *arena, size_t prefix, size_t capacity, size_t rows, int causal,
111 size_t positions_offset, size_t ranges_offset)
112{
113 int32_t *positions = (int32_t *)(arena + positions_offset);
114 int32_t *ranges = (int32_t *)(arena + ranges_offset);
115 for (size_t i = 0; i < rows; ++i) {
116 positions[i] = (int32_t)(prefix + i);
117 ranges[4*i] = 0;
118 ranges[4*i+1] = (int32_t)prefix;
119 ranges[4*i+2] = (int32_t)capacity;
120 ranges[4*i+3] = (int32_t)(capacity + (causal ? i + 1 : rows));
121 }
122}

124static int encode_clean(ck_diffusion_session *s, const int32_t *ids, size_t rows, int reset,
125 ck_diffusion_cancel cancel, void *context)
126{
127 const ck_diffusion_clean_entry *entry = reset ? &s->prefill : &s->append;
128 const size_t prefix = reset ? 0 : s->valid_rows;
129 span tokens;
130 if (rows != (reset ? s->prompt_rows : s->block_rows) ||
131 rows > s->capacity - prefix || (!reset && !prefix) ||
132 !region(&tokens, ids, rows, sizeof(*ids)) || !external(s, tokens))
133 return -1;
134 for (size_t i = 0; i < rows; ++i)
135 if (ids[i] < 0 || (size_t)ids[i] >= s->vocabulary) return -1;
136 if (cancelled(cancel, context)) return -7;
137 memcpy(s->arena + entry->token_ids, ids, rows * sizeof(*ids));
138 bind_visibility(s->arena, prefix, reset ? 0 : s->capacity, rows, 1, entry->positions, entry->ranges);
139 const int status = entry->entry(s->arena, s->arena_bytes);
140 if (status) return status;
141 if (cancelled(cancel, context)) return -7;
142
143 const size_t stride = s->kv_width * sizeof(uint16_t);
144 for (size_t i = 0; i < s->plane_count; ++i) {
145 const ck_diffusion_cache_plane *plane = &s->planes[i];
146 uint8_t *destination = s->arena + plane->cache;
147 const size_t source = reset ? plane->prefill_rows : plane->append_rows;
148 memcpy(destination + prefix * stride, s->arena + source, rows * stride);
149 if (reset)
150 memset(destination + rows * stride, 0, (s->capacity - rows) * stride);
151 }
152 s->valid_rows = prefix + rows;
153 return 0;
154}

170int ck_generate_diffusion_block(ck_diffusion_session *s, const ck_diffusion_block_request *r)
171{
172 if (!r || (uintptr_t)r % _Alignof(ck_diffusion_block_request) || !valid_session(s) ||
173 !s->valid_rows || s->block_rows > s->capacity - s->valid_rows ||
174 !isfinite(r->guidance) || r->latent_elements != s->latent_elements ||
175 r->staged_capacity < s->block_rows || r->output_capacity < s->block_rows)
176 return -1;
177 span operands[6];
178 if (!region(&operands[0], r, sizeof(*r), 1) ||
179 !region(&operands[1], r->output_ids, r->output_capacity, sizeof(int32_t)) ||
180 !region(&operands[2], r->staged_ids, r->staged_capacity, sizeof(int32_t)) ||
181 !region(&operands[3], r->latent, r->latent_elements, sizeof(double)) ||
182 !region(&operands[4], r->schedule, r->schedule_count, sizeof(double)) ||
183 !region(&operands[5], r->noise, r->noise_elements, sizeof(double)))
184 return -1;
185 for (size_t i = 0; i < sizeof operands / sizeof operands[0]; ++i) {
186 if (!external(s, operands[i])) return -1;
187 for (size_t j = 0; j < i; ++j)
188 if (!disjoint(operands[i], operands[j])) return -1;
189 }
190 if (cancelled(r->should_cancel, r->cancel_context)) return -7;
191
192 bind_visibility(s->arena, s->valid_rows, s->capacity, s->block_rows, 0, s->positions, s->conditional_ranges);
193 int32_t *ranges = (int32_t *)(s->arena + s->unconditional_ranges);
194 for (size_t i = 0; i < s->block_rows; ++i) {
195 const int32_t end = (int32_t)s->block_rows;
196 const int32_t row[4] = {0, end, end, end};
197 memcpy(ranges + 4*i, row, sizeof(row));
198 }
199 memcpy(s->arena + s->guidance, &r->guidance, sizeof(r->guidance));
200 const ck_diffusion_schedule_request step = {
201 .step = s->step, .arena = s->arena, .arena_bytes = s->arena_bytes, .bindings = s->step_bindings,
202 .latent_elements = s->latent_elements, .token_count = s->block_rows, .vocabulary = s->vocabulary,
203 .initial_latent = r->latent, .schedule = r->schedule, .schedule_count = r->schedule_count,
204 .noise = r->noise, .noise_elements = r->noise_elements, .output_ids = r->staged_ids,
205 .output_capacity = r->staged_capacity, .should_cancel = r->should_cancel, .cancel_context = r->cancel_context,
206 };
207 int status = ck_run_checked_diffusion_schedule(&step);
208 if (status) return status;
209 status = encode_clean(s, r->staged_ids, s->block_rows, 0, r->should_cancel, r->cancel_context);
210 if (status) return status;
211 memcpy(r->output_ids, r->staged_ids, s->block_rows * sizeof(int32_t));
212 return 0;
213}
