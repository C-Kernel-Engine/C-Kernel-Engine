24void deepseek_mhc_mix_f32(const float *streams,
25 const float *mix,
26 float *out,
27 int tokens,
28 int n_streams,
29 int dim)
30{
31 if (!streams || !mix || !out || tokens <= 0 || n_streams <= 0 || dim <= 0) return;
32
33 for (int t = 0; t < tokens; ++t) {
34 for (int os = 0; os < n_streams; ++os) {
35 for (int d = 0; d < dim; ++d) {
36 float acc = 0.0f;
37 for (int is = 0; is < n_streams; ++is) {
38 acc += mix[ds_mix_idx(t, os, is, n_streams)] *
39 streams[ds_mhc_idx(t, is, d, n_streams, dim)];
40 }
41 out[ds_mhc_idx(t, os, d, n_streams, dim)] = acc;
42 }
43 }
44 }
45}

47void deepseek_mhc_mix_backward_f32(const float *d_out,
48 const float *streams,
49 const float *mix,
50 float *d_streams,
51 float *d_mix,
52 int tokens,
53 int n_streams,
54 int dim)
55{
56 if (!d_out || !streams || !mix || !d_streams || !d_mix ||
57 tokens <= 0 || n_streams <= 0 || dim <= 0) return;
58
59 const size_t stream_count = (size_t)tokens * (size_t)n_streams * (size_t)dim;
60 const size_t mix_count = (size_t)tokens * (size_t)n_streams * (size_t)n_streams;
61 for (size_t i = 0; i < stream_count; ++i) d_streams[i] = 0.0f;
62 for (size_t i = 0; i < mix_count; ++i) d_mix[i] = 0.0f;
63
64 for (int t = 0; t < tokens; ++t) {
65 for (int os = 0; os < n_streams; ++os) {
66 for (int is = 0; is < n_streams; ++is) {
67 float d_mix_acc = 0.0f;
68 const float m = mix[ds_mix_idx(t, os, is, n_streams)];
69 for (int d = 0; d < dim; ++d) {
70 const float go = d_out[ds_mhc_idx(t, os, d, n_streams, dim)];
71 d_streams[ds_mhc_idx(t, is, d, n_streams, dim)] += m * go;
72 d_mix_acc += go * streams[ds_mhc_idx(t, is, d, n_streams, dim)];
73 }
74 d_mix[ds_mix_idx(t, os, is, n_streams)] = d_mix_acc;
75 }
76 }
77 }

80void deepseek_dsa_topk_softmax_f32(const float *scores,
81 int *indices,
82 float *weights,
83 int tokens,
84 int heads,
85 int key_count,
86 int top_k)
87{
88 if (!scores || !indices || !weights || tokens <= 0 || heads <= 0 ||
89 key_count <= 0 || top_k <= 0) return;
90
91 if (top_k > key_count) top_k = key_count;
92
93 for (int t = 0; t < tokens; ++t) {
94 for (int h = 0; h < heads; ++h) {
95 const float *row = scores + ((size_t)t * (size_t)heads + (size_t)h) * (size_t)key_count;
96 int *idx = indices + ((size_t)t * (size_t)heads + (size_t)h) * (size_t)top_k;
97 float *w = weights + ((size_t)t * (size_t)heads + (size_t)h) * (size_t)top_k;
98
99 for (int k = 0; k < top_k; ++k) {
100 idx[k] = -1;
101 w[k] = -FLT_MAX;
102 }
103
104 for (int j = 0; j < key_count; ++j) {
105 const float v = row[j];
106 int pos = -1;
107 for (int k = 0; k < top_k; ++k) {
108 if (idx[k] < 0 || v > w[k] || (v == w[k] && j < idx[k])) {
109 pos = k;
110 break;
111 }
112 }
113 if (pos >= 0) {
114 for (int k = top_k - 1; k > pos; --k) {
115 idx[k] = idx[k - 1];
116 w[k] = w[k - 1];
117 }
118 idx[pos] = j;
119 w[pos] = v;
120 }
121 }
122
123 float max_v = w[0];
124 for (int k = 1; k < top_k; ++k) if (w[k] > max_v) max_v = w[k];
125 float sum = 0.0f;
126 for (int k = 0; k < top_k; ++k) {
127 w[k] = expf(w[k] - max_v);
128 sum += w[k];
129 }
130 if (sum > 0.0f) {
131 const float inv = 1.0f / sum;
132 for (int k = 0; k < top_k; ++k) w[k] *= inv;
133 }
134 }
135 }
136}

173static void ds_softmax(float *x, int n)
174{
175 if (n <= 0) return;
176 float max_v = x[0];
177 for (int i = 1; i < n; ++i) if (x[i] > max_v) max_v = x[i];
178 float sum = 0.0f;
179 for (int i = 0; i < n; ++i) {
180 x[i] = expf(x[i] - max_v);
181 sum += x[i];
182 }
183 if (sum > 0.0f) {
184 const float inv = 1.0f / sum;
185 for (int i = 0; i < n; ++i) x[i] *= inv;
186 }
187}

189void deepseek_csa_attention_f32(const float *q,
190 const float *k,
191 const float *v,
192 const int *indices,
193 float *out,
194 float *attn,
195 int query_tokens,
196 int key_tokens,
197 int heads,
198 int dim,
199 int top_k,
200 float scale)
201{
202 if (!q || !k || !v || !indices || !out || query_tokens <= 0 || key_tokens <= 0 ||
203 heads <= 0 || dim <= 0 || top_k <= 0) return;
204
205 for (int tq = 0; tq < query_tokens; ++tq) {
206 for (int h = 0; h < heads; ++h) {
207 float local_scores[top_k];
208 int valid = 0;
209 for (int j = 0; j < top_k; ++j) {
210 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
211 if (tk < 0 || tk >= key_tokens) {
212 local_scores[j] = -FLT_MAX;
213 continue;
214 }
215 float dot = 0.0f;
216 for (int d = 0; d < dim; ++d) {
217 dot += q[ds_qkv_idx(tq, h, d, heads, dim)] * k[ds_qkv_idx(tk, h, d, heads, dim)];
218 }
219 local_scores[j] = dot * scale;
220 valid++;
221 }
222
223 float *out_row = out + ds_qkv_idx(tq, h, 0, heads, dim);
224 for (int d = 0; d < dim; ++d) out_row[d] = 0.0f;
225 if (valid == 0) {
226 if (attn) {
227 for (int j = 0; j < top_k; ++j) {
228 attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j] = 0.0f;
229 }
230 }
231 continue;
232 }
233
234 ds_softmax(local_scores, top_k);
235 for (int j = 0; j < top_k; ++j) {
236 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
237 const float a = (tk >= 0 && tk < key_tokens) ? local_scores[j] : 0.0f;
238 if (attn) attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j] = a;
239 if (a == 0.0f) continue;
240 for (int d = 0; d < dim; ++d) {
241 out_row[d] += a * v[ds_qkv_idx(tk, h, d, heads, dim)];
242 }
243 }
244 }
245 }
246}

248void deepseek_csa_attention_backward_f32(const float *d_out,
249 const float *q,
250 const float *k,
251 const float *v,
252 const int *indices,
253 const float *attn,
254 float *d_q,
255 float *d_k,
256 float *d_v,
257 int query_tokens,
258 int key_tokens,
259 int heads,
260 int dim,
261 int top_k,
262 float scale)
263{
264 if (!d_out || !q || !k || !v || !indices || !attn || !d_q || !d_k || !d_v ||
265 query_tokens <= 0 || key_tokens <= 0 || heads <= 0 || dim <= 0 || top_k <= 0) return;
266
267 const size_t q_count = (size_t)query_tokens * (size_t)heads * (size_t)dim;
268 const size_t kv_count = (size_t)key_tokens * (size_t)heads * (size_t)dim;
269 for (size_t i = 0; i < q_count; ++i) d_q[i] = 0.0f;
270 for (size_t i = 0; i < kv_count; ++i) {
271 d_k[i] = 0.0f;
272 d_v[i] = 0.0f;
273 }
274
275 for (int tq = 0; tq < query_tokens; ++tq) {
276 for (int h = 0; h < heads; ++h) {
277 float d_attn[top_k];
278 float attn_dot = 0.0f;
279 for (int j = 0; j < top_k; ++j) {
280 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
281 float da = 0.0f;
282 if (tk >= 0 && tk < key_tokens) {
283 const float a = attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
284 for (int d = 0; d < dim; ++d) {
285 const float go = d_out[ds_qkv_idx(tq, h, d, heads, dim)];
286 da += go * v[ds_qkv_idx(tk, h, d, heads, dim)];
287 d_v[ds_qkv_idx(tk, h, d, heads, dim)] += a * go;
288 }
289 attn_dot += a * da;
290 }
291 d_attn[j] = da;
292 }
293
294 for (int j = 0; j < top_k; ++j) {
295 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
296 if (tk < 0 || tk >= key_tokens) continue;
297 const float a = attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
298 const float d_score = a * (d_attn[j] - attn_dot);
299 for (int d = 0; d < dim; ++d) {
300 const float qv = q[ds_qkv_idx(tq, h, d, heads, dim)];
301 const float kv = k[ds_qkv_idx(tk, h, d, heads, dim)];
302 d_q[ds_qkv_idx(tq, h, d, heads, dim)] += scale * d_score * kv;
303 d_k[ds_qkv_idx(tk, h, d, heads, dim)] += scale * d_score * qv;
304 }
305 }
306 }
307 }
308}
