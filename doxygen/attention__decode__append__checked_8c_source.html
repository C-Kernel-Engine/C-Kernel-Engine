9static int ck_mul_size(size_t a, size_t b, size_t *result) {
10 if (a && b > SIZE_MAX / a) return 0;
11 *result = a * b;
12 return 1;
13}

15static int ck_region(const void *pointer, size_t elements,
16 uintptr_t *begin, uintptr_t *end) {
17 size_t bytes;
18 if (!pointer || !ck_mul_size(elements, sizeof(float), &bytes)) return 0;
19 *begin = (uintptr_t)pointer;
20 if (bytes > UINTPTR_MAX - *begin) return 0;
21 *end = *begin + bytes;
22 return 1;
23}

25static int ck_disjoint(const void *a, size_t a_elements,
26 const void *b, size_t b_elements) {
27 uintptr_t ab, ae, bb, be;
28 return ck_region(a, a_elements, &ab, &ae) &&
29 ck_region(b, b_elements, &bb, &be) &&
30 (ae <= bb || be <= ab);
31}

131 if (score > maximum) maximum = score;

135 denominator += exp((double)scores[head * cache_capacity + token] -
136 (double)maximum);

138 for (size_t d = 0; d < head_dim; ++d) {
139 double numerator = 0.0;

143 v_cache + kv_head * cache_head + token * head_dim;
144 double probability = exp(
145 (double)scores[head * cache_capacity + token] -
146 (double)maximum) / denominator;
147 numerator += probability * (double)value[d];
148 }
149 float result = (float)numerator;

151 staged_output[head * head_dim + d] = result;
152 }
153 }

157 memcpy(v_cache + head * cache_head + position * head_dim,
158 v_new + head * head_dim, head_dim * sizeof(float));
159 }

161 return 0;
162}
