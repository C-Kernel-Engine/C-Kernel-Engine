27void deepseek_mhc_mix_f32(const float *streams,
28 const float *mix,
29 float *out,
30 int tokens,
31 int n_streams,
32 int dim)
33{
34 if (!streams || !mix || !out || tokens <= 0 || n_streams <= 0 || dim <= 0) return;
35
36 for (int t = 0; t < tokens; ++t) {
37 for (int os = 0; os < n_streams; ++os) {
38 for (int d = 0; d < dim; ++d) {
39 float acc = 0.0f;
40 for (int is = 0; is < n_streams; ++is) {
41 acc += mix[ds_mix_idx(t, os, is, n_streams)] *
42 streams[ds_mhc_idx(t, is, d, n_streams, dim)];
43 }
44 out[ds_mhc_idx(t, os, d, n_streams, dim)] = acc;
45 }
46 }
47 }

50void deepseek_mhc_mix_backward_f32(const float *d_out,
51 const float *streams,
52 const float *mix,
53 float *d_streams,
54 float *d_mix,
55 int tokens,
56 int n_streams,
57 int dim)
58{
59 if (!d_out || !streams || !mix || !d_streams || !d_mix ||
60 tokens <= 0 || n_streams <= 0 || dim <= 0) return;
61
62 const size_t stream_count = (size_t)tokens * (size_t)n_streams * (size_t)dim;
63 const size_t mix_count = (size_t)tokens * (size_t)n_streams * (size_t)n_streams;
64 for (size_t i = 0; i < stream_count; ++i) d_streams[i] = 0.0f;
65 for (size_t i = 0; i < mix_count; ++i) d_mix[i] = 0.0f;
66
67 for (int t = 0; t < tokens; ++t) {
68 for (int os = 0; os < n_streams; ++os) {
69 for (int is = 0; is < n_streams; ++is) {
70 float d_mix_acc = 0.0f;
71 const float m = mix[ds_mix_idx(t, os, is, n_streams)];
72 for (int d = 0; d < dim; ++d) {
73 const float go = d_out[ds_mhc_idx(t, os, d, n_streams, dim)];
74 d_streams[ds_mhc_idx(t, is, d, n_streams, dim)] += m * go;
75 d_mix_acc += go * streams[ds_mhc_idx(t, is, d, n_streams, dim)];
76 }
77 d_mix[ds_mix_idx(t, os, is, n_streams)] = d_mix_acc;
78 }
79 }
80 }

83void deepseek_dsa_topk_softmax_f32(const float *scores,
84 int *indices,
85 float *weights,
86 int tokens,
87 int heads,
88 int key_count,
89 int top_k)
90{
91 if (!scores || !indices || !weights || tokens <= 0 || heads <= 0 ||
92 key_count <= 0 || top_k <= 0) return;
93
94 if (top_k > key_count) top_k = key_count;
95
96 for (int t = 0; t < tokens; ++t) {
97 for (int h = 0; h < heads; ++h) {
98 const float *row = scores + ((size_t)t * (size_t)heads + (size_t)h) * (size_t)key_count;
99 int *idx = indices + ((size_t)t * (size_t)heads + (size_t)h) * (size_t)top_k;
100 float *w = weights + ((size_t)t * (size_t)heads + (size_t)h) * (size_t)top_k;
101
102 for (int k = 0; k < top_k; ++k) {
103 idx[k] = -1;
104 w[k] = -FLT_MAX;
105 }
106
107 for (int j = 0; j < key_count; ++j) {
108 const float v = row[j];
109 int pos = -1;
110 for (int k = 0; k < top_k; ++k) {
111 if (idx[k] < 0 || v > w[k] || (v == w[k] && j < idx[k])) {
112 pos = k;
113 break;
114 }
115 }
116 if (pos >= 0) {
117 for (int k = top_k - 1; k > pos; --k) {
118 idx[k] = idx[k - 1];
119 w[k] = w[k - 1];
120 }
121 idx[pos] = j;
122 w[pos] = v;
123 }
124 }
125
126 float max_v = w[0];
127 for (int k = 1; k < top_k; ++k) if (w[k] > max_v) max_v = w[k];
128 float sum = 0.0f;
129 for (int k = 0; k < top_k; ++k) {
130 w[k] = expf(w[k] - max_v);
131 sum += w[k];
132 }
133 if (sum > 0.0f) {
134 const float inv = 1.0f / sum;
135 for (int k = 0; k < top_k; ++k) w[k] *= inv;
136 }
137 }
138 }
139}

198 for (int t = 0; t < tokens; ++t) {

210 const int out_col = h * out_per_head + qk_nope_dim + d;
211 float acc = 0.0f;
212 for (int r = 0; r < kv_lora_rank; ++r) {
213 acc += kv_b_proj[(size_t)out_col * (size_t)kv_lora_rank + (size_t)r] *

262static void ds_mla_apply_kimi_rope(const float *src,
263 float *dst,
264 const float *cos_row,
265 const float *sin_row,
266 int dim)
267{
268 const int half = dim / 2;
269 for (int i = 0; i < half; ++i) {
270 const float x_first = src[2 * i];
271 const float x_second = src[2 * i + 1];
272 const float c = cos_row[i];
273 const float s = sin_row[i];
274 dst[i] = x_first * c - x_second * s;
275 dst[half + i] = x_second * c + x_first * s;
276 }
277}

279void deepseek_mla_partial_rope_concat_f32(const float *q_nope,
280 const float *q_pe,
281 const float *k_nope,
282 const float *k_pe,
283 const float *cos,
284 const float *sin,
285 float *query,
286 float *key,
287 int tokens,
288 int heads,
289 int qk_nope_dim,
290 int qk_rope_dim)
291{
292 if (!q_nope || !q_pe || !k_nope || !k_pe || !cos || !sin || !query || !key ||
293 tokens <= 0 || heads <= 0 || qk_nope_dim <= 0 || qk_rope_dim <= 0 || (qk_rope_dim % 2) != 0) {
294 return;
295 }
296
297 const int q_head_dim = qk_nope_dim + qk_rope_dim;
298 for (int t = 0; t < tokens; ++t) {
299 const float *cos_row = cos + (size_t)t * (size_t)(qk_rope_dim / 2);
300 const float *sin_row = sin + (size_t)t * (size_t)(qk_rope_dim / 2);
301 for (int h = 0; h < heads; ++h) {
302 float *q_out = query + ds_mla_thd_idx(t, h, 0, heads, q_head_dim);
303 float *k_out = key + ds_mla_thd_idx(t, h, 0, heads, q_head_dim);
304 const float *qn = q_nope + ds_mla_thd_idx(t, h, 0, heads, qk_nope_dim);
305 const float *qp = q_pe + ds_mla_thd_idx(t, h, 0, heads, qk_rope_dim);
306 const float *kn = k_nope + ds_mla_thd_idx(t, h, 0, heads, qk_nope_dim);
307 const float *kp = k_pe + ds_mla_tok_idx(t, 0, qk_rope_dim);
308 for (int d = 0; d < qk_nope_dim; ++d) {
309 q_out[d] = qn[d];
310 k_out[d] = kn[d];
311 }
312 ds_mla_apply_kimi_rope(qp, q_out + qk_nope_dim, cos_row, sin_row, qk_rope_dim);
313 ds_mla_apply_kimi_rope(kp, k_out + qk_nope_dim, cos_row, sin_row, qk_rope_dim);
314 }
315 }

318void deepseek_mla_partial_rope_concat_packed_f32(const float *q_packed,
319 const float *k_nope,
320 const float *kv_a_packed,
321 const float *cos,
322 const float *sin,
323 float *query,
324 float *key,
325 int tokens,
326 int heads,
327 int kv_lora_rank,
328 int qk_nope_dim,
329 int qk_rope_dim)
330{
331 if (!q_packed || !k_nope || !kv_a_packed || !cos || !sin || !query || !key ||
332 tokens <= 0 || heads <= 0 || kv_lora_rank <= 0 || qk_nope_dim <= 0 ||
333 qk_rope_dim <= 0 || (qk_rope_dim % 2) != 0) {
334 return;
335 }
336
337 const int q_head_dim = qk_nope_dim + qk_rope_dim;
338 const int kv_a_dim = kv_lora_rank + qk_rope_dim;
339 const int half = qk_rope_dim / 2;
340 for (int t = 0; t < tokens; ++t) {
341 const float *cos_row = cos + (size_t)t * (size_t)half;
342 const float *sin_row = sin + (size_t)t * (size_t)half;
343 const float *kp = kv_a_packed + (size_t)t * (size_t)kv_a_dim + (size_t)kv_lora_rank;
344 for (int h = 0; h < heads; ++h) {
345 const float *q_in = q_packed + ds_mla_thd_idx(t, h, 0, heads, q_head_dim);
346 const float *kn = k_nope + ds_mla_thd_idx(t, h, 0, heads, qk_nope_dim);
347 float *q_out = query + ds_mla_thd_idx(t, h, 0, heads, q_head_dim);
348 float *k_out = key + ds_mla_thd_idx(t, h, 0, heads, q_head_dim);
349
350 for (int d = 0; d < qk_nope_dim; ++d) {
351 q_out[d] = q_in[d];
352 k_out[d] = kn[d];
353 }
354
355 const float *qp = q_in + qk_nope_dim;
356 float q_pe_tmp[256];
357 if (qk_rope_dim > (int)(sizeof(q_pe_tmp) / sizeof(q_pe_tmp[0]))) {
358 return;
359 }
360 for (int i = 0; i < qk_rope_dim; ++i) q_pe_tmp[i] = qp[i];
361 for (int i = 0; i < half; ++i) {
362 const float q_first = q_pe_tmp[2 * i];
363 const float q_second = q_pe_tmp[2 * i + 1];
364 const float k_first = kp[2 * i];
365 const float k_second = kp[2 * i + 1];
366 const float c = cos_row[i];
367 const float ss = sin_row[i];
368 q_out[qk_nope_dim + i] = q_first * c - q_second * ss;
369 q_out[qk_nope_dim + half + i] = q_second * c + q_first * ss;
370 k_out[qk_nope_dim + i] = k_first * c - k_second * ss;
371 k_out[qk_nope_dim + half + i] = k_second * c + k_first * ss;
372 }
373 }
374 }

382static void ds_softmax(float *x, int n)
383{
384 if (n <= 0) return;
385 float max_v = x[0];
386 for (int i = 1; i < n; ++i) if (x[i] > max_v) max_v = x[i];
387 float sum = 0.0f;
388 for (int i = 0; i < n; ++i) {
389 x[i] = expf(x[i] - max_v);
390 sum += x[i];
391 }
392 if (sum > 0.0f) {
393 const float inv = 1.0f / sum;
394 for (int i = 0; i < n; ++i) x[i] *= inv;
395 }
396}

398void deepseek_csa_attention_f32(const float *q,
399 const float *k,
400 const float *v,
401 const int *indices,
402 float *out,
403 float *attn,
404 int query_tokens,
405 int key_tokens,
406 int heads,
407 int dim,
408 int top_k,
409 float scale)
410{
411 if (!q || !k || !v || !indices || !out || query_tokens <= 0 || key_tokens <= 0 ||
412 heads <= 0 || dim <= 0 || top_k <= 0) return;
413
414 for (int tq = 0; tq < query_tokens; ++tq) {
415 for (int h = 0; h < heads; ++h) {
416 float local_scores[top_k];
417 int valid = 0;
418 for (int j = 0; j < top_k; ++j) {
419 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
420 if (tk < 0 || tk >= key_tokens) {
421 local_scores[j] = -FLT_MAX;
422 continue;
423 }
424 float dot = 0.0f;
425 for (int d = 0; d < dim; ++d) {
426 dot += q[ds_qkv_idx(tq, h, d, heads, dim)] * k[ds_qkv_idx(tk, h, d, heads, dim)];
427 }
428 local_scores[j] = dot * scale;
429 valid++;
430 }
431
432 float *out_row = out + ds_qkv_idx(tq, h, 0, heads, dim);
433 for (int d = 0; d < dim; ++d) out_row[d] = 0.0f;
434 if (valid == 0) {
435 if (attn) {
436 for (int j = 0; j < top_k; ++j) {
437 attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j] = 0.0f;
438 }
439 }
440 continue;
441 }
442
443 ds_softmax(local_scores, top_k);
444 for (int j = 0; j < top_k; ++j) {
445 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
446 const float a = (tk >= 0 && tk < key_tokens) ? local_scores[j] : 0.0f;
447 if (attn) attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j] = a;
448 if (a == 0.0f) continue;
449 for (int d = 0; d < dim; ++d) {
450 out_row[d] += a * v[ds_qkv_idx(tk, h, d, heads, dim)];
451 }
452 }
453 }
454 }
455}

457void deepseek_csa_attention_backward_f32(const float *d_out,
458 const float *q,
459 const float *k,
460 const float *v,
461 const int *indices,
462 const float *attn,
463 float *d_q,
464 float *d_k,
465 float *d_v,
466 int query_tokens,
467 int key_tokens,
468 int heads,
469 int dim,
470 int top_k,
471 float scale)
472{
473 if (!d_out || !q || !k || !v || !indices || !attn || !d_q || !d_k || !d_v ||
474 query_tokens <= 0 || key_tokens <= 0 || heads <= 0 || dim <= 0 || top_k <= 0) return;
475
476 const size_t q_count = (size_t)query_tokens * (size_t)heads * (size_t)dim;
477 const size_t kv_count = (size_t)key_tokens * (size_t)heads * (size_t)dim;
478 for (size_t i = 0; i < q_count; ++i) d_q[i] = 0.0f;
479 for (size_t i = 0; i < kv_count; ++i) {
480 d_k[i] = 0.0f;
481 d_v[i] = 0.0f;
482 }
483
484 for (int tq = 0; tq < query_tokens; ++tq) {
485 for (int h = 0; h < heads; ++h) {
486 float d_attn[top_k];
487 float attn_dot = 0.0f;
488 for (int j = 0; j < top_k; ++j) {
489 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
490 float da = 0.0f;
491 if (tk >= 0 && tk < key_tokens) {
492 const float a = attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
493 for (int d = 0; d < dim; ++d) {
494 const float go = d_out[ds_qkv_idx(tq, h, d, heads, dim)];
495 da += go * v[ds_qkv_idx(tk, h, d, heads, dim)];
496 d_v[ds_qkv_idx(tk, h, d, heads, dim)] += a * go;
497 }
498 attn_dot += a * da;
499 }
500 d_attn[j] = da;
501 }
502
503 for (int j = 0; j < top_k; ++j) {
504 const int tk = indices[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
505 if (tk < 0 || tk >= key_tokens) continue;
506 const float a = attn[((size_t)tq * (size_t)heads + (size_t)h) * (size_t)top_k + (size_t)j];
507 const float d_score = a * (d_attn[j] - attn_dot);
508 for (int d = 0; d < dim; ++d) {
509 const float qv = q[ds_qkv_idx(tq, h, d, heads, dim)];
510 const float kv = k[ds_qkv_idx(tk, h, d, heads, dim)];
511 d_q[ds_qkv_idx(tq, h, d, heads, dim)] += scale * d_score * kv;
512 d_k[ds_qkv_idx(tk, h, d, heads, dim)] += scale * d_score * qv;
513 }
514 }
515 }
516 }
517}

564 return;
565 }
566
567 float *scores = (float *)malloc((size_t)num_tokens * sizeof(float));
568 if (!scores) return;

571 for (int t = 0; t < num_tokens; ++t) {
572 for (int h = 0; h < num_heads; ++h) {

592 sum += e;
