14/* Each SIMD lane owns a key. Advance every lane in channel order. */
15static inline __attribute__((target("avx2"))) void scores_for_four_keys(
16 const float *query, const float *keys, size_t dim, double scores[4])
17{
18 __m256d sums = _mm256_setzero_pd();
19 size_t channel = 0;
20 for (; channel + 4 <= dim; channel += 4) {
21 __m128 k0 = _mm_loadu_ps(keys + channel);
22 __m128 k1 = _mm_loadu_ps(keys + dim + channel);
23 __m128 k2 = _mm_loadu_ps(keys + 2 * dim + channel);
24 __m128 k3 = _mm_loadu_ps(keys + 3 * dim + channel);
25 _MM_TRANSPOSE4_PS(k0, k1, k2, k3);
26 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel]), _mm256_cvtps_pd(k0)));
27 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel + 1]), _mm256_cvtps_pd(k1)));
28 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel + 2]), _mm256_cvtps_pd(k2)));
29 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel + 3]), _mm256_cvtps_pd(k3)));
30 }
31 for (; channel < dim; ++channel) {
32 const __m256d k = _mm256_setr_pd(keys[channel], keys[dim + channel],
33 keys[2 * dim + channel], keys[3 * dim + channel]);
34 sums = _mm256_add_pd(sums, _mm256_mul_pd(_mm256_set1_pd(query[channel]), k));
35 }
36 _mm256_storeu_pd(scores, sums);
37}

39static __attribute__((target("avx2"))) void attention_row_avx2(
40 const float *query, const float *key, const float *value, const float *relative, float sink,
41 float *output, double *accumulator, size_t dim, size_t begin, size_t position,
42 size_t relative_extent, double scale, size_t ring_capacity)
43{
44 double max_logit = sink, denom = 1.0;
45 for (size_t d = 0; d < dim; ++d) accumulator[d] = 0.0;
46 for (size_t first = begin; first <= position;) {
47 const size_t physical = ring_capacity ? first % ring_capacity : first;
48 size_t available = position - first + 1;
49 if (ring_capacity && available > ring_capacity - physical)
50 available = ring_capacity - physical;
51 double dots[4];
52 if (available >= 4) {
53 scores_for_four_keys(query, key + physical * dim, dim, dots);
54 } else {
55 for (size_t lane = 0; lane < available; ++lane) {
56 const float *keys = key + (physical + lane) * dim;
57 double dot = 0.0;
58 for (size_t d = 0; d < dim; ++d) dot += (double)query[d] * keys[d];
59 dots[lane] = dot;
60 }
61 }
62 const size_t count = available < 4 ? available : 4;
63 for (size_t lane = 0; lane < count; ++lane) {
64 const size_t distance = position - first - lane;
65 const double score = dots[lane] * scale +
66 (distance < relative_extent ? relative[distance] : 0.0);
67 const double next_max = fmax(score, max_logit);
68 const double old_factor = max_logit == next_max ? 1.0 : exp(max_logit - next_max);
69 const double new_factor = score == next_max ? 1.0 : exp(score - next_max);
70 denom = denom * old_factor + new_factor;
71 const float *values = value + (physical + lane) * dim;
72 const __m256d old_vector = _mm256_set1_pd(old_factor), new_vector = _mm256_set1_pd(new_factor);
73 size_t d = 0;
74 for (; d + 4 <= dim; d += 4) {
75 const __m256d previous = _mm256_loadu_pd(accumulator + d);
76 const __m256d current = _mm256_cvtps_pd(_mm_loadu_ps(values + d));
77 _mm256_storeu_pd(accumulator + d, _mm256_add_pd(
78 _mm256_mul_pd(previous, old_vector), _mm256_mul_pd(current, new_vector)));
79 }
80 for (; d < dim; ++d)
81 accumulator[d] = accumulator[d] * old_factor + (double)values[d] * new_factor;
82 max_logit = next_max;
83 }
84 first += count;
85 }
86 for (size_t d = 0; d < dim; ++d) output[d] = (float)(accumulator[d] / denom);
87}

97/* Read the causal span in absolute order, including across a ring wrap. */
98static int gqa_relative_sink_checked(GQA_RELATIVE_ARGS, int circular)
99{
100 if (!heads || heads > 256 || !kv_heads || heads % kv_heads ||
101 !dim || dim > 4096 || !capacity || capacity > SIZE_MAX / kv_heads ||
102 !tokens || tokens > capacity || position > SIZE_MAX - tokens ||
103 (!circular && position > capacity - tokens) || (circular && !window) ||
104 !relative_extent || relative_extent > 16384 || window > 16384)
105 return -2;
106
107 size_t rows, q_size, kv_size, rel_size;
108 if (!ck_checked_span(tokens, heads, heads, &rows) ||
109 !ck_checked_span(rows, dim, dim, &q_size) ||
110 !ck_checked_span(kv_heads * capacity, dim, dim, &kv_size) ||
111 !ck_checked_span(rows, relative_extent, relative_extent, &rel_size) ||
112 q_size > query_elements || q_size > output_elements ||
113 kv_size > key_elements || kv_size > value_elements ||
114 rel_size > relative_elements || rows > sink_elements || dim > work_elements)
115 return -2;
116
117 size_t workspace = dim;
118#ifdef _OPENMP
119 if (q_size <= SIZE_MAX / sizeof(double) && work_elements >= q_size)
120 workspace = q_size;
121#endif
122 const size_t scratch_bytes = workspace * sizeof(double);
123 if (!work || (uintptr_t)work % _Alignof(double) ||
124 (uintptr_t)work > UINTPTR_MAX - scratch_bytes ||
125 !ck_checked_region(output, q_size)) return -1;
126 const float *inputs[] = {query, key, value, relative, sink, output};
127 const size_t counts[] = {q_size, kv_size, kv_size, rel_size, rows, q_size};
128 for (size_t i = 0; i < 6; ++i) {
129 if (!ck_checked_region(inputs[i], counts[i])) return -1;
130 const uintptr_t a = (uintptr_t)inputs[i], b = (uintptr_t)work;
131 if ((a < b + scratch_bytes && b < a + counts[i] * sizeof(float)) ||
132 (i != 5 && ck_checked_overlap(output, q_size, inputs[i], counts[i])))
133 return -1;
134 }
135
136 if (!ck_checked_finite(query, q_size) || !ck_checked_finite(relative, rel_size) ||
137 !ck_checked_finite(sink, rows)) return -3;
138 const size_t first = window && position >= window ? position - window + 1 : 0;
139 const size_t visible = position + tokens - first;
140 /* The writer commits the entire chunk first, so every query's visible key
141 * must fit at once. Otherwise the newest rows can evict an earlier query. */
142 if (circular && visible > capacity) return -2;
143 const size_t physical = circular ? first % capacity : first;
144 const size_t contiguous = visible < capacity - physical ? visible : capacity - physical;
145 for (size_t h = 0; h < kv_heads; ++h) {
146 const size_t offset = (h * capacity + physical) * dim;
147 const size_t prefix = contiguous * dim;
148 const size_t suffix = (visible - contiguous) * dim;
149 const size_t head_start = h * capacity * dim;
150 if (!ck_checked_finite(key + offset, prefix) || !ck_checked_finite(value + offset, prefix) ||
151 (suffix && (!ck_checked_finite(key + head_start, suffix) ||
152 !ck_checked_finite(value + head_start, suffix))))
153 return -3;
154 }
155
156 const double scale = 1.0 / sqrt((double)dim);
157 const size_t group = heads / kv_heads;
158#ifdef CK_GQA_X86_AVX2
159 const int use_avx2 = dim >= 16 && __builtin_cpu_supports("avx2");
160#endif
161#ifdef _OPENMP
162 const int parallel = workspace > dim && visible >= 128;
163#pragma omp parallel for if(parallel)
164#endif
165 for (size_t row = 0; row < rows; ++row) {
166 const size_t qpos = position + row / heads;
167 const size_t begin = window && qpos >= window ? qpos - window + 1 : 0;
168 const float *q = query + row * dim;
169 const float *rel = relative + row * relative_extent;
170 const size_t cache = ((row % heads) / group) * capacity;
171 double *accumulator = work + (workspace > dim ? row * dim : 0);
172#ifdef CK_GQA_X86_AVX2
173 if (use_avx2 && qpos - begin >= 7) {
174 attention_row_avx2(q, key + cache * dim, value + cache * dim, rel, sink[row],
175 output + row * dim, accumulator, dim, begin, qpos, relative_extent, scale, circular ? capacity : 0);
176 continue;
177 }
178#endif
179 double max_logit = sink[row], denom = 1.0;
180 for (size_t d = 0; d < dim; ++d) accumulator[d] = 0.0;
181 for (size_t k = begin; k <= qpos; ++k) {
182 const size_t row_index = circular ? k % capacity : k;
183 const float *keys = key + (cache + row_index) * dim;
184 const float *values = value + (cache + row_index) * dim;
185 double dot = 0.0;
186 /* Each product of finite FP32 operands is exact in FP64. */
187 for (size_t d = 0; d < dim; ++d) dot += (double)q[d] * (double)keys[d];
188 const size_t distance = qpos - k;
189 const double score = dot * scale + (distance < relative_extent ? rel[distance] : 0.0);
190 const double next_max = fmax(score, max_logit);
191 const double old_factor = max_logit == next_max ? 1.0 : exp(max_logit - next_max);
192 const double new_factor = score == next_max ? 1.0 : exp(score - next_max);
193 denom = denom * old_factor + new_factor;
194 for (size_t d = 0; d < dim; ++d)
195 accumulator[d] = accumulator[d] * old_factor + (double)values[d] * new_factor;
196 max_logit = next_max;
197 }
198 for (size_t d = 0; d < dim; ++d)
199 output[row * dim + d] = (float)(accumulator[d] / denom);
200 }
201 return 0;
202}
