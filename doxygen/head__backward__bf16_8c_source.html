7typedef struct { const void *ptr; size_t elements, unit; } Region;
8static int product(size_t a,size_t b,size_t *v) {
9 if (!a || !b || a>SIZE_MAX/b) return 0;
10 *v=a*b; return 1;
11}
12static int sum_size(size_t a,size_t b,size_t *v) {
13 if (a>SIZE_MAX-b) return 0;
14 *v=a+b; return 1;
15}
16static int extent(size_t rows,size_t cols,size_t stride,size_t size) {
17 return rows && cols && stride>=cols && rows-1<=(SIZE_MAX-cols)/stride &&
18 (rows-1)*stride+cols<=size;
19}
20static int regions(const Region *r,size_t count,size_t reads) {
21 for (size_t i=0;i<count;++i) {
22 if (!r[i].ptr || !r[i].elements || (uintptr_t)r[i].ptr%r[i].unit ||
23 r[i].elements>SIZE_MAX/r[i].unit ||
24 (uintptr_t)r[i].ptr>UINTPTR_MAX-r[i].elements*r[i].unit) return 0;
25 }
26 for (size_t i=reads;i<count;++i) for (size_t j=0;j<i;++j)
27 if ((uintptr_t)r[i].ptr<(uintptr_t)r[j].ptr+r[j].elements*r[j].unit &&
28 (uintptr_t)r[j].ptr<(uintptr_t)r[i].ptr+r[i].elements*r[i].unit) return 0;
29 return 1;
30}
31static int expand(const uint16_t *src,float *dst,size_t rows,size_t cols,size_t stride) {
32 for (size_t r=0;r<rows;++r) for (size_t c=0;c<cols;++c) {
33 float x=bf16_to_float(src[r*stride+c]);
34 if (!isfinite(x)) return 0;
35 dst[r*cols+c]=x;
36 }
37 return 1;
38}
39static int roundable(const float *src,size_t n) {
40 for (size_t i=0;i<n;++i)
41 if (!isfinite(src[i]) || (float_to_bf16(src[i])&0x7f80u)==0x7f80u) return 0;
42 return 1;
43}
44static void publish(uint16_t *dst,const float *src,size_t rows,size_t cols,size_t stride) {
45 for (size_t r=0;r<rows;++r) for (size_t c=0;c<cols;++c)
46 dst[r*stride+c]=float_to_bf16(src[r*cols+c]);
47}
48int ck_token_span_mean_backward_bf16(
49 const uint16_t *d_output,size_t d_output_elements,size_t d_output_stride,
50 const int32_t *spans,size_t span_elements,size_t span_count,
51 uint16_t *d_input,size_t d_input_elements,size_t d_input_stride,
52 float *scratch,size_t scratch_elements,
53 size_t token_capacity,size_t valid_tokens,size_t channels) {
54 size_t sc,tc,vc,needed;
55 if (!valid_tokens || valid_tokens>token_capacity || span_count>SIZE_MAX/2 ||
56 !product(span_count,channels,&sc) || !product(token_capacity,channels,&tc) ||
57 !product(valid_tokens,channels,&vc) || !sum_size(sc,tc,&needed) ||
58 !sum_size(needed,vc,&needed) || scratch_elements<needed || span_elements<2*span_count ||
59 !extent(span_count,channels,d_output_stride,d_output_elements) ||
60 !extent(token_capacity,channels,d_input_stride,d_input_elements)) return -2;
61 Region r[]={{d_output,d_output_elements,2},{spans,span_elements,4},
62 {d_input,d_input_elements,2},{scratch,scratch_elements,4}};
63 if (!regions(r,4,2)) return -1;
64 if (!expand(d_output,scratch,span_count,channels,d_output_stride)) return -1;
65 float *result=scratch+sc,*work=result+tc;
66 int rc=ck_token_span_mean_backward_f32(scratch,sc,channels,spans,span_elements,span_count,
67 result,tc,channels,work,vc,token_capacity,valid_tokens,channels);
68 if (rc) return rc;
69 if (!roundable(result,vc)) return -1;
70 publish(d_input,result,valid_tokens,channels,d_input_stride); return 0;
71}

115 {group_counts,count_elements,4},{d_output,d_output_elements,2},

123 int rc=ck_grouped_weighted_rows_backward_f32(x,oc,channels,f,gc,channels,
124 group_counts,count_elements,dy,gc,channels,dx,oc,channels,df,gc,channels,
125 work,inner,groups,option_rows,channels);
126 if (rc) return rc;

128 publish(d_options,dx,option_rows,channels,d_options_stride);
