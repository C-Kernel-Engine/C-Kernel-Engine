34/* One variance-preserving DDPM step with externally supplied Gaussian noise.
35 * All spans have count elements. Identical output/input spans are forbidden. */
36int ck_diffusion_vp_ddpm_step_f64(const double *latent, const double *estimate,
37 const double *noise, double *output, size_t count,
38 double gamma_from, double gamma_to)
39{
40 if (!count || !isfinite(gamma_from) || !isfinite(gamma_to) ||
41 gamma_from < gamma_to || !vp_region_valid(latent, count) ||
42 !vp_region_valid(estimate, count) || !vp_region_valid(noise, count) ||
43 !vp_region_valid(output, count)) return -1;
44 const size_t bytes = count * sizeof(double);
45 if (vp_overlaps(output, latent, bytes) || vp_overlaps(output, estimate, bytes) ||
46 vp_overlaps(output, noise, bytes)) return -1;
47
48 const double from = vp_sigmoid(-gamma_from);
49 const double to = vp_sigmoid(-gamma_to);
50 if (from <= 0.0 || to <= 0.0) return -1;
51 const double blend = -expm1(gamma_to - gamma_from);
52 const double residual_gain = (1.0 - blend) * sqrt(to / from);
53 const double estimate_gain = blend * sqrt(to);
54 const double noise_gain = sqrt(blend * (1.0 - to));
55 if (!isfinite(residual_gain) || !isfinite(estimate_gain) || !isfinite(noise_gain)) return -1;
56
57 for (size_t i = 0; i < count; ++i) {
58 if (!isfinite(latent[i]) || !isfinite(estimate[i]) || !isfinite(noise[i]) ||
59 !isfinite(vp_blend(latent[i], estimate[i], noise[i],
60 residual_gain, estimate_gain, noise_gain))) return -1;
61 }
62 for (size_t i = 0; i < count; ++i) {
63 output[i] = vp_blend(latent[i], estimate[i], noise[i],
64 residual_gain, estimate_gain, noise_gain);
65 }
66 return 0;
67}
