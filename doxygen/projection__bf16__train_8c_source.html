8static int product(size_t a, size_t b, size_t *n) {
9 if (!a || !b || a > SIZE_MAX/b) return 0;
10 *n=a*b; return 1;
11}
12static int regions(const void **p, const size_t *n, const size_t *width, size_t count, size_t reads) {
13 for (size_t i=0;i<count;i++) {
14 if (!p[i] || !n[i] || (uintptr_t)p[i]%width[i] || n[i]>SIZE_MAX/width[i] ||
15 (uintptr_t)p[i]>UINTPTR_MAX-n[i]*width[i]) return 0;
16 if (i>=reads) for (size_t j=0;j<i;j++)
17 if ((uintptr_t)p[i]<(uintptr_t)p[j]+n[j]*width[j] &&
18 (uintptr_t)p[j]<(uintptr_t)p[i]+n[i]*width[i]) return 0;
19 }
20 return 1;
21}
22static float rounded(float x) { return bf16_to_float(float_to_bf16(x)); }
23static float add_product(float sum, float a, float b) {
24 volatile float p=a*b; /* prohibit FMA, including under normal -O3 */
25 volatile float next=sum+p;
26 return next;
27}
28static int projection_forward(
29 const float *input, size_t input_elements, const float *weight, size_t weight_elements,
30 const float *bias, size_t bias_elements,
31 float *output, size_t output_elements, uint16_t *saved_input, size_t saved_input_elements,
32 uint16_t *saved_weight, size_t saved_weight_elements, float *scratch, size_t scratch_elements,
33 size_t rows, size_t inputs, size_t outputs) {
34 size_t x,w,y;
35 if (!product(rows,inputs,&x) || !product(outputs,inputs,&w) || !product(rows,outputs,&y) ||
36 input_elements<x || weight_elements<w || output_elements<y ||
37 saved_input_elements<x || saved_weight_elements<w || scratch_elements<y) return -2;
38 const void *p[]={input,weight,output,saved_input,saved_weight,scratch};
39 const size_t n[]={input_elements,weight_elements,output_elements,saved_input_elements,saved_weight_elements,scratch_elements};
40 const size_t widths[]={4,4,4,2,2,4};
41 if (!regions(p,n,widths,6,2)) return -1;
42 if (bias) {
43 if (bias_elements < outputs) return -2;
44 const void *bp[]={input,weight,bias,output,saved_input,saved_weight,scratch};
45 const size_t bn[]={input_elements,weight_elements,bias_elements,output_elements,saved_input_elements,saved_weight_elements,scratch_elements};
46 const size_t bw[]={4,4,4,4,2,2,4};
47 if (!regions(bp,bn,bw,7,3)) return -1;
48 for(size_t i=0;i<outputs;i++) if(!isfinite(bias[i])) return -1;
49 }
50 for(size_t i=0;i<x;i++) if(!isfinite(input[i]) || !isfinite(rounded(input[i]))) return -1;
51 for(size_t i=0;i<w;i++) if(!isfinite(weight[i]) || !isfinite(rounded(weight[i]))) return -1;
52 for(size_t r=0;r<rows;r++) for(size_t o=0;o<outputs;o++) {
53 float sum=0;
54 for(size_t k=0;k<inputs;k++) sum=add_product(sum,rounded(input[r*inputs+k]),rounded(weight[o*inputs+k]));
55 if(bias) { volatile float biased=sum+bias[o]; sum=biased; }
56 if(!isfinite(sum)) return -1;
57 scratch[r*outputs+o]=sum;
58 }
59 for(size_t i=0;i<x;i++) saved_input[i]=float_to_bf16(input[i]);
60 for(size_t i=0;i<w;i++) saved_weight[i]=float_to_bf16(weight[i]);
61 memcpy(output,scratch,y*sizeof(float)); return 0;
62}
63static int projection_backward(
64 const float *d_output, size_t d_output_elements,
65 const uint16_t *saved_input, size_t saved_input_elements,
66 const uint16_t *saved_weight, size_t saved_weight_elements,
67 float *d_input, size_t d_input_elements, float *d_weight, size_t d_weight_elements,
68 float *d_bias, size_t d_bias_elements,
69 float *scratch, size_t scratch_elements, size_t rows, size_t inputs, size_t outputs) {
70 size_t x,w,y;
71 if(!product(rows,inputs,&x) || !product(outputs,inputs,&w) || !product(rows,outputs,&y) ||
72 x>SIZE_MAX-w || d_output_elements<y || saved_input_elements<x || saved_weight_elements<w ||
73 d_input_elements<x || d_weight_elements<w || scratch_elements<x+w) return -2;
74 const void *p[]={d_output,saved_input,saved_weight,d_input,d_weight,scratch};
75 const size_t n[]={d_output_elements,saved_input_elements,saved_weight_elements,d_input_elements,d_weight_elements,scratch_elements};
76 const size_t widths[]={4,2,2,4,4,4};
77 if(!regions(p,n,widths,6,3)) return -1;
78 if (d_bias) {
79 if (d_bias_elements<outputs || x+w>SIZE_MAX-outputs || scratch_elements<x+w+outputs) return -2;
80 const void *bp[]={d_output,saved_input,saved_weight,d_input,d_weight,d_bias,scratch};
81 const size_t bn[]={d_output_elements,saved_input_elements,saved_weight_elements,d_input_elements,d_weight_elements,d_bias_elements,scratch_elements};
82 const size_t bw[]={4,2,2,4,4,4,4};
83 if(!regions(bp,bn,bw,7,3)) return -1;
84 }
85 for(size_t i=0;i<y;i++) if(!isfinite(d_output[i])) return -1;
86 for(size_t i=0;i<x;i++) if(!isfinite(bf16_to_float(saved_input[i]))) return -1;
87 for(size_t i=0;i<w;i++) if(!isfinite(bf16_to_float(saved_weight[i]))) return -1;
88 for(size_t r=0;r<rows;r++) for(size_t k=0;k<inputs;k++) {
89 float sum=0;
90 for(size_t o=0;o<outputs;o++) sum=add_product(sum,d_output[r*outputs+o],bf16_to_float(saved_weight[o*inputs+k]));
91 if(!isfinite(sum)) return -1;
92 scratch[r*inputs+k]=sum;
93 }
94 for(size_t o=0;o<outputs;o++) for(size_t k=0;k<inputs;k++) {
95 float sum=0;
96 for(size_t r=0;r<rows;r++) sum=add_product(sum,d_output[r*outputs+o],bf16_to_float(saved_input[r*inputs+k]));
97 if(!isfinite(sum)) return -1;
98 scratch[x+o*inputs+k]=sum;
99 }
100 if(d_bias) for(size_t o=0;o<outputs;o++) {
101 float sum=0;
102 for(size_t r=0;r<rows;r++) { volatile float next=sum+d_output[r*outputs+o]; sum=next; }
103 if(!isfinite(sum)) return -1;
104 scratch[x+w+o]=sum;
105 }
106 if(d_bias) memcpy(d_bias,scratch+x+w,outputs*sizeof(float));
107 memcpy(d_input,scratch,x*sizeof(float)); memcpy(d_weight,scratch+x,w*sizeof(float)); return 0;
108}
