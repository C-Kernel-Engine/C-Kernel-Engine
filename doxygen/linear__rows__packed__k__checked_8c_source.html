22static void packed_linear_outputs(size_t begin, size_t end, void *opaque)
23{
24 struct packed_linear_work *work = opaque;
25 size_t row = begin / work->channels;
26 size_t channel = begin % work->channels;
27 for (size_t i = begin; i < end; ++i) {
28 const unsigned char *packed = work->weights + channel * work->row_bytes;
29 const float value = work->dot(work->input + row * work->width, packed, work->width);
30 if (!isfinite(value)) {
31 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
32 return;
33 }
34 if (work->publish) work->output[i] = value;
35 if (++channel == work->channels) {
36 channel = 0;
37 ++row;
38 }
39 }
40}
