25static double attention_dot(const void *q, const void *k, size_t q_offset,
26 size_t k_offset, size_t width, int bf16)
27{
28 double sum = 0.0;
29 for (size_t i = 0; i < width; ++i) {
30 volatile double product = (double)read_element(q, q_offset + i, bf16) *
31 read_element(k, k_offset + i, bf16);
32 sum += product;
33 }
34 return sum;
35}

37/* Per-query visibility is two ordered, disjoint half-open key ranges.
38 * An empty second range means one interval. Output commits atomically. */
39static int attention_ranges_checked(
40 const void *query, size_t query_elements,
41 const void *key, size_t key_elements,
42 const void *value, size_t value_elements,
43 const int32_t *ranges, size_t range_elements,
44 void *output, size_t output_elements,
45 float *scratch, size_t scratch_elements,
46 size_t query_tokens, size_t memory_tokens,
47 size_t heads, size_t kv_heads, size_t head_dim, float scale, int bf16, int round_intermediates)
48{
49 if (!query_tokens || !memory_tokens || memory_tokens > INT32_MAX ||
50 !heads || !kv_heads || heads % kv_heads || !head_dim ||
51 heads > SIZE_MAX / head_dim || query_tokens > SIZE_MAX / 4 ||
52 !isfinite(scale) || scale <= 0.0f) return -2;
53 const size_t qw = heads * head_dim;
54 const size_t kw = kv_heads * head_dim;
55 if (query_tokens > SIZE_MAX / qw || memory_tokens > SIZE_MAX / kw) return -2;
56 const size_t q = query_tokens * qw;
57 const size_t k = memory_tokens * kw;
58 if (q > SIZE_MAX - memory_tokens || range_elements < 4 * query_tokens ||
59 query_elements < q || key_elements < k || value_elements < k ||
60 output_elements < q || scratch_elements < q + memory_tokens) return -2;
61
62 const void *inputs[] = {query, key, value, ranges};
63 const size_t counts[] = {query_elements, key_elements, value_elements, range_elements};
64 const size_t element_width = bf16 ? sizeof(uint16_t) : sizeof(float);
65 const size_t widths[] = {element_width, element_width, element_width, sizeof(int32_t)};
66 if (!region_valid(output, output_elements, element_width) ||
67 !region_valid(scratch, scratch_elements, sizeof(float))) return -1;
68 const size_t output_bytes = output_elements * element_width;
69 const size_t scratch_bytes = scratch_elements * sizeof(float);
70 if (regions_overlap(output, output_bytes, scratch, scratch_bytes)) return -1;
71 for (size_t i = 0; i < 4; ++i) {
72 if (!region_valid(inputs[i], counts[i], widths[i])) return -1;
73 const size_t bytes = counts[i] * widths[i];
74 if (regions_overlap(output, output_bytes, inputs[i], bytes) ||
75 regions_overlap(scratch, scratch_bytes, inputs[i], bytes)) return -1;
76 }
77 for (size_t t = 0; t < query_tokens; ++t) {
78 const int32_t *r = ranges + 4 * t;
79 if (r[0] < 0 || r[0] > r[1] || r[1] > r[2] ||
80 r[2] > r[3] || (size_t)r[3] > memory_tokens ||
81 (r[0] == r[1] && r[2] == r[3])) return -2;
82 }
83 for (size_t i = 0; i < q; ++i)
84 if (!isfinite(read_element(query, i, bf16))) return -1;
85 for (size_t i = 0; i < k; ++i)
86 if (!isfinite(read_element(key, i, bf16)) || !isfinite(read_element(value, i, bf16))) return -1;
87
88 float *scores = scratch;
89 float *staged = scratch + memory_tokens;
90 for (size_t t = 0; t < query_tokens; ++t) {
91 const int32_t *r = ranges + 4 * t;
92 for (size_t h = 0; h < heads; ++h) {
93 const size_t query_head = t * qw + h * head_dim;
94 const size_t head_offset = (h / (heads / kv_heads)) * head_dim;
95 float largest = -INFINITY;
96 for (size_t span = 0; span < 4; span += 2) {
97 for (size_t j = (size_t)r[span]; j < (size_t)r[span + 1]; ++j) {
98 double dot = attention_dot(query, key, query_head, j * kw + head_offset, head_dim, bf16);
99 if (round_intermediates) dot = bf16_to_float(float_to_bf16((float)dot));
100 const double s = dot * scale;
101 if (!isfinite(s) || fabs(s) > FLT_MAX) return -1;
102 scores[j] = round_intermediates ? bf16_to_float(float_to_bf16((float)s)) : (float)s;
103 if (!isfinite(scores[j])) return -1;
104 if (scores[j] > largest) largest = scores[j];
105 }
106 }
107 double denom = 0.0;
108 for (size_t span = 0; span < 4; span += 2) {
109 for (size_t j = (size_t)r[span]; j < (size_t)r[span + 1]; ++j) {
110 scores[j] = (float)exp((double)scores[j] - largest);
111 denom += scores[j];
112 }
113 }
114 if (round_intermediates) {
115 for (size_t span = 0; span < 4; span += 2) {
116 for (size_t j = (size_t)r[span]; j < (size_t)r[span + 1]; ++j) {
117 scores[j] = bf16_to_float(float_to_bf16((float)(scores[j] / denom)));
118 }
119 }
120 }
121 for (size_t d = 0; d < head_dim; ++d) {
122 double sum = 0.0;
123 for (size_t span = 0; span < 4; span += 2) {
124 for (size_t j = (size_t)r[span]; j < (size_t)r[span + 1]; ++j) {
125 volatile double product = (double)scores[j] *
126 read_element(value, j * kw + head_offset + d, bf16);
127 sum += product;
128 }
129 }
130 float v = (float)(round_intermediates ? sum : sum / denom);
131 if (bf16) v = bf16_to_float(float_to_bf16(v));
132 if (!isfinite(v)) return -1;
133 staged[t * qw + h * head_dim + d] = v;
134 }
135 }
136 }
137 if (bf16) {
138 for (size_t i = 0; i < q; ++i) ((uint16_t *)output)[i] = float_to_bf16(staged[i]);
139 } else {
140 for (size_t i = 0; i < q; ++i) ((float *)output)[i] = staged[i];
141 }
142 return 0;
143}
