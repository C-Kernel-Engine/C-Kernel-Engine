27static size_t packed_block_bytes(enum expert_storage storage)
28{
29 switch (storage) {
30 case EXPERT_Q8_0: return sizeof(block_q8_0);
31 case EXPERT_Q4_K: return sizeof(block_q4_K);
32 case EXPERT_Q5_0: return sizeof(block_q5_0);

48/* Packed blocks are read in GGUF byte order and expanded inside the dot.
49 * The FP32 activation and ordered FP64 accumulation policy is shared. */
50static float expert_dot(const float *input, const unsigned char *weight,
51 size_t width, enum expert_storage storage, const unsigned char *activation)
52{
53 if (activation) return ck_q8_activation_dot(weight, activation, width);
54 if (storage == EXPERT_Q8_0) return ck_packed_q8_dot_ordered(input, weight, width);
55 if (storage == EXPERT_Q4_K) return ck_packed_q4_k_dot_ordered(input, weight, width);
56 if (storage == EXPERT_Q5_0) return ck_packed_q5_0_dot_ordered(input, weight, width);
57 double sum = 0.0;
58 if (storage == EXPERT_F32) {
59 for (size_t i = 0; i < width; ++i) {
60 float value;
61 memcpy(&value, weight + i * sizeof(float), sizeof(float));
62 sum += (double)input[i] * (double)value;
63 }
64 } else {
65 const size_t block_bytes = packed_block_bytes(storage);
66 for (size_t i = 0; i < width; i += QK8_0, weight += block_bytes) {
67 const float scale = ck_fp16_to_fp32_soft(
68 (uint16_t)(weight[0] | ((uint16_t)weight[1] << 8)));
69 for (size_t j = 0; j < QK8_0; ++j) {
70 const unsigned pair = weight[2 + j % 16];
71 const float expanded = scale * (float)iq4_nl_values[j < 16 ? pair & 15 : pair >> 4];
72 sum += (double)input[i + j] * (double)expanded;
73 }
74 }
75 }
76 return (float)sum;
77}

156static unsigned char *expert_packed_activation(const struct routed_projection *p, float *scratch)
157{
158 unsigned char *after_activation = (unsigned char *)(scratch + 3 * p->inner);

164static int prepare_expert_activation(const struct routed_projection *p, float *scratch,
165 unsigned char *packed_activation)
166{
167 float *gate_values = scratch, *up_values = gate_values + p->inner, *activation = up_values + p->inner;
168 if (p->reference) {
169 for (size_t i = 0; i < p->inner; ++i) {
170 const float g = gate_values[i], u = up_values[i];
171 if (!isfinite(g) || !isfinite(u)) return -3;
172 const float capped_gate = p->gate_cap * tanhf(g * p->gate_reciprocal);
173 const float capped_up = p->up_cap * tanhf(u * p->up_reciprocal);
174 const float sigmoid = 1.0f / (1.0f + expf(-g));
175 activation[i] = (capped_gate * sigmoid) * capped_up;
176 if (!isfinite(activation[i])) return -3;
177 }
178 } else if (softcapped_swiglu_rows_checked_f32(gate_values, p->inner, up_values, p->inner,
179 activation, p->inner, p->inner, p->gate_cap, p->up_cap)) return -3;
180 return packed_activation ? ck_pack_q8_activation_rne(activation, packed_activation, p->inner) : 0;
181}

183static int finish_expert(const struct routed_projection *p, size_t slot, float *scratch, float *output)
184{
185 const unsigned char *down = p->down + slot * p->down_expert_bytes;
186 const float *activation = scratch + 2 * p->inner;
187 unsigned char *packed_activation = expert_packed_activation(p, scratch);
188 if (prepare_expert_activation(p, scratch, packed_activation)) return -3;
189 if (p->packed_q4 && p->packed_q4->down_q5 && p->down_storage == EXPERT_Q5_0) {
190 p->packed_q4->down_q5(output, down, packed_activation, (int)p->width, (int)p->inner);
191 if (!ck_checked_finite(output, p->width)) return -3;
192 } else {
193 for (size_t i = 0; i < p->width; ++i) {
194 output[i] = expert_dot(activation, down + i * p->down_row, p->inner, p->down_storage, packed_activation);
195 if (!isfinite(output[i])) return -3;
196 }
197 }
198 return 0;
199}

201static int project_expert(const struct routed_projection *p, const float *input,
202 const unsigned char *packed_hidden, size_t slot, float *scratch,
203 float *output)
204{
205 const size_t offset = slot * p->gate_expert_bytes;
206 const unsigned char *gate = p->gate + offset, *up = p->up + offset;
207 float *gate_values = scratch, *up_values = scratch + p->inner;
208 if (p->packed_q4) {
209 p->packed_q4->matvec(gate_values, gate, packed_hidden, (int)p->inner, (int)p->width);
210 p->packed_q4->matvec(up_values, up, packed_hidden, (int)p->inner, (int)p->width);
211 } else {
212 for (size_t i = 0; i < p->inner; ++i) {
213 if (packed_hidden) {
214 ck_q8_activation_dot_pair(gate + i * p->gate_row, up + i * p->gate_row, packed_hidden,
215 p->width, gate_values + i, up_values + i);
216 } else {
217 gate_values[i] = expert_dot(input, gate + i * p->gate_row, p->width, p->storage, NULL);
218 up_values[i] = expert_dot(input, up + i * p->gate_row, p->width, p->storage, NULL);
219 }
220 }
221 }
222 return finish_expert(p, slot, scratch, output);
223}

225static int project_expert_pair(const struct routed_projection *p, size_t slot,
226 const unsigned char *x0, const unsigned char *x1,
227 float *work0, float *work1, float *out0, float *out1)
228{
229 const size_t offset = slot * p->gate_expert_bytes;
230 p->packed_q4->matvec_pair(work0, work1, p->gate + offset, x0, x1, (int)p->inner, (int)p->width);
231 p->packed_q4->matvec_pair(work0 + p->inner, work1 + p->inner, p->up + offset,
232 x0, x1, (int)p->inner, (int)p->width);
233 if (p->packed_q4->down_q5_pair && p->down_storage == EXPERT_Q5_0) {
234 unsigned char *packed0 = expert_packed_activation(p, work0);
235 unsigned char *packed1 = expert_packed_activation(p, work1);
236 if (prepare_expert_activation(p, work0, packed0) || prepare_expert_activation(p, work1, packed1))
237 return -3;
238 p->packed_q4->down_q5_pair(out0, out1, p->down + slot * p->down_expert_bytes,
239 packed0, packed1, (int)p->width, (int)p->inner);
240 return ck_checked_finite(out0, p->width) && ck_checked_finite(out1, p->width) ? 0 : -3;
241 }
242 if (finish_expert(p, slot, work0, out0)) return -3;
243 return finish_expert(p, slot, work1, out1);
244}

246static inline int combine_expert(const float *expert, float *mixture, size_t width,
247 float weight, size_t index, int reference)
248{
249 for (size_t i = 0; i < width; ++i) {
250 if (reference) {
251 const volatile float weighted = expert[i] * weight;
252 mixture[i] = index == 0 ? weighted : mixture[i] + weighted;
253 } else {
254 mixture[i] += expert[i] * weight;
255 }
256 if (!isfinite(mixture[i])) return -3;
257 }
258 return 0;
259}

302static void project_routes(size_t begin, size_t end, void *data)
303{
304 struct routed_work *work = data;
305 const struct routed_projection *p = work->projection;
306 for (size_t route = begin; route < end; ++route) {
307 const unsigned char *packed_hidden = hidden_workspace(p, work->row_work);
308 if (project_expert(p, work->hidden, packed_hidden, work->slots[work->ids[route]],
309 work->row_work + route * work->row_scratch,
310 work->staged_experts + route * p->width)) {
311 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
312 return;
313 }
314 }

317static void project_rows(size_t begin, size_t end, void *data)
318{
319 struct routed_work *work = data;
320 const struct routed_projection *p = work->projection;
321 for (size_t row = begin; row < end; ++row) {
322 float *scratch = work->row_work + (work->scratch_rows > 1 ? row * work->row_scratch : 0);
323 unsigned char *packed_hidden = hidden_workspace(p, scratch);
324 const float *input = work->hidden + row * p->width;
325 if (pack_hidden(p, input, packed_hidden)) {
326 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
327 return;
328 }
329 size_t ordered_slots[1024];
330 if (!p->reference) order_slots(work->ids + row * work->top_k, work->experts, work->top_k, ordered_slots);
331 for (size_t index = 0; index < work->top_k; ++index) {
332 const size_t route = row * work->top_k + (p->reference ? index : ordered_slots[index]);
333 float *expert = work->staged_experts + route * p->width;
334 if (project_expert(p, input, packed_hidden, work->slots[work->ids[route]], scratch, expert) ||
335 combine_expert(expert, work->staged_output + row * p->width, p->width,
336 work->route_weights[route], index, p->reference)) {
337 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
338 return;
339 }
340 }
341 }

344/* Keep each expert's weights hot across routed rows. Route results remain in
345 * their original slots and are mixed only after all expert work completes. */
346static void project_grouped_experts(size_t begin, size_t end, void *opaque)
347{
348 struct routed_work *work = opaque;
349 const struct routed_projection *p = work->projection;
350 const size_t scratch_per_expert = work->row_scratch * (p->packed_q4->matvec_pair ? 2 : 1);
351 for (size_t id = begin; id < end; ++id) {
352 if (work->slots[id] < 0) continue;
353 float *scratch = work->row_work + id * scratch_per_expert;
354 size_t pending = SIZE_MAX;
355 for (size_t row = 0; row < work->rows; ++row) {
356 const unsigned char *packed = work->packed_rows + row * work->packed_row_bytes;
357 for (size_t i = 0; i < work->top_k; ++i) {
358 const size_t route = row * work->top_k + i;
359 if ((size_t)work->ids[route] != id) continue;
360 if (p->packed_q4->matvec_pair && pending == SIZE_MAX) {
361 pending = route;
362 break;
363 }
364 const int status = pending == SIZE_MAX
365 ? project_expert(p, work->hidden + row * p->width, packed, work->slots[id], scratch,
366 work->staged_experts + route * p->width)
367 : project_expert_pair(p, work->slots[id],
368 work->packed_rows + pending / work->top_k * work->packed_row_bytes, packed,
369 scratch, scratch + work->row_scratch,
370 work->staged_experts + pending * p->width, work->staged_experts + route * p->width);
371 if (status) {
372 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
373 return;
374 }
375 pending = SIZE_MAX;
376 break;
377 }
378 }
379 if (pending != SIZE_MAX) {
380 const size_t row = pending / work->top_k;
381 if (project_expert(p, work->hidden + row * p->width,
382 work->packed_rows + row * work->packed_row_bytes, work->slots[id], scratch,
383 work->staged_experts + pending * p->width)) {
384 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
385 return;
386 }
387 }
388 }
389}

391static size_t collect_expert_pair_jobs(const struct routed_work *work, struct expert_pair_job *jobs)
392{
393 size_t count = 0;
394 for (size_t id = 0; id < work->experts; ++id) {
395 uint32_t pending = UINT32_MAX;
396 for (size_t route = 0; route < work->rows * work->top_k; ++route) {
397 if ((size_t)work->ids[route] != id) continue;
398 if (pending == UINT32_MAX) {
399 pending = (uint32_t)route;
400 } else {
401 jobs[count++] = (struct expert_pair_job){pending, (uint32_t)route};
402 pending = UINT32_MAX;
403 }
404 }
405 if (pending != UINT32_MAX) jobs[count++] = (struct expert_pair_job){pending, UINT32_MAX};
406 }
407 return count;
408}

410static void project_expert_pair_jobs(size_t begin, size_t end, void *opaque)
411{
412 struct routed_work *work = opaque;
413 const struct routed_projection *p = work->projection;
414 const int worker = ck_threadpool_thread_id(work->pool);
415 const size_t scratch_slot = worker < 0 ? 0 : (size_t)worker;
416 float *scratch = work->row_work + scratch_slot * 2 * work->row_scratch;
417 for (size_t i = begin; i < end; ++i) {
418 const uint32_t first = work->jobs[i].first, second = work->jobs[i].second;
419 const size_t row = first / work->top_k;
420 const size_t slot = work->slots[work->ids[first]];
421 const unsigned char *packed = work->packed_rows + row * work->packed_row_bytes;
422 float *out = work->staged_experts + (size_t)first * p->width;
423 const int status = second == UINT32_MAX
424 ? project_expert(p, work->hidden + row * p->width, packed, slot, scratch, out)
425 : project_expert_pair(p, slot, packed,
426 work->packed_rows + second / work->top_k * work->packed_row_bytes,
427 scratch, scratch + work->row_scratch, out, work->staged_experts + (size_t)second * p->width);
428 if (status) {
429 atomic_store_explicit(&work->failed, 1, memory_order_relaxed);
430 return;
431 }
432 }

435static int routed_softcapped_experts(ROUTED_EXPERT_ARGS, size_t down_matrix_bytes,
436 enum expert_storage storage, enum expert_storage down_storage,
437 int quantize, int reference_moe, const struct q4_hidden_ops *packed_q4)
438{
439 /* Bounds keep workspace arithmetic valid on supported size_t widths. */
440 if (!rows || rows > CHECKED_ROUTED_EXPERT_MAX_ROWS || !hidden_dim || hidden_dim > 8192 ||
441 !inner || inner > 8192 || !experts || experts > 1024 || !slots ||
442 slots > experts || !top_k || top_k > experts ||
443 !isfinite(gate_cap) || gate_cap <= 0.0f || !isfinite(up_cap) || up_cap <= 0.0f)
444 return -1;
445 const size_t gate_block = packed_block_elements(storage);
446 const size_t down_block = packed_block_elements(down_storage);
447 if (hidden_dim % gate_block || inner % down_block) return -1;
448 const size_t gate_row = hidden_dim / gate_block * packed_block_bytes(storage);
449 const size_t down_row = inner / down_block * packed_block_bytes(down_storage);
450 const size_t gate_expert_bytes = inner * gate_row;
451 const size_t down_expert_bytes = hidden_dim * down_row;
452 const size_t route_count = rows * top_k;
453 const size_t output_elements = rows * hidden_dim;
454 const size_t row_outputs = route_count * hidden_dim;
455 if (packed_q4 && ((uintptr_t)gate % _Alignof(block_q4_K) ||
456 (uintptr_t)up % _Alignof(block_q4_K))) return -1;
457 if (packed_q4 && packed_q4->down_q5 && down_storage == EXPERT_Q5_0 &&
458 (uintptr_t)down % _Alignof(block_q5_0)) return -1;
459 size_t packed_bytes = packed_q4 ? hidden_dim / QK_K * sizeof(block_q8_K) :
460 quantize ? gate_row + down_row : 0;
461 if (packed_q4 && packed_q4->down_q5) packed_bytes += inner / QK8_0 * sizeof(block_q8_0);
462 const size_t packed_elements = (packed_bytes + sizeof(float) - 1) / sizeof(float);
463 const size_t row_scratch = 3 * inner + packed_elements;
464 const size_t staged_elements = row_outputs + output_elements;
465 const size_t scratch_rows = work_elements >= staged_elements + rows * row_scratch ? rows : 1;
466 const size_t route_scratch = rows == 1 && top_k > 1 &&
467 work_elements >= staged_elements + top_k * row_scratch ? top_k : 0;
468 const size_t packed_row_bytes = hidden_dim / QK_K * sizeof(block_q8_K);
469 const int grouped = packed_q4 && packed_q4->group_rows && rows > 1;
470 const size_t grouped_pack_elements = grouped ? rows * packed_row_bytes / sizeof(float) : 0;
471 const int dispatch_pairs = grouped && packed_q4->dispatch_pairs;
472 const size_t jobs_elements = dispatch_pairs ? 2 * route_count : 0;
473 const size_t scratch_count = dispatch_pairs ? 2 * CK_THREADPOOL_MAX_THREADS :
474 grouped ? experts * (packed_q4->matvec_pair ? 2 : 1) : route_scratch ? route_scratch : scratch_rows;
475 const size_t scratch_elements = staged_elements + grouped_pack_elements + jobs_elements + scratch_count * row_scratch;
476 if (hidden_elements < output_elements || id_elements < route_count || route_elements < route_count ||
477 expert_slot_elements < experts || expert_output_elements < row_outputs ||
478 output_capacity < output_elements || work_elements < scratch_elements ||
479 scratch_elements > SIZE_MAX / sizeof(float)) return -1;
480 if (gate_expert_bytes > SIZE_MAX / slots || down_expert_bytes > SIZE_MAX / slots ||
481 matrix_bytes != gate_expert_bytes * slots ||
482 down_matrix_bytes != down_expert_bytes * slots) return -1;
483 const void *buffers[] = {hidden, ids, route_weights, expert_slots, gate, up, down,
484 expert_output, output, work};
485 const size_t sizes[] = {output_elements * sizeof(float), route_count * sizeof(int32_t),
486 route_count * sizeof(float), experts * sizeof(int32_t),
487 matrix_bytes, matrix_bytes, down_matrix_bytes, row_outputs * sizeof(float),
488 output_elements * sizeof(float),
489 scratch_elements * sizeof(float)};
490 for (size_t i = 0; i < sizeof(buffers) / sizeof(buffers[0]); ++i) {
491 if (!region(buffers[i], sizes[i])) return -1;
492 if (i < 4 || i >= 7) {
493 if ((uintptr_t)buffers[i] % _Alignof(float)) return -1;
494 }
495 if (i < 7) continue;
496 for (size_t j = 0; j < i; ++j)
497 if (contains_overlap(buffers[i], sizes[i], buffers[j], sizes[j])) return -1;
498 }
499 if (!ck_checked_finite(hidden, output_elements) ||
500 !ck_checked_finite(route_weights, route_count)) return -2;
501
502 unsigned char assigned[1024] = {0};
503 for (size_t id = 0; id < experts; ++id) {
504 const int32_t slot = expert_slots[id];
505 if (slot == -1) continue;
506 if (slot < 0 || (size_t)slot >= slots || assigned[slot]) return -2;
507 assigned[slot] = 1;
508 }
509 for (size_t row = 0; row < rows; ++row) {
510 for (size_t k = 0; k < top_k; ++k) {
511 const size_t route = row * top_k + k;
512 const int32_t id = ids[route];
513 if (id < 0 || (size_t)id >= experts || expert_slots[id] < 0 ||
514 (size_t)expert_slots[id] >= slots || route_weights[route] < 0.0f) return -2;
515 for (size_t prev = 0; prev < k; ++prev)
516 if (id == ids[row * top_k + prev]) return -2;
517 }
518 }
519
520 const float gate_reciprocal = reference_moe ? 1.0f / gate_cap : 0.0f;
521 const float up_reciprocal = reference_moe ? 1.0f / up_cap : 0.0f;
522 if (!isfinite(gate_reciprocal) || !isfinite(up_reciprocal)) return -3;
523
524 float *staged_experts = work;
525 float *staged_output = staged_experts + row_outputs;
526 float *packed_rows = staged_output + output_elements;
527 float *jobs_buffer = packed_rows + grouped_pack_elements;
528 float *row_work = jobs_buffer + jobs_elements;
529 memset(staged_output, 0, output_elements * sizeof(float));
530 const struct routed_projection projection = {
531 .gate = gate, .up = up, .down = down, .width = hidden_dim, .inner = inner,
532 .gate_row = gate_row, .down_row = down_row,
533 .gate_expert_bytes = gate_expert_bytes, .down_expert_bytes = down_expert_bytes,
534 .gate_cap = gate_cap, .up_cap = up_cap, .gate_reciprocal = gate_reciprocal, .up_reciprocal = up_reciprocal,
535 .storage = storage, .down_storage = down_storage, .quantize = quantize, .reference = reference_moe, .packed_q4 = packed_q4,
536 };
537 struct routed_work task = {
538 .projection = &projection, .hidden = hidden, .route_weights = route_weights,
539 .ids = ids, .slots = expert_slots, .row_work = row_work,
540 .staged_experts = staged_experts, .staged_output = staged_output,
541 .packed_rows = (const unsigned char *)packed_rows, .packed_row_bytes = packed_row_bytes,
542 .row_scratch = row_scratch, .scratch_rows = scratch_rows, .rows = rows, .experts = experts, .top_k = top_k,
543 };
544 atomic_init(&task.failed, 0);
545 if (grouped) {
546 unsigned char *packs = (unsigned char *)packed_rows;
547 for (size_t row = 0; row < rows; ++row) {
548 if (pack_hidden(&projection, hidden + row * hidden_dim, packs + row * packed_row_bytes)) return -3;
549 }
550 if (dispatch_pairs) {
551 struct expert_pair_job *jobs = (struct expert_pair_job *)jobs_buffer;
552 const size_t count = collect_expert_pair_jobs(&task, jobs);
553 task.jobs = jobs;
554 task.pool = ck_threadpool_global();
555 ck_checked_parallel_for(count, 1, project_expert_pair_jobs, &task);
556 } else {
557 ck_checked_parallel_for(experts, 1, project_grouped_experts, &task);
558 }
559 if (atomic_load_explicit(&task.failed, memory_order_relaxed)) return -3;
560 for (size_t row = 0; row < rows; ++row) {
561 size_t ordered_slots[1024];
562 order_slots(ids + row * top_k, experts, top_k, ordered_slots);
563 for (size_t index = 0; index < top_k; ++index) {
564 const size_t route = row * top_k + ordered_slots[index];
565 if (combine_expert(staged_experts + route * hidden_dim, staged_output + row * hidden_dim,
566 hidden_dim, route_weights[route], index, 0)) return -3;
567 }
568 }
569 } else if (route_scratch) {
570 unsigned char *packed_hidden = hidden_workspace(&projection, row_work);
571 if (pack_hidden(&projection, hidden, packed_hidden)) return -3;
572 /* Each task writes its declared slot. Mixing preserves the FP32 order. */
573 ck_checked_parallel_for(top_k, 1, project_routes, &task);
574 if (atomic_load_explicit(&task.failed, memory_order_relaxed)) return -3;
575 size_t ordered_slots[1024];
576 if (!reference_moe) order_slots(ids, experts, top_k, ordered_slots);
577 for (size_t index = 0; index < top_k; ++index) {
578 const size_t route = reference_moe ? index : ordered_slots[index];
579 if (combine_expert(staged_experts + route * hidden_dim, staged_output, hidden_dim,
580 route_weights[route], index, reference_moe)) return -3;
581 }
582 } else {
583 ck_checked_parallel_for(rows, scratch_rows > 1 ? 1 : (int)rows, project_rows, &task);
584 if (atomic_load_explicit(&task.failed, memory_order_relaxed)) return -3;
585 }
586 memcpy(expert_output, staged_experts, row_outputs * sizeof(float));
587 memcpy(output, staged_output, output_elements * sizeof(float));
588 return 0;
589}
