26/* Portable reference projection. Products and the ascending reduction
27 * stay FP64. A FP32 intermediate is rounded to BF16 RNE on publication. */
28int ck_linear_rows_bf16_checked(
29 const uint16_t *input, size_t input_elements,
30 const uint16_t *weight, size_t weight_elements,
31 uint16_t *output, size_t output_elements,
32 uint16_t *staged, size_t staged_elements,
33 size_t rows, size_t input_width, size_t output_width)
34{
35 if (!rows || !input_width || !output_width ||
36 rows > SIZE_MAX / input_width || output_width > SIZE_MAX / input_width ||
37 rows > SIZE_MAX / output_width) return -2;
38 const size_t in_count = rows * input_width;
39 const size_t w_count = output_width * input_width;
40 const size_t out_count = rows * output_width;
41 if (input_elements < in_count || weight_elements < w_count ||
42 output_elements < out_count || staged_elements < out_count) return -2;
43 if (!region_valid(input, input_elements) || !region_valid(weight, weight_elements) ||
44 !region_valid(output, output_elements) || !region_valid(staged, staged_elements) ||
45 overlap(output, output_elements, input, input_elements) ||
46 overlap(output, output_elements, weight, weight_elements) ||
47 overlap(output, output_elements, staged, staged_elements) ||
48 overlap(staged, staged_elements, input, input_elements) ||
49 overlap(staged, staged_elements, weight, weight_elements)) return -1;
50 for (size_t i = 0; i < in_count; ++i)
51 if (!isfinite(bf16_to_float(input[i]))) return -1;
52 for (size_t i = 0; i < w_count; ++i)
53 if (!isfinite(bf16_to_float(weight[i]))) return -1;
54
55 for (size_t r = 0; r < rows; ++r) {
56 for (size_t o = 0; o < output_width; ++o) {
57 const uint16_t *src = input + r * input_width;
58 const uint16_t *w = weight + o * input_width;
59 double sum = 0.0;
60 for (size_t k = 0; k < input_width; ++k) {
61 volatile double product = (double)bf16_to_float(src[k]) * bf16_to_float(w[k]);
62 sum += product;
63 }
64 const uint16_t b = float_to_bf16((float)sum);
65 if (!isfinite(bf16_to_float(b))) return -1;
66 staged[r * output_width + o] = b;
67 }
68 }
69 for (size_t i = 0; i < out_count; ++i) output[i] = staged[i];
70 return 0;
71}
