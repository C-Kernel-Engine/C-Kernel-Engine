14static double dot_bias(const float *input, const float *weight,
15 size_t width, float bias)
16{
17 double sum = (double)bias;
18 for (size_t c = 0; c < width; ++c) {

31static void compute_four_rows(const float *input, size_t input_stride,
32 const float *weight, size_t weight_stride,
33 const float *bias, float *output, size_t output_stride,
34 size_t rows, size_t width, size_t channels)
35{
36 size_t r = 0;
37 for (; r + 4 <= rows; r += 4) {
38 for (size_t o = 0; o < channels; ++o) {
39 __m256d sum = _mm256_set1_pd(bias ? (double)bias[o] : 0.0);
40 for (size_t c = 0; c < width; ++c) {
41 __m256d values = _mm256_set_pd(input[(r+3)*input_stride+c],
42 input[(r+2)*input_stride+c], input[(r+1)*input_stride+c],
43 input[r*input_stride+c]);
44 __m256d w = _mm256_set1_pd((double)weight[o*weight_stride+c]);
45 sum = _mm256_add_pd(sum, _mm256_mul_pd(values, w));
46 }
47 double lanes[4];
48 _mm256_storeu_pd(lanes, sum);
49 for (size_t lane = 0; lane < 4; ++lane)
50 output[(r+lane)*output_stride+o] = (float)lanes[lane];
51 }
52 }
53 for (; r < rows; ++r)
54 for (size_t o = 0; o < channels; ++o)
55 output[r*output_stride+o] = (float)dot_bias(input+r*input_stride,
56 weight+o*weight_stride, width, bias ? bias[o] : 0.0f);
57}

60{

69int linear_rows_checked_f32(

79 !span_fits(rows, output_stride, output_channels, output_elements) ||

81 return -1;

83 /* Validate every accessed value before the first write. A conservative
84 * absolute bound permits one compute pass for ordinary model values.
85 * Values near the FP32 limit retain the exact two-pass staging path. */
86 double max_input = 0.0;
87 double max_weight_l1 = 0.0;
88 double max_bias = 0.0;
89 for (size_t r = 0; r < rows; ++r)
90 for (size_t c = 0; c < input_channels; ++c)
91 {
92 const float value = input[r * input_stride + c];
93 if (!isfinite(value)) return -1;
94 const double absolute = fabs((double)value);
95 if (absolute > max_input) max_input = absolute;
96 }
97 for (size_t o = 0; o < output_channels; ++o) {
98 if (bias && !isfinite(bias[o])) return -1;
99 if (bias && fabs((double)bias[o]) > max_bias)
100 max_bias = fabs((double)bias[o]);
101 double weight_l1 = 0.0;
102 for (size_t c = 0; c < input_channels; ++c)
103 {
104 const float value = weight[o * weight_stride + c];
105 if (!isfinite(value)) return -1;
106 weight_l1 += fabs((double)value);
107 }
108 if (weight_l1 > max_weight_l1) max_weight_l1 = weight_l1;
109 }
110 /* At most UINT32_MAX positive terms keep FP64 summation error far below
111 * the 4x guard, including error in the L1 bound. Every result therefore
112 * fits FP32 and publishing cannot fail after the first output write. */
113 if (input_channels <= (SIZE_MAX / 4) && input_channels <= UINT32_MAX &&
114 max_input * max_weight_l1 + max_bias <= (double)FLT_MAX / 4.0) {
115#if (defined(__x86_64__) || defined(__i386__)) && (defined(__GNUC__) || defined(__clang__))
116 __builtin_cpu_init();
117 if (rows >= 4 && __builtin_cpu_supports("avx2") &&
118 disjoint(output, output_elements, input, input_elements) &&
119 disjoint(output, output_elements, weight, weight_elements) &&
120 disjoint(output, output_elements, bias, bias_elements)) {
121 compute_four_rows(input, input_stride, weight, weight_stride, bias,
122 output, output_stride, rows, input_channels,
123 output_channels);
124 return 0;
125 }
126#endif
127 for (size_t r = 0; r < rows; ++r)
128 for (size_t o = 0; o < output_channels; ++o)
129 output[r * output_stride + o] = (float)dot_bias(
130 input + r * input_stride, weight + o * weight_stride,
131 input_channels, bias ? bias[o] : 0.0f);
132 return 0;
133 }
134 for (size_t r = 0; r < rows; ++r)
135 for (size_t o = 0; o < output_channels; ++o) {
136 const double result = dot_bias(input + r * input_stride,
137 weight + o * weight_stride,
138 input_channels, bias ? bias[o] : 0.0f);
139 if (!isfinite(result) || fabs(result) > FLT_MAX) return -1;
140 }
141 for (size_t r = 0; r < rows; ++r)
142 for (size_t o = 0; o < output_channels; ++o)
143 output[r * output_stride + o] = (float)dot_bias(
144 input + r * input_stride, weight + o * weight_stride,
145 input_channels, bias ? bias[o] : 0.0f);
146 return 0;
