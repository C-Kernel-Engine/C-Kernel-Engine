29/* Packed blocks are read in GGUF byte order and expanded inside the dot.
30 * The FP32 activation and ordered FP64 accumulation policy is shared. */
31static float expert_dot(const float *input, const unsigned char *weight,
32 size_t width, enum expert_storage storage, const unsigned char *activation)
33{
34 if (activation) return ck_q8_activation_dot(weight, activation, width);
35 if (storage == EXPERT_Q8_0) return ck_packed_q8_dot_ordered(input, weight, width);
36 double sum = 0.0;
37 if (storage == EXPERT_F32) {
38 for (size_t i = 0; i < width; ++i) {
39 float value;
40 memcpy(&value, weight + i * sizeof(float), sizeof(float));
41 sum += (double)input[i] * (double)value;
42 }
43 } else {
44 const size_t block_bytes = packed_block_bytes(storage);
45 for (size_t i = 0; i < width; i += QK8_0, weight += block_bytes) {
46 const float scale = ck_fp16_to_fp32_soft(
47 (uint16_t)(weight[0] | ((uint16_t)weight[1] << 8)));
48 for (size_t j = 0; j < QK8_0; ++j) {
49 const unsigned pair = weight[2 + j % 16];
50 const float expanded = scale * (float)iq4_nl_values[j < 16 ? pair & 15 : pair >> 4];
51 sum += (double)input[i + j] * (double)expanded;
52 }
53 }
54 }
55 return (float)sum;
56}

85static int project_expert(const struct routed_projection *p, const float *input,
86 const unsigned char *packed_hidden, size_t slot, float *scratch,
87 float *output)
88{
89 const size_t offset = slot * p->expert_bytes;
90 const unsigned char *gate = p->gate + offset, *up = p->up + offset, *down = p->down + offset;
91 float *gate_values = scratch, *up_values = gate_values + p->inner, *activation = up_values + p->inner;
92 unsigned char *packed_activation = p->quantize ? (unsigned char *)(activation + p->inner) + p->gate_row : NULL;
93
94 for (size_t i = 0; i < p->inner; ++i) {
95 if (packed_hidden) {
96 ck_q8_activation_dot_pair(gate + i * p->gate_row, up + i * p->gate_row, packed_hidden,
97 p->width, gate_values + i, up_values + i);
98 } else {
99 gate_values[i] = expert_dot(input, gate + i * p->gate_row, p->width, p->storage, NULL);
100 up_values[i] = expert_dot(input, up + i * p->gate_row, p->width, p->storage, NULL);
101 }
102 }
103 if (p->reference) {
104 for (size_t i = 0; i < p->inner; ++i) {
105 const float g = gate_values[i], u = up_values[i];
106 if (!isfinite(g) || !isfinite(u)) return -3;
107 const float capped_gate = p->gate_cap * tanhf(g * p->gate_reciprocal);
108 const float capped_up = p->up_cap * tanhf(u * p->up_reciprocal);
109 const float sigmoid = 1.0f / (1.0f + expf(-g));
110 activation[i] = (capped_gate * sigmoid) * capped_up;
111 if (!isfinite(activation[i])) return -3;
112 }
113 } else if (softcapped_swiglu_rows_checked_f32(gate_values, p->inner, up_values, p->inner,
114 activation, p->inner, p->inner, p->gate_cap, p->up_cap)) return -3;
115 if (packed_activation && ck_pack_q8_activation_rne(activation, packed_activation, p->inner)) return -3;
116 for (size_t i = 0; i < p->width; ++i) {
117 output[i] = expert_dot(activation, down + i * p->down_row, p->inner, p->storage, packed_activation);
118 if (!isfinite(output[i])) return -3;
119 }
120 return 0;
121}

123static inline int combine_expert(const float *expert, float *mixture, size_t width,
124 float weight, size_t index, int reference)
125{
126 for (size_t i = 0; i < width; ++i) {
127 if (reference) {
128 const volatile float weighted = expert[i] * weight;
129 mixture[i] = index == 0 ? weighted : mixture[i] + weighted;
130 } else {
131 mixture[i] += expert[i] * weight;
132 }
133 if (!isfinite(mixture[i])) return -3;
134 }
135 return 0;
136}

146static int routed_softcapped_experts(ROUTED_EXPERT_ARGS, enum expert_storage storage, int quantize, int reference_moe)
147{
148 /* Bounds keep workspace arithmetic valid on supported size_t widths. */
149 if (!rows || rows > 16 || !hidden_dim || hidden_dim > 8192 ||
150 !inner || inner > 8192 || !experts || experts > 1024 || !slots ||
151 slots > experts || !top_k || top_k > experts ||
152 !isfinite(gate_cap) || gate_cap <= 0.0f || !isfinite(up_cap) || up_cap <= 0.0f)
153 return -1;
154 const int packed = storage != EXPERT_F32;
155 if (packed && (hidden_dim % QK8_0 || inner % QK8_0)) return -1;
156 const size_t block_bytes = packed ? packed_block_bytes(storage) : sizeof(float);
157 const size_t block_elements = packed ? QK8_0 : 1;
158 const size_t gate_row = hidden_dim / block_elements * block_bytes;
159 const size_t down_row = inner / block_elements * block_bytes;
160 const size_t expert_bytes = inner * gate_row;
161 const size_t route_count = rows * top_k;
162 const size_t output_elements = rows * hidden_dim;
163 const size_t row_outputs = route_count * hidden_dim;
164 const size_t packed_elements = quantize ? (gate_row + down_row + sizeof(float) - 1) / sizeof(float) : 0;
165 const size_t row_scratch = 3 * inner + packed_elements;
166 const size_t staged_elements = row_outputs + output_elements;
167 size_t scratch_rows = 1;
168#ifdef _OPENMP
169 if (work_elements >= staged_elements + rows * row_scratch) scratch_rows = rows;
170#endif
171 size_t route_scratch = 0;
172#ifdef _OPENMP
173 if (quantize && rows == 1 && top_k > 1 && omp_get_max_threads() > 1 &&
174 work_elements >= staged_elements + top_k * row_scratch)
175 route_scratch = top_k;
176#endif
177 const size_t scratch_elements = staged_elements + (route_scratch ? route_scratch : scratch_rows) * row_scratch;
178 if (hidden_elements < output_elements || id_elements < route_count || route_elements < route_count ||
179 expert_slot_elements < experts || expert_output_elements < row_outputs ||
180 output_capacity < output_elements || work_elements < scratch_elements) return -1;
181 if (expert_bytes != hidden_dim * down_row || expert_bytes > SIZE_MAX / slots ||
182 matrix_bytes != expert_bytes * slots) return -1;
183 const void *buffers[] = {hidden, ids, route_weights, expert_slots, gate, up, down,
184 expert_output, output, work};
185 const size_t sizes[] = {output_elements * sizeof(float), route_count * sizeof(int32_t),
186 route_count * sizeof(float), experts * sizeof(int32_t),
187 matrix_bytes, matrix_bytes, matrix_bytes, row_outputs * sizeof(float),
188 output_elements * sizeof(float),
189 scratch_elements * sizeof(float)};
190 for (size_t i = 0; i < sizeof(buffers) / sizeof(buffers[0]); ++i) {
191 if (!region(buffers[i], sizes[i])) return -1;
192 if (i < 4 || i >= 7) {
193 if ((uintptr_t)buffers[i] % _Alignof(float)) return -1;
194 }
195 if (i < 7) continue;
196 for (size_t j = 0; j < i; ++j)
197 if (contains_overlap(buffers[i], sizes[i], buffers[j], sizes[j])) return -1;
198 }
199 if (!ck_checked_finite(hidden, output_elements) ||
200 !ck_checked_finite(route_weights, route_count)) return -2;
201
202 unsigned char assigned[1024] = {0};
203 for (size_t id = 0; id < experts; ++id) {
204 const int32_t slot = expert_slots[id];
205 if (slot == -1) continue;
206 if (slot < 0 || (size_t)slot >= slots || assigned[slot]) return -2;
207 assigned[slot] = 1;
208 }
209 for (size_t row = 0; row < rows; ++row) {
210 for (size_t k = 0; k < top_k; ++k) {
211 const size_t route = row * top_k + k;
212 const int32_t id = ids[route];
213 if (id < 0 || (size_t)id >= experts || expert_slots[id] < 0 ||
214 (size_t)expert_slots[id] >= slots || route_weights[route] < 0.0f) return -2;
215 for (size_t prev = 0; prev < k; ++prev)
216 if (id == ids[row * top_k + prev]) return -2;
217 }
218 }
219
220 const float gate_reciprocal = reference_moe ? 1.0f / gate_cap : 0.0f;
221 const float up_reciprocal = reference_moe ? 1.0f / up_cap : 0.0f;
222 if (!isfinite(gate_reciprocal) || !isfinite(up_reciprocal)) return -3;
223
224 float *staged_experts = work;
225 float *staged_output = staged_experts + row_outputs;
226 float *row_work = staged_output + output_elements;
227 memset(staged_output, 0, output_elements * sizeof(float));
228 const struct routed_projection projection = {
229 .gate = gate, .up = up, .down = down, .width = hidden_dim, .inner = inner,
230 .gate_row = gate_row, .down_row = down_row, .expert_bytes = expert_bytes,
231 .gate_cap = gate_cap, .up_cap = up_cap, .gate_reciprocal = gate_reciprocal, .up_reciprocal = up_reciprocal,
232 .storage = storage, .quantize = quantize, .reference = reference_moe,
233 };
234 int failed = 0;
235#ifdef _OPENMP
236 if (route_scratch) {
237 unsigned char *packed_hidden = (unsigned char *)(row_work + 3 * inner);
238 if (ck_pack_q8_activation_rne(hidden, packed_hidden, hidden_dim)) return -3;
239 /* Each task writes its declared slot. Mixing below preserves the
240 * original expert-ID or router-slot FP32 order. */
241 const int max_threads = omp_get_max_threads();
242 const int threads = (size_t)max_threads < top_k ? max_threads : (int)top_k;
243#pragma omp parallel for num_threads(threads) reduction(|:failed)
244 for (size_t route = 0; route < top_k; ++route) {
245 failed |= project_expert(&projection, hidden, packed_hidden, expert_slots[ids[route]],
246 row_work + route * row_scratch, staged_experts + route * hidden_dim) != 0;
247 }
248 if (failed) return -3;
249 size_t ordered_slots[1024];
250 if (!reference_moe) order_slots(ids, experts, top_k, ordered_slots);
251 for (size_t index = 0; index < top_k; ++index) {
252 const size_t route = reference_moe ? index : ordered_slots[index];
253 if (combine_expert(staged_experts + route * hidden_dim, staged_output, hidden_dim,
254 route_weights[route], index, reference_moe)) return -3;
255 }
256 } else
257#endif
258 {
259#ifdef _OPENMP
260#pragma omp parallel for if(scratch_rows > 1) reduction(|:failed)
261#endif
262 for (size_t row = 0; row < rows; ++row) {
263 float *scratch = row_work + (scratch_rows > 1 ? row * row_scratch : 0);
264 unsigned char *packed_hidden = quantize ? (unsigned char *)(scratch + 3 * inner) : NULL;
265 const float *input = hidden + row * hidden_dim;
266 if (packed_hidden && ck_pack_q8_activation_rne(input, packed_hidden, hidden_dim)) {
267 failed = 1;
268 continue;
269 }
270 size_t ordered_slots[1024];
271 if (!reference_moe) order_slots(ids + row * top_k, experts, top_k, ordered_slots);
272 for (size_t index = 0; index < top_k; ++index) {
273 const size_t route = row * top_k + (reference_moe ? index : ordered_slots[index]);
274 float *expert = staged_experts + route * hidden_dim;
275 if (project_expert(&projection, input, packed_hidden, expert_slots[ids[route]], scratch, expert) ||
276 combine_expert(expert, staged_output + row * hidden_dim, hidden_dim,
277 route_weights[route], index, reference_moe)) {
278 failed = 1;
279 break;
280 }
281 }
282 }
283 if (failed) return -3;
284 }
285 memcpy(expert_output, staged_experts, row_outputs * sizeof(float));
286 memcpy(output, staged_output, output_elements * sizeof(float));
287 return 0;
288}
