13static double rms_scale(const float *x, size_t width, double epsilon)
14{
15 double sum = 0.0;
16 for (size_t c = 0; c < width; ++c) sum += (double)x[c] * x[c];

27static void mix_depth_rows(size_t begin, size_t end, void *opaque)
28{
29 const struct depth_mix_rows *work = opaque;
30 const float *bank = work->bank, *current = work->current, *query = work->query;
31 float *output = work->output;
32 const size_t states = work->states, width = work->width;
33 const float epsilon = work->epsilon;
34 const double temperature = 1.0 / sqrt((double)width);
35 for (size_t r = begin; r < end; ++r) {
36 double scale[65], logits[65], max_logit = -INFINITY;
37 for (size_t s = 0; s <= states; ++s) {
38 const float *x = s == states ? current + r * width : bank + (r * states + s) * width;
39 scale[s] = rms_scale(x, width, epsilon);
40 double dot = 0.0;
41 for (size_t c = 0; c < width; ++c) {
42 const float normalized = (float)(x[c] * scale[s]);
43 dot += (double)query[c] * normalized;
44 }
45 logits[s] = dot * temperature;
46 max_logit = fmax(max_logit, logits[s]);
47 }
48 double denominator = 0.0;
49 for (size_t s = 0; s <= states; ++s) {
50 logits[s] = exp(logits[s] - max_logit);
51 denominator += logits[s];
52 }
53 for (size_t c = 0; c < width; ++c) {
54 double sum = 0.0;
55 for (size_t s = 0; s <= states; ++s) {
56 const float *x = s == states ? current + r * width : bank + (r * states + s) * width;
57 const float normalized = (float)(x[c] * scale[s]);
58 sum += logits[s] * normalized;
59 }
60 output[r * width + c] = (float)(sum / denominator);
61 }
62 }
63}
